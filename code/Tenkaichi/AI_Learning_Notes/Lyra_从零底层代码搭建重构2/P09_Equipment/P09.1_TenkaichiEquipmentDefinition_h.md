// 版权声明（Lyra 项目自带，保留）

#pragma once

#include "Templates/SubclassOf.h"

#include "TenkaichiEquipmentDefinition.generated.h"

class AActor;
class UTenkaichiAbilitySet;
class UTenkaichiEquipmentInstance;

// 装备要生成的 Actor 描述：生成什么、挂哪个插槽、什么偏移
USTRUCT()
struct FTenkaichiEquipmentActorToSpawn
{
	GENERATED_BODY()

	// 默认构造（空）
	FTenkaichiEquipmentActorToSpawn()
	{}

	// 要生成的 Actor 类
	UPROPERTY(EditAnywhere, Category=Equipment)
	TSubclassOf<AActor> ActorToSpawn;

	// 挂到角色身上的哪个插槽（如 "hand_r"）
	UPROPERTY(EditAnywhere, Category=Equipment)
	FName AttachSocket;

	// 挂载的偏移（位置/旋转/缩放）
	UPROPERTY(EditAnywhere, Category=Equipment)
	FTransform AttachTransform;
};


/**
 * UTenkaichiEquipmentDefinition
 *
 * 一件"可以穿到角色身上"的装备的定义
 */
UCLASS(Blueprintable, Const, Abstract, BlueprintType)
class UTenkaichiEquipmentDefinition : public UObject
{
	GENERATED_BODY()

public:
	// 构造函数（ObjectInitializer 有默认值，可无参构造）
	UTenkaichiEquipmentDefinition(const FObjectInitializer& ObjectInitializer = FObjectInitializer::Get());

	// 用什么实例类来承载这件装备（默认 UTenkaichiEquipmentInstance）
	UPROPERTY(EditDefaultsOnly, Category=Equipment)
	TSubclassOf<UTenkaichiEquipmentInstance> InstanceType;

	// 穿上这件装备时要授予的技能集（GAS AbilitySet）
	UPROPERTY(EditDefaultsOnly, Category=Equipment)
	TArray<TObjectPtr<const UTenkaichiAbilitySet>> AbilitySetsToGrant;

	// 穿上时要在角色身上生成的 Actor（如武器模型）
	UPROPERTY(EditDefaultsOnly, Category=Equipment)
	TArray<FTenkaichiEquipmentActorToSpawn> ActorsToSpawn;
};

## 三、逐段讲解

### 1. include 与前置声明

```cpp
#include "Templates/SubclassOf.h"
```

- `Templates/SubclassOf.h`：`TSubclassOf`（类型安全的类引用模板）。因为 `.h` 里多个字段用 `TSubclassOf<...>`，需要完整定义。

```cpp
class AActor;
class UTenkaichiAbilitySet;
class UTenkaichiEquipmentInstance;
```

- 三个**前置声明**：`AActor`（引擎）、`UTenkaichiAbilitySet`（P05 已教）、`UTenkaichiEquipmentInstance`（P09.2 待教）。
- **为什么前置声明够**：它们都只以 `TSubclassOf<...>` 或 `TObjectPtr<...>` 出现（不实例化、不继承），编译期只需类型名。

### 2. 结构体：`FTenkaichiEquipmentActorToSpawn`

```cpp
USTRUCT()
struct FTenkaichiEquipmentActorToSpawn
{
	GENERATED_BODY()
	FTenkaichiEquipmentActorToSpawn()
	{}
	UPROPERTY(EditAnywhere, Category=Equipment)
	TSubclassOf<AActor> ActorToSpawn;
	UPROPERTY(EditAnywhere, Category=Equipment)
	FName AttachSocket;
	UPROPERTY(EditAnywhere, Category=Equipment)
	FTransform AttachTransform;
};
```

- "要生成一个 Actor"的完整描述：`ActorToSpawn`（什么类）+ `AttachSocket`（挂哪个插槽）+ `AttachTransform`（偏移）。
- `USTRUCT()`：**没有 `BlueprintType`**（Lyra 源码原样）——这个结构体不直接暴露给蓝图，是 C++ 内部数据结构。
- 注意：`AttachTransform` 是 `FTransform`（完整类型），但这里**没有 include** `FTransform` 的头——因为它作为 `UPROPERTY` 成员，UE 反射系统能处理；实际 `FTransform` 定义由引擎的其他头（通过 UHT 生成的 `.generated.h`）间接提供。这是 Lyra 原样写法，照抄即可。

### 3. 类：`UTenkaichiEquipmentDefinition`

```cpp
UCLASS(Blueprintable, Const, Abstract, BlueprintType)
class UTenkaichiEquipmentDefinition : public UObject
{
	GENERATED_BODY()
public:
	UTenkaichiEquipmentDefinition(const FObjectInitializer& ObjectInitializer = FObjectInitializer::Get());
	UPROPERTY(EditDefaultsOnly, Category=Equipment)
	TSubclassOf<UTenkaichiEquipmentInstance> InstanceType;
	UPROPERTY(EditDefaultsOnly, Category=Equipment)
	TArray<TObjectPtr<const UTenkaichiAbilitySet>> AbilitySetsToGrant;
	UPROPERTY(EditDefaultsOnly, Category=Equipment)
	TArray<FTenkaichiEquipmentActorToSpawn> ActorsToSpawn;
};
```

- 继承 `UObject`（最基础的对象基类，不是 Actor/组件）。
- **三个类标记**（总 md 已讲）：
  - `Blueprintable`：可建蓝图子类。
  - `Const`：实例只读（运行时不可改）。
  - `Abstract`：抽象类，必须建蓝图子类用。
- **三个字段**（都是 `EditDefaultsOnly`，只在类默认值里配）：
  - `InstanceType`：`TSubclassOf<UTenkaichiEquipmentInstance>` —— 用哪个实例类承载。
  - `AbilitySetsToGrant`：`TArray<TObjectPtr<const UTenkaichiAbilitySet>>` —— 授予的技能集（`const` 说明是只读引用，不拥有）。
  - `ActorsToSpawn`：`TArray<FTenkaichiEquipmentActorToSpawn>` —— 生成的 Actor 列表。
