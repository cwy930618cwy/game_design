```cpp
// 版权声明（Lyra 项目自带，保留）

#include "InventoryFragment_SetStats.h"

#include "Inventory/TenkaichiInventoryItemInstance.h"

#include UE_INLINE_GENERATED_CPP_BY_NAME(InventoryFragment_SetStats)

// 覆写钩子：物品实例创建时，把 InitialItemStats 表里每个 Tag 灌进实例的 StatTags
void UInventoryFragment_SetStats::OnInstanceCreated(UTenkaichiInventoryItemInstance* Instance) const
{
	Super::OnInstanceCreated(Instance);

	for (const auto& KVP : InitialItemStats)
	{
		Instance->AddStatTagStack(KVP.Key, KVP.Value);
	}
}

// 按 Tag 查初始属性表里的值（查不到返回 0）
int32 UInventoryFragment_SetStats::GetItemStatByTag(FGameplayTag Tag) const
{
	const int32* FoundStat = InitialItemStats.Find(Tag);
	return (FoundStat != nullptr) ? *FoundStat : 0;
}
```

## 三、逐段讲解

### 1. include

```cpp
#include "InventoryFragment_SetStats.h"
#include "Gameplay/TenkaichiInventoryItemInstance.h"
```

- 第一个是它自己的 `.h`。
- `Gameplay/TenkaichiInventoryItemInstance.h`：`.h` 里只是前置声明，`.cpp` 里要调 `Instance->AddStatTagStack`，需要**完整定义**，所以 include。（⚠️ 注意实际路径是 `Inventory/TenkaichiInventoryItemInstance.h`，与 P08.2 一致）

```cpp
#include UE_INLINE_GENERATED_CPP_BY_NAME(InventoryFragment_SetStats)
```

- UE5 固定宏（P04 讲过），参数 = 类名（无前缀时就是 `InventoryFragment_SetStats`）。

### 2. 覆写钩子：灌初始属性

```cpp
void UInventoryFragment_SetStats::OnInstanceCreated(UTenkaichiInventoryItemInstance* Instance) const
{
	Super::OnInstanceCreated(Instance);

	for (const auto& KVP : InitialItemStats)
	{
		Instance->AddStatTagStack(KVP.Key, KVP.Value);
	}
}
```

- `Super::OnInstanceCreated(Instance)`：先调基类钩子（P08.1 空实现，但保留调用，符合 C++ 链式虚函数习惯）。
- `for (const auto& KVP : InitialItemStats)`：遍历初始属性表（`TMap` 的元素是 `TPair<Tag, int32>`，`const auto&` 避免拷贝）。
- `Instance->AddStatTagStack(KVP.Key, KVP.Value)`：**关键调用**——把 `Key`（Tag）加 `Value`（数量）灌进实例的 `StatTags`（P08.2 实现的薄转发方法）。**这就是初始属性的"注入点"**。

### 3. 查询函数

```cpp
int32 UInventoryFragment_SetStats::GetItemStatByTag(FGameplayTag Tag) const
{
	const int32* ValueStat = InitialItemStats.Find(Tag);
	return (ValueStat != nullptr) ? *ValueStat : 0;
}
```

- `InitialItemStats.Find(Tag)`：在 `TMap` 里找 `Tag`，**找不到返回 nullptr，找到返回指针**。
- 三目运算：指针非空则解引用取值，否则返回 0。
