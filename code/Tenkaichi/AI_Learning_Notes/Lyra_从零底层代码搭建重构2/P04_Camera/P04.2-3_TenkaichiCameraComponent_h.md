#pragma once

#include "Camera/CameraComponent.h"
#include "GameFramework/Actor.h"

#include "TenkaichiCameraComponent.generated.h"

class UCanvas;
class UTenkaichiCameraMode;
class UTenkaichiCameraModeStack;
class UObject;
struct FFrame;
struct FGameplayTag;
struct FMinimalViewInfo;
template <class TClass> class TSubclassOf;

DECLARE_DELEGATE_RetVal(TSubclassOf<UTenkaichiCameraMode>, FTenkaichiCameraModeDelegate);


/**
 * UTenkaichiCameraComponent
 *
 *	本项目使用的基础相机组件类。
 */
UCLASS()
class TENKAICHIGAME_API UTenkaichiCameraComponent : public UCameraComponent
{
	GENERATED_BODY()

public:

	// 构造函数。
	UTenkaichiCameraComponent(const FObjectInitializer& ObjectInitializer);

	// 如果指定 Actor 上存在相机组件，返回它。
	UFUNCTION(BlueprintPure, Category = "Tenkaichi|Camera")
	static UTenkaichiCameraComponent* FindCameraComponent(const AActor* Actor) { return (Actor ? Actor->FindComponentByClass<UTenkaichiCameraComponent>() : nullptr); }

	// 返回相机正在观察的目标 Actor。
	virtual AActor* GetTargetActor() const { return GetOwner(); }

	// 用来查询"当前最佳相机模式"的委托。
	FTenkaichiCameraModeDelegate DetermineCameraModeDelegate;

	// 给视野加一个偏移。偏移只生效一帧，应用后会被清掉。
	void AddFieldOfViewOffset(float FovOffset) { FieldOfViewOffset += FovOffset; }

	// 绘制调试信息（showdebug 相机）。
	virtual void DrawDebug(UCanvas* Canvas) const;

	// 获取最顶层模式对应的标签及其混合权重。
	void GetBlendInfo(float& OutWeightOfTopLayer, FGameplayTag& OutTagOfTopLayer) const;

protected:

	// 组件注册到 Actor 时调用：创建相机模式栈。
	virtual void OnRegister() override;

	// 核心方法，覆盖引擎：每帧求最终相机视图并同步给控制器和相机组件。
	virtual void GetCameraView(float DeltaTime, FMinimalViewInfo& DesiredView) override;

	// 通过委托查询当前应该使用的相机模式，并压入相机模式栈。
	virtual void UpdateCameraModes();

protected:

	// 用来混合相机模式的栈。
	UPROPERTY()
	TObjectPtr<UTenkaichiCameraModeStack> CameraModeStack;

	// 应用到视野的偏移。偏移只生效一帧，应用后会被清掉。
	float FieldOfViewOffset;

};
```

```

| 行 | 干什么 | 归属 |
|----|--------|------|
| `Camera/CameraComponent.h` | 引擎相机组件基类 `UCameraComponent` | ✅ 引擎 |
| `GameFramework/Actor.h` | 引擎 `AActor`（`FindCameraComponent` 参数用） | ✅ 引擎 |
| `TenkaichiCameraComponent.generated.h` | UHT 生成头（**必须最后 include**） | 生成 |
| `class UCanvas;` 等前置声明 | 避免 include 对应头文件（减少编译耦合），只声明指针/引用用 | ✅ 引擎 |
| `DECLARE_DELEGATE_RetVal(...)` | 声明一个**带返回值**的委托：外部绑定它，返回"当前该用哪个相机模式类型"（`TSubclassOf<UTenkaichiCameraMode>`） | 宏 |

> **关键：`DECLARE_DELEGATE_RetVal`**——普通委托没有返回值（`DECLARE_DELEGATE`），这个 `_RetVal` 变体让委托回调**能返回一个 `TSubclassOf<UTenkaichiCameraMode>`**。因为相机组件不自己决定用哪个模式，而是**问外部**："现在应该用哪种模式？"外部（通常是角色 Pawn）绑一个 lambda/函数，返回类型。这实现了"模式决定逻辑外置"，相机组件只负责执行。
