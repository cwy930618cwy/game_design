void FTenkaichiCameraModeView::Blend(const FTenkaichiCameraModeView& Other, float OtherWeight)
{
	if (OtherWeight <= 0.0f)
	{
		return;
	}
	else if (OtherWeight >= 1.0f)
	{
		*this = Other;
		return;
	}

	Location = FMath::Lerp(Location, Other.Location, OtherWeight);

	const FRotator DeltaRotation = (Other.Rotation - Rotation).GetNormalized();
	Rotation = Rotation + (OtherWeight * DeltaRotation);

	const FRotator DeltaControlRotation = (Other.ControlRotation - ControlRotation).GetNormalized();
	ControlRotation = ControlRotation + (OtherWeight * DeltaControlRotation);

	FieldOfView = FMath::Lerp(FieldOfView, Other.FieldOfView, OtherWeight);
}

FVector UTenkaichiCameraMode::GetPivotLocation() const
{
	const AActor* TargetActor = GetTargetActor();
	check(TargetActor);

	if (const APawn* TargetPawn = Cast<APawn>(TargetActor))
	{
		// 针对角色做高度调整，以考虑蹲伏的情况。
		if (const ACharacter* TargetCharacter = Cast<ACharacter>(TargetPawn))
		{
			const ACharacter* TargetCharacterCDO = TargetCharacter->GetClass()->GetDefaultObject<ACharacter>();
			check(TargetCharacterCDO);

			const UCapsuleComponent* CapsuleComp = TargetCharacter->GetCapsuleComponent();
			check(CapsuleComp);

			const UCapsuleComponent* CapsuleCompCDO = TargetCharacterCDO->GetCapsuleComponent();
			check(CapsuleCompCDO);

			const float DefaultHalfHeight = CapsuleCompCDO->GetUnscaledCapsuleHalfHeight();
			const float ActualHalfHeight = CapsuleComp->GetUnscaledCapsuleHalfHeight();
			const float HeightAdjustment = (DefaultHalfHeight - ActualHalfHeight) + TargetCharacterCDO->BaseEyeHeight;

			return TargetCharacter->GetActorLocation() + (FVector::UpVector * HeightAdjustment);
		}

		return TargetPawn->GetPawnViewLocation();
	}

	return TargetActor->GetActorLocation();
}

FRotator UTenkaichiCameraMode::GetPivotRotation() const
{
	const AActor* TargetActor = GetTargetActor();
	check(TargetActor);

	if (const APawn* TargetPawn = Cast<APawn>(TargetActor))
	{
		return TargetPawn->GetViewRotation();
	}

	return TargetActor->GetActorRotation();
}

void UTenkaichiCameraMode::UpdateCameraMode(float DeltaTime)
{
	UpdateView(DeltaTime);
	UpdateBlending(DeltaTime);
}

void UTenkaichiCameraMode::UpdateView(float DeltaTime)
{
	FVector PivotLocation = GetPivotLocation();
	FRotator PivotRotation = GetPivotRotation();

	PivotRotation.Pitch = FMath::ClampAngle(PivotRotation.Pitch, ViewPitchMin, ViewPitchMax);

	View.Location = PivotLocation;
	View.Rotation = PivotRotation;
	View.ControlRotation = View.Rotation;
	View.FieldOfView = FieldOfView;
}

void UTenkaichiCameraMode::SetBlendWeight(float Weight)
{
	BlendWeight = FMath::Clamp(Weight, 0.0f, 1.0f);

	// 因为我们直接设置了混合权重，所以需要反算出混合 alpha，以抵消混合函数的影响。
	const float InvExponent = (BlendExponent > 0.0f) ? (1.0f / BlendExponent) : 1.0f;

	switch (BlendFunction)
	{
	case ETenkaichiCameraModeBlendFunction::Linear:
		BlendAlpha = BlendWeight;
		break;

	case ETenkaichiCameraModeBlendFunction::EaseIn:
		BlendAlpha = FMath::InterpEaseIn(0.0f, 1.0f, BlendWeight, InvExponent);
		break;

	case ETenkaichiCameraModeBlendFunction::EaseOut:
		BlendAlpha = FMath::InterpEaseOut(0.0f, 1.0f, BlendWeight, InvExponent);
		break;

	case ETenkaichiCameraModeBlendFunction::EaseInOut:
		BlendAlpha = FMath::InterpEaseInOut(0.0f, 1.0f, BlendWeight, InvExponent);
		break;

	default:
		checkf(false, TEXT("SetBlendWeight: Invalid BlendFunction [%d]\n"), (uint8)BlendFunction);
		break;
	}
}

void UTenkaichiCameraMode::UpdateBlending(float DeltaTime)
{
	if (BlendTime > 0.0f)
	{
		BlendAlpha += (DeltaTime / BlendTime);
		BlendAlpha = FMath::Min(BlendAlpha, 1.0f);
	}
	else
	{
		BlendAlpha = 1.0f;
	}

	const float Exponent = (BlendExponent > 0.0f) ? BlendExponent : 1.0f;

	switch (BlendFunction)
	{
	case ETenkaichiCameraModeBlendFunction::Linear:
		BlendWeight = BlendAlpha;
		break;

	case ETenkaichiCameraModeBlendFunction::EaseIn:
		BlendWeight = FMath::InterpEaseIn(0.0f, 1.0f, BlendAlpha, Exponent);
		break;

	case ETenkaichiCameraModeBlendFunction::EaseOut:
		BlendWeight = FMath::InterpEaseOut(0.0f, 1.0f, BlendAlpha, Exponent);
		break;

	case ETenkaichiCameraModeBlendFunction::EaseInOut:
		BlendWeight = FMath::InterpEaseInOut(0.0f, 1.0f, BlendAlpha, Exponent);
		break;

	default:
		checkf(false, TEXT("UpdateBlending: Invalid BlendFunction [%d]\n"), (uint8)BlendFunction);
		break;
	}
}

void UTenkaichiCameraModeStack::PushCameraMode(TSubclassOf<UTenkaichiCameraMode> CameraModeClass)
{
	if (!CameraModeClass)
	{
		return;
	}

	UTenkaichiCameraMode* CameraMode = GetCameraModeInstance(CameraModeClass);
	check(CameraMode);

	int32 StackSize = CameraModeStack.Num();

	if ((StackSize > 0) && (CameraModeStack[0] == CameraMode))
	{
		// 已经在栈顶了。
		return;
	}

	// 看看它是否已经在栈里，如果在就把它移除。
	// 计算它原本对栈的贡献有多大。
	int32 ExistingStackIndex = INDEX_NONE;
	float ExistingStackContribution = 1.0f;

	for (int32 StackIndex = 0; StackIndex < StackSize; ++StackIndex)
	{
		if (CameraModeStack[StackIndex] == CameraMode)
		{
			ExistingStackIndex = StackIndex;
			ExistingStackContribution *= CameraMode->GetBlendWeight();
			break;
		}
		else
		{
			ExistingStackContribution *= (1.0f - CameraModeStack[StackIndex]->GetBlendWeight());
		}
	}

	if (ExistingStackIndex != INDEX_NONE)
	{
		CameraModeStack.RemoveAt(ExistingStackIndex);
		StackSize--;
	}
	else
	{
		ExistingStackContribution = 0.0f;
	}

	// 决定初始权重。
	const bool bShouldBlend = ((CameraMode->GetBlendTime() > 0.0f) && (StackSize > 0));
	const float BlendWeight = (bShouldBlend ? ExistingStackContribution : 1.0f);

	CameraMode->SetBlendWeight(BlendWeight);

	// 把新条目加到栈顶。
	CameraModeStack.Insert(CameraMode, 0);

	// 确保栈底始终是 100% 权重。
	CameraModeStack.Last()->SetBlendWeight(1.0f);

	// 如果它是新加到栈里的，通知它被激活。
	if (ExistingStackIndex == INDEX_NONE)
	{
		CameraMode->OnActivation();
	}
}

bool UTenkaichiCameraModeStack::EvaluateStack(float DeltaTime, FTenkaichiCameraModeView& OutCameraModeView)
{
	if (!bIsActive)
	{
		return false;
	}

	UpdateStack(DeltaTime);
	BlendStack(OutCameraModeView);

	return true;
}

UTenkaichiCameraMode* UTenkaichiCameraModeStack::GetCameraModeInstance(TSubclassOf<UTenkaichiCameraMode> CameraModeClass)
{
	check(CameraModeClass);

	// 先看看是不是已经创建过一个。
	for (UTenkaichiCameraMode* CameraMode : CameraModeInstances)
	{
		if ((CameraMode != nullptr) && (CameraMode->GetClass() == CameraModeClass))
		{
			return CameraMode;
		}
	}

	// 没找到，就创建一个新的。
	UTenkaichiCameraMode* NewCameraMode = NewObject<UTenkaichiCameraMode>(GetOuter(), CameraModeClass, NAME_None, RF_NoFlags);
	check(NewCameraMode);

	CameraModeInstances.Add(NewCameraMode);

	return NewCameraMode;
}

void UTenkaichiCameraModeStack::UpdateStack(float DeltaTime)
{
	const int32 StackSize = CameraModeStack.Num();
	if (StackSize <= 0)
	{
		return;
	}

	int32 RemoveCount = 0;
	int32 RemoveIndex = INDEX_NONE;

	for (int32 StackIndex = 0; StackIndex < StackSize; ++StackIndex)
	{
		UTenkaichiCameraMode* CameraMode = CameraModeStack[StackIndex];
		check(CameraMode);

		CameraMode->UpdateCameraMode(DeltaTime);

		if (CameraMode->GetBlendWeight() >= 1.0f)
		{
			// 这个模式下面的所有模式都不再需要了，可以移除。
			RemoveIndex = (StackIndex + 1);
			RemoveCount = (StackSize - RemoveIndex);
			break;
		}
	}

	if (RemoveCount > 0)
	{
		// 通知被移除的模式它们被移出栈了。
		for (int32 StackIndex = RemoveIndex; StackIndex < StackSize; ++StackIndex)
		{
			UTenkaichiCameraMode* CameraMode = CameraModeStack[StackIndex];
			check(CameraMode);

			CameraMode->OnDeactivation();
		}

		CameraModeStack.RemoveAt(RemoveIndex, RemoveCount);
	}
}

void UTenkaichiCameraModeStack::BlendStack(FTenkaichiCameraModeView& OutCameraModeView) const
{
	const int32 StackSize = CameraModeStack.Num();
	if (StackSize <= 0)
	{
		return;
	}

	// 从栈底开始，一层层往上混合。
	const UTenkaichiCameraMode* CameraMode = CameraModeStack[StackSize - 1];
	check(CameraMode);

	OutCameraModeView = CameraMode->GetCameraModeView();

	for (int32 StackIndex = (StackSize - 2); StackIndex >= 0; --StackIndex)
	{
		CameraMode = CameraModeStack[StackIndex];
		check(CameraMode);

		OutCameraModeView.Blend(CameraMode->GetCameraModeView(), CameraMode->GetBlendWeight());
	}
}

void UTenkaichiCameraModeStack::DrawDebug(UCanvas* Canvas) const
{
	check(Canvas);

	FDisplayDebugManager& DisplayDebugManager = Canvas->DisplayDebugManager;

	DisplayDebugManager.SetDrawColor(FColor::Green);
	DisplayDebugManager.DrawString(FString(TEXT("   --- Camera Modes (Begin) ---")));

	for (const UTenkaichiCameraMode* CameraMode : CameraModeStack)
	{
		check(CameraMode);
		CameraMode->DrawDebug(Canvas);
	}

	DisplayDebugManager.SetDrawColor(FColor::Green);
	DisplayDebugManager.DrawString(FString::Printf(TEXT("   --- Camera Modes (End) ---")));
}

void UTenkaichiCameraModeStack::GetBlendInfo(float& OutWeightOfTopLayer, FGameplayTag& OutTagOfTopLayer) const
{
	if (CameraModeStack.Num() == 0)
	{
		OutWeightOfTopLayer = 1.0f;
		OutTagOfTopLayer = FGameplayTag();
		return;
	}
	else
	{
		UTenkaichiCameraMode* TopEntry = CameraModeStack.Last();
		check(TopEntry);
		OutWeightOfTopLayer = TopEntry->GetBlendWeight();
		OutTagOfTopLayer = TopEntry->GetCameraTypeTag();
	}
}

	, ControlRotation(ForceInit)

void FTenkaichiCameraModeView::Blend(const FTenkaichiCameraModeView& Other, float OtherWeight)
{
	if (OtherWeight <= 0.0f) { return; }
	else if (OtherWeight >= 1.0f) { *this = Other; return; }

	Location = FMath::Lerp(Location, Other.Location, OtherWeight);
	const FRotator DeltaRotation = (Other.Rotation - Rotation).GetNormalized();
	Rotation = Rotation + (OtherWeight * DeltaRotation);
	const FRotator DeltaControlRotation = (Other.ControlRotation - ControlRotation).GetNormalized();
	ControlRotation = ControlRotation + (OtherWeight * DeltaControlRotation);
	FieldOfView = FMath::Lerp(FieldOfView, Other.FieldOfView, OtherWeight);
}
```

| 段 | 干什么 |
|----|--------|
| `FieldOfView(TENKAICHI_CAMERA_DEFAULT_FOV)` | 初始 FOV 用宏默认值（80°） |
| `OtherWeight <= 0.0f` | 对方权重为 0 → 完全保持自己 |
| `OtherWeight >= 1.0f` | 对方权重为 1 → 完全用对方（`*this = Other`） |
| `FMath::Lerp` | 线性插值（位置/FOV 用） |
| `Rotation + DeltaRotation` | 旋转用角度差插值（避免方向性问题），`.GetNormalized()` 归一化到 -180~180 |

> 类比：**"两张快照叠成一张"**——权重 0 全看自己，权重 1 全看对方，中间按权重插值。旋转用"最短角度差"插值，防止绕远路。

void UTenkaichiCameraMode::SetBlendWeight(float Weight)
{
	BlendWeight = FMath::Clamp(Weight, 0.0f, 1.0f);
	// 因为我们直接设置了混合权重，所以需要反算出混合 alpha，以抵消混合函数的影响。
	const float InvExponent = (BlendExponent > 0.0f) ? (1.0f / BlendExponent) : 1.0f;
	switch (BlendFunction) { ... BlendAlpha = ...; }
}

void UTenkaichiCameraMode::UpdateBlending(float DeltaTime)
{
	if (BlendTime > 0.0f)
	{
		BlendAlpha += (DeltaTime / BlendTime);
		BlendAlpha = FMath::Min(BlendAlpha, 1.0f);
	}
	else { BlendAlpha = 1.0f; }
	const float Exponent = (BlendExponent > 0.0f) ? BlendExponent : 1.0f;
	switch (BlendFunction) { ... BlendWeight = ...; }
}
```

| 段 | 干什么 |
|----|--------|
| `BlendWeight` | 实际用于混合的权重（0~1） |
| `BlendAlpha` | 线性进度（时间累积：`+= DeltaTime / BlendTime`） |
| `SetBlendWeight` | **外部设权重**（入栈时）：用**倒数指数**反推 alpha |
| `UpdateBlending` | **时间推进**（每帧）：alpha 随时间推进，再用指数曲线算出权重 |
| `FMath::InterpEaseIn/Out/InOut` | 引擎缓动函数，指数控制曲线形状 |

> 类比：**"渐入渐出的过渡"**——`UpdateBlending` 每帧把 alpha 从 0 推到 1（`DeltaTime / BlendTime`），再用选定的曲线（Linear/Ease 等）把 alpha 映射成权重。两个相机模式切换时，一个淡出、一个淡入。

### 1.9 PushCameraMode（把模式压入栈）

```cpp
void UTenkaichiCameraModeStack::PushCameraMode(TSubclassOf<UTenkaichiCameraMode> CameraModeClass)
{
	if (!CameraModeClass) { return; }
	UTenkaichiCameraMode* CameraMode = GetCameraModeInstance(CameraModeClass);
	check(CameraMode);
	int32 StackSize = CameraModeStack.Num();
	if ((StackSize > 0) && (CameraModeStack[0] == CameraMode)) { return; } // 已在栈顶

	int32 ExistingStackIndex = INDEX_NONE;
	float ExistingStackContribution = 1.0f;
	for (int32 StackIndex = 0; StackIndex < StackSize; ++StackIndex)
	{
		if (CameraModeStack[StackIndex] == CameraMode)
		{
			ExistingStackIndex = StackIndex;
			ExistingStackContribution *= CameraMode->GetBlendWeight();
			break;
		}
		else
		{
			ExistingStackContribution *= (1.0f - CameraModeStack[StackIndex]->GetBlendWeight());
		}
	}
	if (ExistingStackIndex != INDEX_NONE) { CameraModeStack.RemoveAt(ExistingStackIndex); StackSize--; }
	else { ExistingStackContribution = 0.0f; }

	const bool bShouldBlend = ((CameraMode->GetBlendTime() > 0.0f) && (StackSize > 0));
	const float BlendWeight = (bShouldBlend ? ExistingStackContribution : 1.0f);
	CameraMode->SetBlendWeight(BlendWeight);
	CameraModeStack.Insert(CameraMode, 0);
	CameraModeStack.Last()->SetBlendWeight(1.0f);
	if (ExistingStackIndex == INDEX_NONE) { CameraMode->OnActivation(); }
}
```

| 段 | 干什么 |
|----|--------|
| 已在栈顶？ | 直接返回（不重复压） |
| `ExistingStackContribution` | 如果模式已在栈中，算出它原本的贡献权重，移除后用它做初始权重（平滑续接） |
| `bShouldBlend` | 有过渡时间且栈非空 → 从旧权重开始淡入；否则直接 100% |
| `Insert(0)` | 新模式插到栈顶（栈顶 = 最高优先级） |
| `Last()->SetBlendWeight(1.0f)` | 栈底永远 100% 权重 |
| `OnActivation` | 新入栈的模式收到激活回调 |

> 类比：**"切镜头时，新镜头从旧镜头的权重接管"**——避免瞬跳，让新镜头从旧镜头的混合状态平滑切入。

### 1.10 EvaluateStack / GetCameraModeInstance / UpdateStack / BlendStack / GetBlendInfo

```cpp
bool UTenkaichiCameraModeStack::EvaluateStack(float DeltaTime, FTenkaichiCameraModeView& OutCameraModeView)
{
	if (!bIsActive) { return false; }
	UpdateStack(DeltaTime);
	BlendStack(OutCameraModeView);
	return true;
}

UTenkaichiCameraMode* UTenkaichiCameraModeStack::GetCameraModeInstance(TSubclassOf<UTenkaichiCameraMode> CameraModeClass)
{
	check(CameraModeClass);
	// 先找缓存里有没有同类的实例
	for (UTenkaichiCameraMode* CameraMode : CameraModeInstances)
	{
		if ((CameraMode != nullptr) && (CameraMode->GetClass() == CameraModeClass)) { return CameraMode; }
	}
	// 没有就 new 一个
	UTenkaichiCameraMode* NewCameraMode = NewObject<UTenkaichiCameraMode>(GetOuter(), CameraModeClass, NAME_None, RF_NoFlags);
	check(NewCameraMode);
	CameraModeInstances.Add(NewCameraMode);
	return NewCameraMode;
}

void UTenkaichiCameraModeStack::UpdateStack(float DeltaTime)
{
	const int32 StackSize = CameraModeStack.Num();
	if (StackSize <= 0) { return; }
	int32 RemoveCount = 0;
	int32 RemoveIndex = INDEX_NONE;
	for (int32 StackIndex = 0; StackIndex < StackSize; ++StackIndex)
	{
		UTenkaichiCameraMode* CameraMode = CameraModeStack[StackIndex];
		check(CameraMode);
		CameraMode->UpdateCameraMode(DeltaTime);
		if (CameraMode->GetBlendWeight() >= 1.0f)
		{
			// 这个模式下面的所有模式都不再需要了，可以移除。
			RemoveIndex = (StackIndex + 1);
			RemoveCount = (StackSize - RemoveIndex);
			break;
		}
	}
	if (RemoveCount > 0)
	{
		for (int32 StackIndex = RemoveIndex; StackIndex < StackSize; ++StackIndex)
		{
			UTenkaichiCameraMode* CameraMode = CameraModeStack[StackIndex];
			check(CameraMode);
			CameraMode->OnDeactivation();
		}
		CameraModeStack.RemoveAt(RemoveIndex, RemoveCount);
	}
}

void UTenkaichiCameraModeStack::BlendStack(FTenkaichiCameraModeView& OutCameraModeView) const
{
	const int32 StackSize = CameraModeStack.Num();
	if (StackSize <= 0) { return; }
	// 从栈底开始，一层层往上混合。
	const UTenkaichiCameraMode* CameraMode = CameraModeStack[StackSize - 1];
	check(CameraMode);
	OutCameraModeView = CameraMode->GetCameraModeView();
	for (int32 StackIndex = (StackSize - 2); StackIndex >= 0; --StackIndex)
	{
		CameraMode = CameraModeStack[StackIndex];
		check(CameraMode);
		OutCameraModeView.Blend(CameraMode->GetCameraModeView(), CameraMode->GetBlendWeight());
	}
}

void UTenkaichiCameraModeStack::GetBlendInfo(float& OutWeightOfTopLayer, FGameplayTag& OutTagOfTopLayer) const
{
	if (CameraModeStack.Num() == 0)
	{
		OutWeightOfTopLayer = 1.0f;
		OutTagOfTopLayer = FGameplayTag();
		return;
	}
	else
	{
		UTenkaichiCameraMode* TopEntry = CameraModeStack.Last();
		check(TopEntry);
		OutWeightOfTopLayer = TopEntry->GetBlendWeight();
		OutTagOfTopLayer = TopEntry->GetCameraTypeTag();
	}
}
```

| 段 | 干什么 |
|----|--------|
| `EvaluateStack` | **每帧主入口**：先 `UpdateStack`（推进所有模式 + 清理完权重的），再 `BlendStack`（混合成最终视图） |
| `GetCameraModeInstance` | 实例缓存——同类型模式复用同一个实例，避免每帧 `NewObject` |
| `UpdateStack` | 推进每个模式的过渡；某个模式权重到 1 后，它下面的模式全移除（下面的被完全遮住） |
| `BlendStack` | 从栈底往上，逐层 `Blend` 叠加权重 |
| `GetBlendInfo` | 取栈顶模式的权重 + 标签（给外部查询"当前是什么模式"） |

> 类比：**"每帧算一次最终画面"**——先让每个模式更新自己的快照和过渡，再把栈里所有快照从底到顶按权重叠成一张最终视图，交给相机组件。
