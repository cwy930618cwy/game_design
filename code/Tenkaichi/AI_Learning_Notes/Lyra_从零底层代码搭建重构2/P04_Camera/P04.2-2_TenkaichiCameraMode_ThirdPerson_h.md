```cpp
#pragma once

#include "TenkaichiCameraMode.h"

/**
 * UTenkaichiCameraMode_ThirdPerson
 *
 *	一个基础的第三人称相机模式。
 */
UCLASS(Abstract, Blueprintable)
class UTenkaichiCameraMode_ThirdPerson : public UTenkaichiCameraMode
{
	GENERATED_BODY()

public:

	// 构造函数。
	UTenkaichiCameraMode_ThirdPerson();

protected:

	// 覆盖基类的视图更新，加入第三人称偏移、蹲伏偏移和防穿墙逻辑。
	virtual void UpdateView(float DeltaTime) override;

	// 检测目标是否下蹲，并设置对应的蹲伏偏移目标。
	void UpdateForTarget(float DeltaTime);

	// 防穿墙入口：计算安全点并调整相机位置避免穿墙。
	void UpdatePreventPenetration(float DeltaTime);

	// 核心防穿墙射线检测：用多个 feeler 射线打点，算出被挡比例并平滑收镜。
	void PreventCameraPenetration(class AActor const& ViewTarget, FVector const& SafeLoc, FVector& CameraLoc, float const& DeltaTime, float& DistBlockedPct, bool bSingleRayOnly);

	// 覆盖调试绘制：打印被撞到的 Actor。
	virtual void DrawDebug(UCanvas* Canvas) const override;

protected:

	// 用视图俯仰角求值的曲线，用来定义相对目标的本地空间偏移。
	UPROPERTY(EditDefaultsOnly, Category = "Third Person", Meta = (EditCondition = "!bUseRuntimeFloatCurves"))
	TObjectPtr<const UCurveVector> TargetOffsetCurve;

	// UE-103986：PIE 期间实时编辑 RuntimeFloatCurves 不生效（不像曲线资产）。
	// 该问题解决后它将变成默认方式，TargetOffsetCurve 将被移除。
	UPROPERTY(EditDefaultsOnly, Category = "Third Person")
	bool bUseRuntimeFloatCurves;

	UPROPERTY(EditDefaultsOnly, Category = "Third Person", Meta = (EditCondition = "bUseRuntimeFloatCurves"))
	FRuntimeFloatCurve TargetOffsetX;

	UPROPERTY(EditDefaultsOnly, Category = "Third Person", Meta = (EditCondition = "bUseRuntimeFloatCurves"))
	FRuntimeFloatCurve TargetOffsetY;

	UPROPERTY(EditDefaultsOnly, Category = "Third Person", Meta = (EditCondition = "bUseRuntimeFloatCurves"))
	FRuntimeFloatCurve TargetOffsetZ;

	// 改变蹲伏偏移淡入/淡出的速度。
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Third Person")
	float CrouchOffsetBlendMultiplier = 5.0f;

	// 防穿墙相关
public:
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category="Collision")
	float PenetrationBlendInTime = 0.1f;

	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category="Collision")
	float PenetrationBlendOutTime = 0.15f;

	/** 如果为真，做碰撞检测让相机不会穿进世界物体里。 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category="Collision")
	bool bPreventPenetration = true;

	/** 如果为真，尝试提前探测附近的墙并提前移动相机，帮助避免弹跳。 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category="Collision")
	bool bDoPredictiveAvoidance = true;

	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Collision")
	float CollisionPushOutDistance = 2.f;

	/** 当相机距离因穿墙被推进到其完整距离的这个百分比时 */
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Collision")
	float ReportPenetrationPercent = 0.f;

	/**
	 * 这些是用于找到相机放置位置的探测射线。
	 * 索引 0：用来防止碰撞的常规探测射线。
	 * 索引 1+：当 bDoPredictiveAvoidance=true 时使用这些射线，扫描玩家向那个方向旋转时可能的撞击，
	 *           并提前把相机拉进来，避免撞击到遮挡物。
	 */
	UPROPERTY(EditDefaultsOnly, Category = "Collision")
	TArray<FTenkaichiPenetrationAvoidanceFeeler> PenetrationAvoidanceFeelers;

	UPROPERTY(Transient)
	float AimLineToDesiredPosBlockedPct;

	UPROPERTY(Transient)
	TArray<TObjectPtr<const AActor>> DebugActorsHitDuringCameraPenetration;

#if ENABLE_DRAW_DEBUG
	mutable float LastDrawDebugTime = -MAX_FLT;
#endif

protected:

	// 设置蹲伏偏移的目标值并重置过渡。
	void SetTargetCrouchOffset(FVector NewTargetOffset);

	// 每帧平滑推进蹲伏偏移的过渡。
	void UpdateCrouchOffset(float DeltaTime);

	FVector InitialCrouchOffset = FVector::ZeroVector;
	FVector TargetCrouchOffset = FVector::ZeroVector;
	float CrouchOffsetBlendPct = 1.0f;
	FVector CurrentCrouchOffset = FVector::ZeroVector;

};
```


| 成员 | 干什么 |
|------|--------|
| `TargetOffsetCurve` | **资产曲线**：以俯仰角为 X，查询出一个 `FVector` 偏移，把相机从枢轴点拉开。`EditCondition = "!bUseRuntimeFloatCurves"` 表示只在"不用运行时曲线"时才在编辑器显示 |
| `bUseRuntimeFloatCurves` | 切换开关：用资产曲线（`TargetOffsetCurve`）还是运行时浮点曲线（下面 XYZ 三条） |
| `TargetOffsetX/Y/Z` | **运行时曲线**：分别以俯仰角查 X/Y/Z 偏移，可在 PIE 里实时调整（比资产曲线灵活，但 PIE 下资产曲线不实时生效） |
| `CrouchOffsetBlendMultiplier` | 蹲伏偏移的过渡速度倍数（默认 5.0，越快镜头跟随越跟手） |

> 类比：**"两条路选一条"**——要么用美术做的曲线资产（`TargetOffsetCurve`），要么用引擎内置的运行时曲线（XYZ 三条）。前者方便策划配置，后者方便程序员运行时调。俯仰角抬高时镜头偏移变大（视角抬高、人物下沉），俯角时相反。

```cpp
	float AimLineToDesiredPosBlockedPct;      // 瞄准线被挡百分比（跨帧保留）
```

| 成员 | 干什么 |
|------|--------|
| `PenetrationBlendInTime` / `OutTime` | 穿墙收镜 / 移出墙回弹的过渡时长（0.1s 快收，0.15s 慢回，避免镜头抖动） |
| `bPreventPenetration` | 总开关：是否做防穿墙 |
| `bDoPredictiveAvoidance` | 是否用预测射线（1+ 号 feeler）提前探测，避免相机怼到墙前才突然弹开 |
| `CollisionPushOutDistance` | 相机与碰撞体保持的最小间距 |
| `ReportPenetrationPercent` | 穿墙比例低于它时，通知 assist 接口"相机被穿透了" |
| `PenetrationAvoidanceFeelers` | **P04.3 的射线组**：0 号主射线防碰撞，1+ 号预测射线提前扫描 |
| `AimLineToDesiredPosBlockedPct` | 瞄准线被挡的百分比，跨帧保留（Transient 不序列化） |
| `LastDrawDebugTime` | 调试用：控制调试绘制的刷新间隔 |

> 类比：**"相机被墙挡住怎么办"**——收到"墙"的信息后，把相机往前收（`BlendIn`），等墙过去了再平滑拉回（`BlendOut`）。预测射线让它**提前**收，而不是怼上了才弹开。

### 1.4 蹲伏偏移内部状态

```cpp
	// 设置蹲伏偏移的目标值并重置过渡。
	void SetTargetCrouchOffset(FVector NewTargetOffset);
	// 每帧平滑推进蹲伏偏移的过渡。
	void UpdateCrouchOffset(float DeltaTime);

	FVector InitialCrouchOffset = FVector::ZeroVector;   // 蹲伏偏移起始值
	FVector TargetCrouchOffset = FVector::ZeroVector;    // 蹲伏偏移目标值
	float CrouchOffsetBlendPct = 1.0f;                   // 蹲伏偏移过渡进度 0~1
	FVector CurrentCrouchOffset = FVector::ZeroVector;   // 蹲伏偏移当前值
```

| 成员 | 干什么 |
|------|--------|
| `SetTargetCrouchOffset` | 设目标 + 重置过渡进度（从当前值开始平滑过渡到目标） |
| `UpdateCrouchOffset` | 每帧按 `CrouchOffsetBlendMultiplier` 速度推进 `BlendPct`，插值出 `CurrentCrouchOffset` |
| `Initial/Target/Current` | 经典的"起止当前"三段插值状态 |

> 类比：**"下蹲时镜头跟下来"**——`Initial`（站着眼高）→ `Target`（蹲着眼高），`UpdateCrouchOffset` 每帧把 `Current` 从初始插到目标，镜头就平滑下移，不会瞬跳。
