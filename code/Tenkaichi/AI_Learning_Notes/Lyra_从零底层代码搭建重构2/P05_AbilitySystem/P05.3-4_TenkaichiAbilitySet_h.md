```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#pragma once

#include "ActiveGameplayEffectHandle.h"
#include "Engine/DataAsset.h"
#include "AttributeSet.h"
#include "GameplayTagContainer.h"

#include "GameplayAbilitySpecHandle.h"
#include "TenkaichiAbilitySet.generated.h"

class UAttributeSet;
class UGameplayEffect;
class UTenkaichiAbilitySystemComponent;
class UTenkaichiGameplayAbility;
class UObject;


/**
 * FTenkaichiAbilitySet_GameplayAbility
 *
 *	这个能力集要授予的一个「技能」的数据：技能类 + 等级 + 输入标签
 */
USTRUCT(BlueprintType)
struct FTenkaichiAbilitySet_GameplayAbility
{
	GENERATED_BODY()

public:

	// 要授予的技能类
	UPROPERTY(EditDefaultsOnly)
	TSubclassOf<UTenkaichiGameplayAbility> Ability = nullptr;

	// 授予的技能等级
	UPROPERTY(EditDefaultsOnly)
	int32 AbilityLevel = 1;

	// 用于给这个技能处理输入的标签
	UPROPERTY(EditDefaultsOnly, Meta = (Categories = "InputTag"))
	FGameplayTag InputTag;
};


/**
 * FTenkaichiAbilitySet_GameplayEffect
 *
 *	这个能力集要授予的一个「效果」的数据：效果类 + 等级
 */
USTRUCT(BlueprintType)
struct FTenkaichiAbilitySet_GameplayEffect
{
	GENERATED_BODY()

public:

	// 要授予的效果类
	UPROPERTY(EditDefaultsOnly)
	TSubclassOf<UGameplayEffect> GameplayEffect = nullptr;

	// 授予的效果等级
	UPROPERTY(EditDefaultsOnly)
	float EffectLevel = 1.0f;
};

/**
 * FTenkaichiAbilitySet_AttributeSet
 *
 *	这个能力集要授予的一个「属性集」的数据：属性集类
 */
USTRUCT(BlueprintType)
struct FTenkaichiAbilitySet_AttributeSet
{
	GENERATED_BODY()

public:
	// 要授予的属性集类
	UPROPERTY(EditDefaultsOnly)
	TSubclassOf<UAttributeSet> AttributeSet;

};

/**
 * FTenkaichiAbilitySet_GrantedHandles
 *
 *	记录这个能力集已经授予了哪些技能/效果/属性集（用句柄/指针），方便之后一次性收回
 */
USTRUCT(BlueprintType)
struct FTenkaichiAbilitySet_GrantedHandles
{
	GENERATED_BODY()

public:

	// 记下已授予的技能句柄
	void AddAbilitySpecHandle(const FGameplayAbilitySpecHandle& Handle);
	// 记下已授予的效果句柄
	void AddGameplayEffectHandle(const FActiveGameplayEffectHandle& Handle);
	// 记下已授予的属性集指针
	void AddAttributeSet(UAttributeSet* Set);

	// 把记录的所有技能/效果/属性集从 ASC 上收回
	void TakeFromAbilitySystem(UTenkaichiAbilitySystemComponent* LyraASC);

protected:

	// 已授予的技能句柄列表
	UPROPERTY()
	TArray<FGameplayAbilitySpecHandle> AbilitySpecHandles;

	// 已授予的效果句柄列表
	UPROPERTY()
	TArray<FActiveGameplayEffectHandle> GameplayEffectHandles;

	// 已授予的属性集指针列表
	UPROPERTY()
	TArray<TObjectPtr<UAttributeSet>> GrantedAttributeSets;
};


/**
 * UTenkaichiAbilitySet
 *
 *	不可变的数据资产：一次性授予一组技能/效果/属性集
 */
UCLASS(BlueprintType, Const)
class UTenkaichiAbilitySet : public UPrimaryDataAsset
{
	GENERATED_BODY()

public:

	// 构造函数
	UTenkaichiAbilitySet(const FObjectInitializer& ObjectInitializer = FObjectInitializer::Get());

	// 把这个能力集授予指定的 ASC；返回的句柄收集器可用于之后收回所有已授予内容
	void GiveToAbilitySystem(UTenkaichiAbilitySystemComponent* LyraASC, FTenkaichiAbilitySet_GrantedHandles* OutGrantedHandles, UObject* SourceObject = nullptr) const;

protected:

	// 授予这个能力集时要授予的技能列表
	UPROPERTY(EditDefaultsOnly, Category = "Gameplay Abilities", meta=(TitleProperty=Ability))
	TArray<FTenkaichiAbilitySet_GameplayAbility> GrantedGameplayAbilities;

	// 授予这个能力集时要授予的效果列表
	UPROPERTY(EditDefaultsOnly, Category = "Gameplay Effects", meta=(TitleProperty=GameplayEffect))
	TArray<FTenkaichiAbilitySet_GameplayEffect> GrantedGameplayEffects;

	// 授予这个能力集时要授予的属性集列表
	UPROPERTY(EditDefaultsOnly, Category = "Attribute Sets", meta=(TitleProperty=AttributeSet))
	TArray<FTenkaichiAbilitySet_AttributeSet> GrantedAttributes;
};
```

## 四、逐段讲解

### 1. 头文件与前置声明

```cpp
#pragma once
#include "ActiveGameplayEffectHandle.h"
#include "Engine/DataAsset.h"
#include "AttributeSet.h"
#include "GameplayTagContainer.h"
#include "GameplayAbilitySpecHandle.h"
#include "TenkaichiAbilitySet.generated.h"

class UAttributeSet;
class UGameplayEffect;
class UTenkaichiAbilitySystemComponent;
class UTenkaichiGameplayAbility;
class UObject;
```

- `Engine/DataAsset.h`：基类 `UPrimaryDataAsset`。
- `AttributeSet.h`：`UAttributeSet`（属性集基类）。
- `GameplayTagContainer.h`：`FGameplayTag`（输入标签）。
- `ActiveGameplayEffectHandle.h` / `GameplayAbilitySpecHandle.h`：两种句柄类型。
- 前置声明：`.h` 里只用 `TSubclassOf<...>`（类引用）和指针，前置声明足够 → 零挂起。

### 2. FTenkaichiAbilitySet_GameplayAbility（技能数据）

```cpp
USTRUCT(BlueprintType)
struct FTenkaichiAbilitySet_GameplayAbility
{
	GENERATED_BODY()
public:
	UPROPERTY(EditDefaultsOnly)
	TSubclassOf<UTenkaichiGameplayAbility> Ability = nullptr;

	UPROPERTY(EditDefaultsOnly)
	int32 AbilityLevel = 1;

	UPROPERTY(EditDefaultsOnly, Meta = (Categories = "InputTag"))
	FGameplayTag InputTag;
};
```

- `USTRUCT(BlueprintType)`：能在蓝图里作为变量类型使用。
- `TSubclassOf<UTenkaichiGameplayAbility> Ability = nullptr;`：要授的技能**类**（不是实例），默认空。
- `int32 AbilityLevel = 1;`：技能等级。
- `FGameplayTag InputTag;`：这个技能对应的**输入标签**（用 `Meta = (Categories = "InputTag")` 约束只能在 `InputTag` 分类里选）。
  后续 `.cpp` 里会把 InputTag 加到技能的动态源标签里，用于输入映射。

### 3. FTenkaichiAbilitySet_GameplayEffect（效果数据）

```cpp
USTRUCT(BlueprintType)
struct FTenkaichiAbilitySet_GameplayEffect
{
	GENERATED_BODY()
public:
	UPROPERTY(EditDefaultsOnly)
	TSubclassOf<UGameplayEffect> GameplayEffect = nullptr;

	UPROPERTY(EditDefaultsOnly)
	float EffectLevel = 1.0f;
};
```

- 效果类 `TSubclassOf<UGameplayEffect>` + 等级 `float EffectLevel`（效果等级是浮点）。

### 4. FTenkaichiAbilitySet_AttributeSet（属性集数据）

```cpp
USTRUCT(BlueprintType)
struct FTenkaichiAbilitySet_AttributeSet
{
	GENERATED_BODY()
public:
	UPROPERTY(EditDefaultsOnly)
	TSubclassOf<UAttributeSet> AttributeSet;
};
```

- 只一个：属性集**类**。没有等级（属性集无等级概念）。

### 5. FTenkaichiAbilitySet_GrantedHandles（句柄收集器）

```cpp
USTRUCT(BlueprintType)
struct FTenkaichiAbilitySet_GrantedHandles
{
	GENERATED_BODY()
public:
	void AddAbilitySpecHandle(const FGameplayAbilitySpecHandle& Handle);
	void AddGameplayEffectHandle(const FActiveGameplayEffectHandle& Handle);
	void AddAttributeSet(UAttributeSet* Set);
	void TakeFromAbilitySystem(UTenkaichiAbilitySystemComponent* LyraASC);

protected:
	UPROPERTY()
	TArray<FGameplayAbilitySpecHandle> AbilitySpecHandles;
	UPROPERTY()
	TArray<FActiveGameplayEffectHandle> GameplayEffectHandles;
	UPROPERTY()
	TArray<TObjectPtr<UAttributeSet>> GrantedAttributeSets;
};
```

- **作用**：`GiveToAbilitySystem` 授予时，把每个技能/效果/属性集的句柄（或指针）存进这里；之后用 `TakeFromAbilitySystem` 一次性全部收回。
- 三个 `Add*` 方法：把句柄/指针塞进对应 `TArray`。
- `TakeFromAbilitySystem(ASC)`：遍历三个数组，逐一从 ASC 上 `ClearAbility` / `RemoveActiveGameplayEffect` / `RemoveSpawnedAttribute`，最后 `Reset()` 清空。
- 数据在 `protected`：只有结构体自己（和 `.cpp` 实现）能访问。

### 6. UTenkaichiAbilitySet（数据资产本体）

```cpp
UCLASS(BlueprintType, Const)
class UTenkaichiAbilitySet : public UPrimaryDataAsset
{
	GENERATED_BODY()
public:
	UTenkaichiAbilitySet(const FObjectInitializer& ObjectInitializer = FObjectInitializer::Get());
	void GiveToAbilitySystem(UTenkaichiAbilitySystemComponent* LyraASC, FTenkaichiAbilitySet_GrantedHandles* OutGrantedHandles, UObject* SourceObject = nullptr) const;

protected:
	UPROPERTY(EditDefaultsOnly, Category = "Gameplay Abilities", meta=(TitleProperty=Ability))
	TArray<FTenkaichiAbilitySet_GameplayAbility> GrantedGameplayAbilities;
	UPROPERTY(EditDefaultsOnly, Category = "Gameplay Effects", meta=(TitleProperty=GameplayEffect))
	TArray<FTenkaichiAbilitySet_GameplayEffect> GrantedGameplayEffects;
	UPROPERTY(EditDefaultsOnly, Category = "Attribute Sets", meta=(TitleProperty=AttributeSet))
	TArray<FTenkaichiAbilitySet_AttributeSet> GrantedAttributes;
};
```

- `UCLASS(BlueprintType, Const)`：可在蓝图用；`Const` 表示这个资产运行时不可变（数据只读）。
- `UPrimaryDataAsset`：**主数据资产**——能被 `AssetManager` 管理和异步加载（这是后续 P14 的核心，现在先知道即可）。
- 构造函数带 `FObjectInitializer`（默认参数 `FObjectInitializer::Get()`）。
- `GiveToAbilitySystem(...)`：核心方法，`.cpp` 里实现三件事：① 授予属性集 ② 授予技能 ③ 授予效果。
- 三个 `TArray` 配好编辑器可编辑的"要授什么"清单，`Category` 分组 + `TitleProperty` 决定数组元素在编辑器里显示哪个字段做标题。

## 五、几个要记住的点

1. **DataAsset 是"打包清单"**：三个 TArray 就是可配置的技能/效果/属性集清单，编辑器里拖进资产即可。
2. **GrantedHandles 是"收据"**：授予后记住句柄，才能准确收回。
3. **UPrimaryDataAsset**：为后续 AssetManager 异步加载做准备（P14）。
4. **`.h` 零挂起**：全部前置声明/`TSubclassOf`。
