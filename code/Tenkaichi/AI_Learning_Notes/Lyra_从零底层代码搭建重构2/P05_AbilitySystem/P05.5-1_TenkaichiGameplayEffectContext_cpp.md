```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#include "TenkaichiGameplayEffectContext.h"

#include "AbilitySystem/TenkaichiAbilitySourceInterface.h"
#include "Engine/HitResult.h"
#include "PhysicalMaterials/PhysicalMaterial.h"

#if UE_WITH_IRIS
#include "Iris/ReplicationState/PropertyNetSerializerInfoRegistry.h"
#include "Serialization/GameplayEffectContextNetSerializer.h"
#endif

#include UE_INLINE_GENERATED_CPP_BY_NAME(TenkaichiGameplayEffectContext)

class FArchive;

// 从 handle 里取出包裹的 FTenkaichiGameplayEffectContext，若不存在或类型不对则返回 nullptr
FTenkaichiGameplayEffectContext* FTenkaichiGameplayEffectContext::ExtractEffectContext(struct FGameplayEffectContextHandle Handle)
{
	FGameplayEffectContext* BaseEffectContext = Handle.Get();
	if ((BaseEffectContext != nullptr) && BaseEffectContext->GetScriptStruct()->IsChildOf(FTenkaichiGameplayEffectContext::StaticStruct()))
	{
		return (FTenkaichiGameplayEffectContext*)BaseEffectContext;
	}

	return nullptr;
}

// 网络序列化：先序列化基类，再序列化新增字段
bool FTenkaichiGameplayEffectContext::NetSerialize(FArchive& Ar, class UPackageMap* Map, bool& bOutSuccess)
{
	FGameplayEffectContext::NetSerialize(Ar, Map, bOutSuccess);

	// 目前不序列化以下字段（用于激活后）：
	// CartridgeID

	return true;
}

#if UE_WITH_IRIS
namespace UE::Net
{
	// 转发给 FGameplayEffectContextNetSerializer
	// 注意：如果修改了 FTenkaichiGameplayEffectContext::NetSerialize()，就需要实现一个自定义 NetSerializer，
	// 因为当前的 fallback 就不够用了。
	UE_NET_IMPLEMENT_FORWARDING_NETSERIALIZER_AND_REGISTRY_DELEGATES(TenkaichiGameplayEffectContext, FGameplayEffectContextNetSerializer);
}
#endif

// 设置作为能力来源的对象
void FTenkaichiGameplayEffectContext::SetAbilitySource(const ITenkaichiAbilitySourceInterface* InObject, float InSourceLevel)
{
	AbilitySourceObject = MakeWeakObjectPtr(Cast<const UObject>(InObject));
	//SourceLevel = InSourceLevel;
}

// 返回与来源对象关联的能力来源接口。仅在权威端有效。
const ITenkaichiAbilitySourceInterface* FTenkaichiGameplayEffectContext::GetAbilitySource() const
{
	return Cast<ITenkaichiAbilitySourceInterface>(AbilitySourceObject.Get());
}

// 若有命中结果则返回其物理材质
const UPhysicalMaterial* FTenkaichiGameplayEffectContext::GetPhysicalMaterial() const
{
	if (const FHitResult* HitResultPtr = GetHitResult())
	{
		return HitResultPtr->PhysMaterial.Get();
	}
	return nullptr;
}
```

## 五、逐段讲解

### 1. 头文件与 include

```cpp
#include "TenkaichiGameplayEffectContext.h"
#include "AbilitySystem/TenkaichiAbilitySourceInterface.h"

#include "PhysicalMaterials/PhysicalMaterial.h"

```

- `TenkaichiAbilitySourceInterface.h`：要 `Cast<ITenkaichiAbilitySourceInterface>`，需要完整接口定义。
- `Engine/HitResult.h`：`FHitResult` 类型（`GetHitResult()` 返回它）。
- `PhysicalMaterials/PhysicalMaterial.h`：`UPhysicalMaterial` 类型。
- `UE_WITH_IRIS` 段：UE5 新的 Iris 网络复制系统，用 `#if` 保护，只有启用了 Iris 才编译。
- `UE_INLINE_GENERATED_CPP_BY_NAME`：固定宏（已教过）。
- `class FArchive;`：`.cpp` 里 `NetSerialize` 签名用到 `FArchive`，前置声明。

### 2. ExtractEffectContext（核心入口）

```cpp
FTenkaichiGameplayEffectContext* FTenkaichiGameplayEffectContext::ExtractEffectContext(struct FGameplayEffectContextHandle Handle)
{
	FGameplayEffectContext* BaseEffectContext = Handle.Get();
	if ((BaseEffectContext != nullptr) && BaseEffectContext->GetScriptStruct()->IsChildOf(FTenkaichiGameplayEffectContext::StaticStruct()))
	{
		return (FTenkaichiGameplayEffectContext*)BaseEffectContext;
	}
	return nullptr;
}
```

- `Handle.Get()`：从 handle 取出基类指针 `FGameplayEffectContext*`。
- `GetScriptStruct()->IsChildOf(...)`：判断这个上下文**是不是** `FTenkaichiGameplayEffectContext` 类型（或它的子类）。
- 是 → 安全 `Cast` 返回；不是 → `nullptr`。
- 这就是 DamageExecution 里"安全取出本类型"的源头。

### 3. NetSerialize（网络序列化）

```cpp
bool FTenkaichiGameplayEffectContext::NetSerialize(FArchive& Ar, class UPackageMap* Map, bool& bOutSuccess)
{
	FGameplayEffectContext::NetSerialize(Ar, Map, bOutSuccess);
	// 目前不序列化以下字段（用于激活后）：
	// CartridgeID
	return true;
}
```

- 先调基类 `NetSerialize`（序列化基类自带字段）。
- `CartridgeID` 是激活后才用的字段，Lyra 选择**不序列化**（所以注释里说明，没写序列化代码）。
- 返回 `true` 表示成功。

### 4. Iris 网络序列化注册

```cpp
#if UE_WITH_IRIS
namespace UE::Net
{
	UE_NET_IMPLEMENT_FORWARDING_NETSERIALIZER_AND_REGISTRY_DELEGATES(TenkaichiGameplayEffectContext, FGameplayEffectContextNetSerializer);
}
#endif
```

- **Iris** 是 UE5 新的网络复制系统。
- 这个宏把 `FTenkaichiGameplayEffectContext` 的序列化**转发**给引擎已有的 `FGameplayEffectContextNetSerializer`，并注册到 Iris 注册表。
- 第一个参数 `TenkaichiGameplayEffectContext` 是**去掉了 `F` 前缀**的名字（宏约定）。
- 受 `UE_WITH_IRIS` 保护，旧复制系统（非 Iris）时不编译。

### 5. SetAbilitySource / GetAbilitySource

```cpp
void FTenkaichiGameplayEffectContext::SetAbilitySource(const ITenkaichiAbilitySourceInterface* InObject, float InSourceLevel)
{
	AbilitySourceObject = MakeWeakObjectPtr(Cast<const UObject>(InObject));
	//SourceLevel = InSourceLevel;
}

const ITenkaichiAbilitySourceInterface* FTenkaichiGameplayEffectContext::GetAbilitySource() const
{

```

- **`SetAbilitySource`**：把 `InObject` 转成 `const UObject*`，用 `MakeWeakObjectPtr` 存成弱引用（不持有强引用，防止循环引用/悬空）。
  - `//SourceLevel = InSourceLevel;` 是被注释掉的（Lyra 原样保留，说明 `SourceLevel` 目前没启用）。
- **`GetAbilitySource`**：从弱引用 `.Get()` 取出对象，`Cast` 回接口类型返回。

### 6. GetPhysicalMaterial

```cpp
const UPhysicalMaterial* FTenkaichiGameplayEffectContext::GetPhysicalMaterial() const
{
	if (const FHitResult* HitResultPtr = GetHitResult())
	{
		return HitResultPtr->PhysMaterial.Get();
	}
	return nullptr;
}
```

- `GetHitResult()`：基类方法，返回命中结果（可能为 nullptr）。
- 有命中结果 → 取 `PhysMaterial`（命中点的物理材质）；无 → `nullptr`。
- 这是 DamageExecution 里"物理材质衰减"取材质的地方。
