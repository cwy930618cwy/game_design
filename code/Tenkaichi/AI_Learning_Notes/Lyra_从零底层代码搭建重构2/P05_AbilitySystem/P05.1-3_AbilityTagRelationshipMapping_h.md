```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#pragma once

#include "Engine/DataAsset.h"
#include "GameplayTagContainer.h"

#include "TenkaichiAbilityTagRelationshipMapping.generated.h"

class UObject;

/** 定义「不同技能标签之间的关系」的结构体 */
USTRUCT()
struct FTenkaichiAbilityTagRelationship
{
	GENERATED_BODY()

	/** 这条「容器关系」所关心的标签。单个标签，但一个技能可以有多个这样的关系 */
	UPROPERTY(EditAnywhere, Category = Ability, meta = (Categories = "Gameplay.Action"))
	FGameplayTag AbilityTag;

	/** 使用「这个标签」的技能，会阻断「这些标签」的技能 */
	UPROPERTY(EditAnywhere, Category = Ability)
	FGameplayTagContainer AbilityTagsToBlock;

	/** 使用「这个标签」的技能，会取消「这些标签」的技能 */
	UPROPERTY(EditAnywhere, Category = Ability)
	FGameplayTagContainer AbilityTagsToCancel;

	/** 如果某技能带有「这个标签」，这些标签会被隐式加到它的「激活要求标签」里 */
	UPROPERTY(EditAnywhere, Category = Ability)
	FGameplayTagContainer ActivationRequiredTags;

	/** 如果某技能带有「这个标签」，这些标签会被隐式加到它的「激活禁止标签」里 */
	UPROPERTY(EditAnywhere, Category = Ability)
	FGameplayTagContainer ActivationBlockedTags;
};


/** 描述「技能标签如何阻断或取消其他技能」的映射 */
UCLASS()
class UTenkaichiAbilityTagRelationshipMapping : public UDataAsset
{
	GENERATED_BODY()

private:
	/** 不同 gameplay 标签之间的关系列表（哪些阻断、哪些取消） */
	UPROPERTY(EditAnywhere, Category = Ability, meta=(TitleProperty="AbilityTag"))
	TArray<FTenkaichiAbilityTagRelationship> AbilityTagRelationships;

public:
	// 给一组技能标签，解析出「要阻断/取消的标签」
	void GetAbilityTagsToBlockAndCancel(const FGameplayTagContainer& AbilityTags, FGameplayTagContainer* OutTagsToBlock, FGameplayTagContainer* OutTagsToCancel) const;

	// 给一组技能标签，解析出「激活时额外要求/禁止的标签」
	void GetRequiredAndBlockedActivationTags(const FGameplayTagContainer& AbilityTags, FGameplayTagContainer* OutActivationRequired, FGameplayTagContainer* OutActivationBlocked) const;

	// 判断「这组技能标签」是否被「传入的动作标签」取消
	bool IsAbilityCancelledByTag(const FGameplayTagContainer& AbilityTags, const FGameplayTag& ActionTag) const;
};
```

## 四、逐段讲解

### 1. 头文件

```cpp
#include "Engine/DataAsset.h"
#include "GameplayTagContainer.h"
#include "TenkaichiAbilityTagRelationshipMapping.generated.h"
```

- `Engine/DataAsset.h`：提供 `UDataAsset` 基类（数据资产——一种能在编辑器里保存配置的资产）。
- `GameplayTagContainer.h`：提供 `FGameplayTag`（单个标签）和 `FGameplayTagContainer`（标签容器）。
- `generated.h`：UHT 生成，放最后。

### 2. 结构体 `FTenkaichiAbilityTagRelationship`

```cpp
USTRUCT()
struct FTenkaichiAbilityTagRelationship
{
	GENERATED_BODY()
	...
};
```

- `USTRUCT()`：这是一个 UE 反射结构体（可以在编辑器里显示、编辑）。
- 里面 5 个 `UPROPERTY(EditAnywhere, ...)` 字段，都能在编辑器里改（`EditAnywhere` = 任何地方都可编辑）。

**5 个字段逐个讲**：

| 字段 | 类型 | 含义 |
|------|------|------|
| `AbilityTag` | `FGameplayTag`（单个） | 这条关系"关于哪个标签"。注意 `meta = (Categories = "Gameplay.Action")`：编辑器里这个字段只显示 `Gameplay.Action` 分类下的标签，方便挑选 |
| `AbilityTagsToBlock` | `FGameplayTagContainer`（容器） | 用该标签的技能，会**阻断**这些标签 |
| `AbilityTagsToCancel` | `FGameplayTagContainer` | 用该标签的技能，会**取消**这些标签 |
| `ActivationRequiredTags` | `FGameplayTagContainer` | 有该标签的技能，**隐式额外要求**这些标签 |
| `ActivationBlockedTags` | `FGameplayTagContainer` | 有该标签的技能，**隐式额外禁止**这些标签 |

> 注意 `AbilityTag` 是**单个** `FGameplayTag`，另外 4 个是**容器** `FGameplayTagContainer`（可以放多个）。

### 3. 数据资产类 `UTenkaichiAbilityTagRelationshipMapping`

```cpp
UCLASS()
class UTenkaichiAbilityTagRelationshipMapping : public UDataAsset
{
	GENERATED_BODY()
	...
};
```

- `UCLASS()`：UE 反射类。
- 继承 `UDataAsset`：所以它是一个**数据资产**，能在内容浏览器里创建、在编辑器里配置。

### 4. private 字段

```cpp
UPROPERTY(EditAnywhere, Category = Ability, meta=(TitleProperty="AbilityTag"))
TArray<FTenkaichiAbilityTagRelationship> AbilityTagRelationships;
```

- 一个**数组**，装所有"关系"条目（每条就是一个 `FTenkaichiAbilityTagRelationship`）。
- `meta=(TitleProperty="AbilityTag")`：在编辑器里这个数组的每个元素，用它的 `AbilityTag` 字段作为显示标题（而不是默认的 "0 号元素"）。
- 数组是 private，但带 `EditAnywhere`，所以编辑器能改，代码里不能直接访问。

### 5. public 三个查询函数

```cpp
void GetAbilityTagsToBlockAndCancel(const FGameplayTagContainer& AbilityTags, FGameplayTagContainer* OutTagsToBlock, FGameplayTagContainer* OutTagsToCancel) const;
void GetRequiredAndBlockedActivationTags(const FGameplayTagContainer& AbilityTags, FGameplayTagContainer* OutActivationRequired, FGameplayTagContainer* OutActivationBlocked) const;
bool IsAbilityCancelledByTag(const FGameplayTagContainer& AbilityTags, const FGameplayTag& ActionTag) const;
```

- 三个都是 `const` 函数（不修改自身数据，纯查询）。
- 前两个用**出参（Out 参数）**：结果通过指针 `OutTagsToBlock` 等写出去，因为要"一次算多个结果"。
- 第三个返回 `bool`：是/否。

## 五、几个要记住的点

1. **`UDataAsset`**：数据资产的基类，策划在编辑器里配置。
2. **`USTRUCT` + `UCLASS`**：结构体和类都要用 UHT 反射，才能在编辑器里编辑。
3. **`meta=(Categories=...)`**：限制标签下拉框只显示某分类。
4. **`meta=(TitleProperty=...)`**：数组元素用某字段做标题。
5. **出参用指针**：要一次返回多个结果时，用 `Out` 指针参数，而不是返回值。
