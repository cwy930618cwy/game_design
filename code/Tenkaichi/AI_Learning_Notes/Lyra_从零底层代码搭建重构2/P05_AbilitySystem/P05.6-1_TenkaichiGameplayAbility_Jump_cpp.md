```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#include "TenkaichiGameplayAbility_Jump.h"

#include "AbilitySystem/Abilities/TenkaichiGameplayAbility.h"
#include "Character/TenkaichiCharacter.h"

#include UE_INLINE_GENERATED_CPP_BY_NAME(TenkaichiGameplayAbility_Jump)

struct FGameplayTagContainer;


// 构造函数：设为按 Actor 实例化 + 本地预测（客户端立即响应，不等服务器）
UTenkaichiGameplayAbility_Jump::UTenkaichiGameplayAbility_Jump(const FObjectInitializer& ObjectInitializer)
	: Super(ObjectInitializer)
{
	InstancingPolicy = EGameplayAbilityInstancingPolicy::InstancedPerActor;
	NetExecutionPolicy = EGameplayAbilityNetExecutionPolicy::LocalPredicted;
}

// 判断能否激活：依次检查角色是否存在、角色当前能否跳跃、以及基类是否允许
bool UTenkaichiGameplayAbility_Jump::CanActivateAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayTagContainer* SourceTags, const FGameplayTagContainer* TargetTags, FGameplayTagContainer* OptionalRelevantTags) const
{
	if (!ActorInfo || !ActorInfo->AvatarActor.IsValid())
	{
		return false;
	}

	const ATenkaichiCharacter* TenkaichiCharacter = Cast<ATenkaichiCharacter>(ActorInfo->AvatarActor.Get());
	if (!TenkaichiCharacter || !TenkaichiCharacter->CanJump())
	{
		return false;
	}

	if (!Super::CanActivateAbility(Handle, ActorInfo, SourceTags, TargetTags, OptionalRelevantTags))
	{
		return false;
	}

	return true;
}

// 技能结束：兜底停止跳跃（防蓝图忘记调用），再交给父类处理
void UTenkaichiGameplayAbility_Jump::EndAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayAbilityActivationInfo ActivationInfo, bool bReplicateEndAbility, bool bWasCancelled)
{
	// 万一蓝图没有调用停止跳跃，这里兜底停一次
	CharacterJumpStop();

	Super::EndAbility(Handle, ActorInfo, ActivationInfo, bReplicateEndAbility, bWasCancelled);
}

// 让本地控制的角色开始跳跃（若尚未处于跳跃状态）
void UTenkaichiGameplayAbility_Jump::CharacterJumpStart()
{
	if (ATenkaichiCharacter* TenkaichiCharacter = GetTenkaichiCharacterFromActorInfo())
	{
		if (TenkaichiCharacter->IsLocallyControlled() && !TenkaichiCharacter->bPressedJump)
		{
			TenkaichiCharacter->UnCrouch();
			TenkaichiCharacter->Jump();
		}
	}
}

// 让本地控制的角色停止跳跃（若正处于跳跃状态）
void UTenkaichiGameplayAbility_Jump::CharacterJumpStop()
{
	if (ATenkaichiCharacter* TenkaichiCharacter = GetTenkaichiCharacterFromActorInfo())
	{
		if (TenkaichiCharacter->IsLocallyControlled() && TenkaichiCharacter->bPressedJump)
		{
			TenkaichiCharacter->StopJumping();
		}
	}
}
```

### 2. 构造函数（2 个策略）

```cpp
UTenkaichiGameplayAbility_Jump::UTenkaichiGameplayAbility_Jump(const FObjectInitializer& ObjectInitializer)
	: Super(ObjectInitializer)
{
	InstancingPolicy = EGameplayAbilityInstancingPolicy::InstancedPerActor;
	NetExecutionPolicy = EGameplayAbilityNetExecutionPolicy::LocalPredicted;
}
```

**两个策略逐个讲**：

| 策略 | 值 | 含义 |
|------|-----|------|
| **`InstancingPolicy`** | `InstancedPerActor` | 每个 Actor **一个实例**。因为跳跃需要"记住"当前是不是在跳的状态（虽然本类没加字段，但蓝图子类可能有） |
| **`NetExecutionPolicy`** | **`LocalPredicted`** | **本地预测** —— 客户端**立即本地执行**，同时告知服务器 |

> **`LocalPredicted` 是本类最重要的设定**（对比一下）：
> - 如果设 `ServerInitiated`（像 P05.5-4-1 阶段技能那样）：客户端按跳跃键 → 发请求给服务器 → 服务器确认 → 回包 → 角色才跳起来。**一个 RTT 的延迟**，手感极差。
> - 设 `LocalPredicted`：客户端**立刻跳**，同时告诉服务器；服务器校验不合法时会下发纠正（回滚）。**手感即时，这是所有"玩家操作类技能"的标配**。
>
> 对比 P05.5-4-1 `GamePhaseAbility` 用 `ServerInitiated` —— 因为阶段是全局权威状态，**绝不能**让客户端预测。两类技能的策略选择正好相反，理解了这个就理解了 GAS 策略设计的意图。

### 3. CanActivateAbility（三层判断，本类逻辑核心）

```cpp
bool UTenkaichiGameplayAbility_Jump::CanActivateAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayTagContainer* SourceTags, const FGameplayTagContainer* TargetTags, FGameplayTagContainer* OptionalRelevantTags) const
{
	if (!ActorInfo || !ActorInfo->AvatarActor.IsValid())
	{
		return false;
	}

	const ATenkaichiCharacter* TenkaichiCharacter = Cast<ATenkaichiCharacter>(ActorInfo->AvatarActor.Get());
	if (!TenkaichiCharacter || !TenkaichiCharacter->CanJump())
	{
		return false;
	}

	if (!Super::CanActivateAbility(Handle, ActorInfo, SourceTags, TargetTags, OptionalRelevantTags))
	{
		return false;
	}

	return true;
}
```

**三层"守卫"结构**（UE 里常见的 early-return 写法）：

1. **第一层：`!ActorInfo || !ActorInfo->AvatarActor.IsValid()`**
   - `ActorInfo` 为空 或 Avatar 角色无效 → 拒绝。防止空指针崩溃。
2. **第二层：`Cast<ATenkaichiCharacter>` 失败 或 `!CanJump()`**
   - `Cast` 失败 = Avatar 不是 Lyra 角色（比如是别的 Pawn）→ 拒绝。
   - `CanJump()` 是**引擎 `ACharacter` 的方法**，判断当前物理状态能不能跳（不在空中、没被压制等）。
3. **第三层：`!Super::CanActivateAbility(...)`** ⚠️ **最关键，不能漏**
   - 基类（P05.3-B LyraGA）里有**激活组检查**、**Tag 关系检查**（Required/Blocked）、官方的 CanActivate 判定。
   - **漏了这行 = 绕过整套规则**（比如角色死了还能跳）。
- **全通过 → `return true`**。

> **注意返回类型是 `bool` 且函数是 `const`** —— 判断函数不修改状态。

### 4. EndAbility（兜底）

```cpp
void UTenkaichiGameplayAbility_Jump::EndAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayAbilityActivationInfo ActivationInfo, bool bReplicateEndAbility, bool bWasCancelled)
{
	// 万一蓝图没有调用停止跳跃，这里兜底停一次
	CharacterJumpStop();

	Super::EndAbility(Handle, ActorInfo, ActivationInfo, bReplicateEndAbility, bWasCancelled);
}
```

- **`CharacterJumpStop();`** —— 无条件调一次。
- **为什么安全**：`CharacterJumpStop()` 内部有 `if (bPressedJump)` 判断——**没在跳的时候是空操作**，不会有任何副作用。
- **Lyra 原注释**（已中文化）："in case the ability blueprint doesn't call it" —— 蓝图子类应该在跳跃动画/逻辑结束时调 `CharacterJumpStop()`，但**忘了的话角色会卡在 `bPressedJump = true` 一直跳**，这里兜底。
- **`Super::EndAbility(...)` 在最后** —— ⚠️ 必须调，交回基类做清理（激活组、Tag 移除等）。

### 5. CharacterJumpStart / CharacterJumpStop（对称的一对）

```cpp
void UTenkaichiGameplayAbility_Jump::CharacterJumpStart()
{
	if (ATenkaichiCharacter* TenkaichiCharacter = GetTenkaichiCharacterFromActorInfo())
	{
		if (TenkaichiCharacter->IsLocallyControlled() && !TenkaichiCharacter->bPressedJump)
		{
			TenkaichiCharacter->UnCrouch();
			TenkaichiCharacter->Jump();
		}
	}
}

void UTenkaichiGameplayAbility_Jump::CharacterJumpStop()
{
	if (ATenkaichiCharacter* TenkaichiCharacter = GetTenkaichiCharacterFromActorInfo())
	{
		if (TenkaichiCharacter->IsLocallyControlled() && TenkaichiCharacter->bPressedJump)
		{
			TenkaichiCharacter->StopJumping();
		}
	}
}
```

**结构完全对称**，逐个要素讲：

1. **`GetTenkaichiCharacterFromActorInfo()`** —— P05.3-B `UTenkaichiGameplayAbility` 提供的辅助函数（工程 `TenkaichiGameplayAbility.cpp:103` 已实现），从 `ActorInfo` 里安全取出 Lyra 角色。**比自己 `Cast<ATenkaichiCharacter>(ActorInfo->AvatarActor.Get())` 更简洁**，且基类已经处理了判空。
2. **`IsLocallyControlled()`** —— ⚠️ **只操作本地控制的角色**。网络下每个人控制自己的 Pawn，这个函数保证你不会去操作别人的角色。
3. **`!bPressedJump` / `bPressedJump`** —— **幂等保护**：
   - Start 时：已经按着跳了就不再跳（避免连点重复触发）
   - Stop 时：没在跳就不调（空操作）
   - `bPressedJump` 是引擎 `ACharacter` 的成员，表示"跳跃键是否被按下"。
4. **`UnCrouch()`**（仅 Start 有）—— 蹲着时**先站起来再跳**。引擎不允许直接蹲跳，Lyra 帮你自动处理。⚠️ **Stop 函数里没有这行**，Stop 只是停止跳跃。
5. **`Jump()` / `StopJumping()`** —— 引擎 `ACharacter` 的原生方法，真正的跳跃逻辑（改速度、播动画、走 `CharacterMovementComponent`）。

> **这就是"适配器"的本质**：本类不实现任何跳跃物理，只是把引擎的 `Jump()` / `StopJumping()` **包进 GAS 的激活/结束生命周期**，并加上网络预测、权限判断、幂等保护。

## 五、几个要记住的点

1. **`LocalPredicted` 是手感关键**：玩家操作类技能标配，对比阶段技能用 `ServerInitiated`，策略选择完全相反。
2. **`CanActivateAbility` 三层守卫**，最后必须问 `Super::`（漏了会绕过激活组/Tag 规则）。
3. **`EndAbility` 无条件兜底 `CharacterJumpStop()`**：内部有 `bPressedJump` 判断所以安全。
4. **`UnCrouch()` 只在 Start 里**：蹲着跳要先起身。
5. **`IsLocallyControlled()` 权限判断**：不操作别人的 Pawn。
6. **局部/成员名里的 `Lyra` 也要换**：`LyraCharacter` → `TenkaichiCharacter`（铁律 24）。
7. **`.cpp` 的基类 include 路径和 `.h` 不同**：`.cpp` 里带 `AbilitySystem/Abilities/` 前缀。
