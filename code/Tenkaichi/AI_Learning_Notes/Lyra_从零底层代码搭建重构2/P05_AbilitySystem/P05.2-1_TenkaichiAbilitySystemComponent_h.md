```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#pragma once

#include "Abilities/TenkaichiGameplayAbility.h"
#include "AbilitySystemComponent.h"
#include "NativeGameplayTags.h"

#include "TenkaichiAbilitySystemComponent.generated.h"

// 把 UE_API 宏临时替换成项目的导出宏（TENKAICHIGAME_API），文件结束再还原成引擎默认
#define UE_API TENKAICHIGAME_API

class AActor;
class UGameplayAbility;
class UTenkaichiAbilityTagRelationshipMapping;
class UObject;
struct FFrame;
struct FGameplayAbilityTargetDataHandle;

// 声明"技能输入被阻塞"这个全局游戏标签（实际定义在 .cpp）
TENKAICHIGAME_API UE_DECLARE_GAMEPLAY_TAG_EXTERN(TAG_Gameplay_AbilityInputBlocked);

/**
 * UTenkaichiAbilitySystemComponent
 *
 *	本项目使用的能力系统组件基类。
 */
UCLASS(MinimalAPI)
class UTenkaichiAbilitySystemComponent : public UAbilitySystemComponent
{
	GENERATED_BODY()

public:

	UE_API UTenkaichiAbilitySystemComponent(const FObjectInitializer& ObjectInitializer = FObjectInitializer::Get());

	//~UActorComponent 接口开始
	UE_API virtual void EndPlay(const EEndPlayReason::Type EndPlayReason) override;
	//~UActorComponent 接口结束

	UE_API virtual void InitAbilityActorInfo(AActor* InOwnerActor, AActor* InAvatarActor) override;

	// 取消技能用的函数类型：传入"是否取消该技能"的判断回调
	typedef TFunctionRef<bool(const UTenkaichiGameplayAbility* TenkaichiAbility, FGameplayAbilitySpecHandle Handle)> TShouldCancelAbilityFunc;
	UE_API void CancelAbilitiesByFunc(TShouldCancelAbilityFunc ShouldCancelFunc, bool bReplicateCancelAbility);

	// 取消所有"输入触发/按住激活"的技能
	UE_API void CancelInputActivatedAbilities(bool bReplicateCancelAbility);

	// 输入标签被按下：把输入标签翻译成"待激活"的技能句柄
	UE_API void AbilityInputTagPressed(const FGameplayTag& InputTag);
	// 输入标签被松开：把输入标签翻译成"待释放"的技能句柄
	UE_API void AbilityInputTagReleased(const FGameplayTag& InputTag);

	// 每帧处理输入：按住/按下/松开，批量激活技能
	UE_API void ProcessAbilityInput(float DeltaTime, bool bGamePaused);
	// 清空输入句柄缓存
	UE_API void ClearAbilityInput();

	// 激活组是否被阻塞（排他组被占用）
	UE_API bool IsActivationGroupBlocked(ELyraAbilityActivationGroup Group) const;
	// 技能加入激活组（计数+1，排他时取消同组其他）
	UE_API void AddAbilityToActivationGroup(ELyraAbilityActivationGroup Group, UTenkaichiGameplayAbility* TenkaichiAbility);
	// 技能离开激活组（计数-1）
	UE_API void RemoveAbilityFromActivationGroup(ELyraAbilityActivationGroup Group, UTenkaichiGameplayAbility* TenkaichiAbility);
	// 取消指定激活组的所有技能
	UE_API void CancelActivationGroupAbilities(ELyraAbilityActivationGroup Group, UTenkaichiGameplayAbility* IgnoreTenkaichiAbility, bool bReplicateCancelAbility);

	// 用一个 GameplayEffect 添加指定的动态授予标签
	UE_API void AddDynamicTagGameplayEffect(const FGameplayTag& Tag);

	// 移除所有用来添加指定动态授予标签的 GameplayEffect 实例
	UE_API void RemoveDynamicTagGameplayEffect(const FGameplayTag& Tag);

	// 取与给定技能句柄和激活信息关联的技能目标数据
	UE_API void GetAbilityTargetData(const FGameplayAbilitySpecHandle AbilityHandle, FGameplayAbilityActivationInfo ActivationInfo, FGameplayAbilityTargetDataHandle& OutTargetDataHandle);

	// 设置当前的标签关系映射表，如果传 null 则清空它
	UE_API void SetTagRelationshipMapping(UTenkaichiAbilityTagRelationshipMapping* NewMapping);

	// 查看技能标签，收集额外的"必需标签"和"阻塞标签"
	UE_API void GetAdditionalActivationTagRequirements(const FGameplayTagContainer& AbilityTags, FGameplayTagContainer& OutActivationRequired, FGameplayTagContainer& OutActivationBlocked) const;

	// 遍历技能，触发 OnSpawn（出生即触发）策略的技能激活
	UE_API void TryActivateAbilitiesOnSpawn();

protected:

	UE_API virtual void AbilitySpecInputPressed(FGameplayAbilitySpec& Spec) override;
	UE_API virtual void AbilitySpecInputReleased(FGameplayAbilitySpec& Spec) override;

	UE_API virtual void NotifyAbilityActivated(const FGameplayAbilitySpecHandle Handle, UGameplayAbility* Ability) override;
	UE_API virtual void NotifyAbilityFailed(const FGameplayAbilitySpecHandle Handle, UGameplayAbility* Ability, const FGameplayTagContainer& FailureReason) override;
	UE_API virtual void NotifyAbilityEnded(FGameplayAbilitySpecHandle Handle, UGameplayAbility* Ability, bool bWasCancelled) override;
	UE_API virtual void ApplyAbilityBlockAndCancelTags(const FGameplayTagContainer& AbilityTags, UGameplayAbility* RequestingAbility, bool bEnableBlockTags, const FGameplayTagContainer& BlockTags, bool bExecuteCancelTags, const FGameplayTagContainer& CancelTags) override;
	UE_API virtual void HandleChangeAbilityCanBeCanceled(const FGameplayTagContainer& AbilityTags, UGameplayAbility* RequestingAbility, bool bCanBeCanceled) override;

	// 通知客户端：某个技能激活失败
	UFUNCTION(Client, Unreliable)
	UE_API void ClientNotifyAbilityFailed(const UGameplayAbility* Ability, const FGameplayTagContainer& FailureReason);

	UE_API void HandleAbilityFailed(const UGameplayAbility* Ability, const FGameplayTagContainer& FailureReason);
protected:

	// 若设置了这个表，就用它查找激活/取消的标签关系
	UPROPERTY()
	TObjectPtr<UTenkaichiAbilityTagRelationshipMapping> TagRelationshipMapping;

	// 本帧输入被按下的技能句柄
	TArray<FGameplayAbilitySpecHandle> InputPressedSpecHandles;

	// 本帧输入被松开的技能句柄
	TArray<FGameplayAbilitySpecHandle> InputReleasedSpecHandles;

	// 输入一直被按住的技能句柄
	TArray<FGameplayAbilitySpecHandle> InputHeldSpecHandles;

	// 每个激活组正在运行的技能数量
	int32 ActivationGroupCounts[(uint8)ELyraAbilityActivationGroup::MAX];
};

// 文件末尾还原 UE_API 为引擎默认（不要影响其他文件）
#undef UE_API
```

## 四、逐段讲解

### 1. 开头的 `#define UE_API` / `#undef UE_API`（同 P05.3-B 技巧）

```cpp
#define UE_API TENKAICHIGAME_API
...
#undef UE_API
```

- 引擎里 `UE_API` 默认是 `ENGINE_API`。Lyra 把它**临时替换成 `LYRAGAME_API`**，让本文件所有标记 `UE_API` 的函数都导出项目模块的 API。
- 换成我们项目：`LYRAGAME_API` → `TENKAICHIGAME_API`。
- **`#undef UE_API` 必须在文件末尾**：还原回引擎默认，避免污染下一个 include 该头的文件。

### 2. include 的三样东西

```cpp
#include "Abilities/TenkaichiGameplayAbility.h"   // P05.3-B 技能基类（带来 ELyraAbilityActivationGroup）
#include "AbilitySystemComponent.h"                // 引擎 GAS 能力组件基类
#include "NativeGameplayTags.h"                    // 原生游戏标签宏（UE_DECLARE_GAMEPLAY_TAG_EXTERN）
```

| include | 用途 |
|---------|------|
| `TenkaichiGameplayAbility.h` | **本类继承链的根**（`UGameplayAbility`）的来源，还带来 `ELyraAbilityActivationGroup` 枚举和 `UTenkaichiGameplayAbility` 完整定义（`TShouldCancelAbilityFunc` 用） |
| `AbilitySystemComponent.h` | 引擎 `UAbilitySystemComponent`——本类直接继承它 |
| `NativeGameplayTags.h` | 提供 `UE_DECLARE_GAMEPLAY_TAG_EXTERN` 宏，声明全局标签 |

### 3. 声明全局标签 `TAG_Gameplay_AbilityInputBlocked`

```cpp
TENKAICHIGAME_API UE_DECLARE_GAMEPLAY_TAG_EXTERN(TAG_Gameplay_AbilityInputBlocked);
```

- 一个**全局游戏标签**"Gameplay.AbilityInputBlocked"（输入被阻塞）。
- `UE_DECLARE_GAMEPLAY_TAG_EXTERN` = 只在 `.h` 里**声明**（extern），真正的定义在 `.cpp`（`UE_DEFINE_GAMEPLAY_TAG(TAG_Gameplay_AbilityInputBlocked, "Gameplay.AbilityInputBlocked")`）。下次教 `.cpp` 时定义它。
- 前面加 `TENKAICHIGAME_API` 导出宏，让别的模块（UI 等）也能引用这个标签。

### 4. 类声明

```cpp
UCLASS(MinimalAPI)
class UTenkaichiAbilitySystemComponent : public UAbilitySystemComponent
{
	GENERATED_BODY()
```

- 继承**引擎 `UAbilitySystemComponent`**——GAS 的能力组件基类，本类在其上加 Lyra 定制（输入处理/激活组/标签关系）。
- `MinimalAPI`：只导出类名级别的最小 API（函数用 `UE_API` 逐个导出），加快编译。

### 5. 公开方法组（输入处理 + 激活组 + 标签）

`.h` 里的公开方法分四组，职责一目了然：

| 方法 | 一句话职责 |
|------|-----------|
| `EndPlay` | 组件销毁时从全局系统注销 ASC（挂起） |
| `InitAbilityActorInfo` | 设置 Owner/Avatar 时通知技能新化身（部分挂起） |
| `CancelAbilitiesByFunc` | 按回调条件取消一批技能 |
| `CancelInputActivatedAbilities` | 取消所有输入触发/按住激活的技能 |
| `AbilityInputTagPressed/Released` | 输入标签 → 待激活/待释放的技能句柄 |
| `ProcessAbilityInput` | **每帧核心**：处理按住/按下/松开，批量激活 |
| `ClearAbilityInput` | 清空输入句柄缓存 |
| `IsActivationGroupBlocked` | 激活组是否被排他占用 |
| `Add/RemoveAbilityToActivationGroup` | 技能进/出激活组（计数增减） |
| `CancelActivationGroupAbilities` | 取消某激活组全部技能 |
| `Add/RemoveDynamicTagGameplayEffect` | 用效果动态加/删标签（挂起） |
| `GetAbilityTargetData` | 取技能目标数据 |
| `SetTagRelationshipMapping` | 设置 Tag 关系映射表（P05.1） |
| `GetAdditionalActivationTagRequirements` | 用映射展开技能附加的 required/blocked 标签 |
| `TryActivateAbilitiesOnSpawn` | 遍历技能触发 OnSpawn 激活 |

- **`TShouldCancelAbilityFunc`**：一个函数类型（`TFunctionRef<bool(const UTenkaichiGameplayAbility*, FGameplayAbilitySpecHandle)>`），`CancelAbilitiesByFunc` 收一个"判断是否取消"的回调，遍历技能时对每个技能调用它。
- **`ELyraAbilityActivationGroup`**：P05.3-B 教的激活组枚举（Independent / Exclusive_Replaceable / Exclusive_Blocking），这里用它做激活组管理。

### 6. protected 的 override（引擎虚函数 + 客户端通知）

```cpp
AbilitySpecInputPressed/Released     // 输入按下/松开时派发事件给技能实例
NotifyAbilityActivated/Ended         // 技能激活/结束时增减激活组计数
NotifyAbilityFailed                  // 技能失败时路由到本地处理或客户端通知
ApplyAbilityBlockAndCancelTags       // 用 Tag 映射展开 block/cancel 标签
HandleChangeAbilityCanBeCanceled     // 处理技能可取消状态变化
ClientNotifyAbilityFailed            // 客户端收到技能失败通知（UFUNCTION Client, Unreliable）
HandleAbilityFailed                  // 调技能本体的失败回调（触发 P05.3-B 的失败消息）
```

| 要点 | 含义 |
|------|------|
| `override` | 覆盖引擎 `UAbilitySystemComponent` 的虚函数，实现 GAS 定制的回调 |
| `UFUNCTION(Client, Unreliable)` | **RPC**：只在客户端执行，不可靠传输（失败通知丢了也无妨）。服务端调它，客户端收到后走 `HandleAbilityFailed` |

### 7. protected 数据成员

```cpp
TObjectPtr<UTenkaichiAbilityTagRelationshipMapping> TagRelationshipMapping;  // 标签关系映射表（P05.1）
TArray<FGameplayAbilitySpecHandle> InputPressedSpecHandles;   // 本帧按下的技能句柄
TArray<FGameplayAbilitySpecHandle> InputReleasedSpecHandles;  // 本帧松开的技能句柄
TArray<FGameplayAbilitySpecHandle> InputHeldSpecHandles;      // 一直按住的技能句柄
int32 ActivationGroupCounts[(uint8)ELyraAbilityActivationGroup::MAX];  // 每个激活组的运行数
```

- **`TagRelationshipMapping`**（`UPROPERTY`）：指向 P05.1 已教 `UTenkaichiAbilityTagRelationshipMapping`。`.h` 只存指针，前置声明就够，所以 **P05.1 不必在这里 include**。
- **三个输入句柄数组**：`InputPressed/Released/Held` 记录输入事件对应的技能句柄，`ProcessAbilityInput` 每帧读它们来批量激活/释放。这就是"输入→技能"的中转站。
- **`ActivationGroupCounts`**：按 `ELyraAbilityActivationGroup` 枚举的下标（`MAX` = 3）统计每个激活组正在跑几个技能。排他组计数 > 0 就 `IsActivationGroupBlocked`。

### 8. 前置声明（`.h` 为什么能独立编译）

```cpp
class AActor;
class UGameplayAbility;
class UTenkaichiAbilityTagRelationshipMapping;  // P05.1 ✅ 只碰指针
class UObject;
struct FFrame;
struct FGameplayAbilityTargetDataHandle;
```

- `UTenkaichiAbilityTagRelationshipMapping`（P05.1）和 `UGameplayAbility`（引擎）在这里**只用了指针/引用**，前置声明就够，不需要完整定义——**`.h` 能独立编译**。
- 真正需要完整定义的是 `UTenkaichiGameplayAbility`（`TShouldCancelAbilityFunc` 里用了它的指针，但 `GetAbilitySource` 等需要完整定义时在 `.cpp` include）。
- **挂起不在这里**：ASC `.h` 的依赖（P05.3-B GameplayAbility ✅ / P05.1 TagRelationshipMapping ✅ / 引擎 GAS ✅）全就位，所以 `.h` **零挂起**。

## 五、几个要记住的点

1. **`#define UE_API TENKAICHIGAME_API`**：同 P05.3-B 技巧，把 UE_API 临时替换成项目导出宏，末尾 `#undef` 还原。
2. **继承 `UAbilitySystemComponent`**：引擎 GAS 能力组件 → 本项目 ASC。本类是 GAS 的"中央处理器"。
3. **输入三件套**：`InputPressed/Released/Held` 三个句柄数组 + `ProcessAbilityInput` 每帧驱动 = "输入→技能"的核心机制。
4. **激活组管理**：`ActivationGroupCounts[]` 按枚举下标统计，配合 `Add/RemoveAbilityToActivationGroup` 实现技能互斥。
5. **标签关系映射**：`SetTagRelationshipMapping` / `GetAdditionalActivationTagRequirements` / `ApplyAbilityBlockAndCancelTags` 都是接 P05.1 的表做 Tag 展开。
6. **`TAG_Gameplay_AbilityInputBlocked`**：`.h` 只声明（extern），`.cpp` 才定义。
7. **`.h` 零挂起**：依赖（P05.3-B / P05.1 / 引擎 GAS）全就位。挂起全在下次 `.cpp`。
8. **接口分界标记中文化**：`//~UActorComponent 接口开始/结束`。
