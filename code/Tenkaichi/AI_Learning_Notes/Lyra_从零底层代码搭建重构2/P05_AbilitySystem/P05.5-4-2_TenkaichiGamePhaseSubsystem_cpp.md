	UWorld* World = GetWorld();

	if (ensure(GameState_ASC))
	{
		FGameplayAbilitySpec PhaseSpec(PhaseAbility, 1, 0, this);

		{
			FTenkaichiGamePhaseEntry& Entry = ActivePhaseMap.FindOrAdd(SpecHandle);
			Entry.PhaseEndedCallback = PhaseEndedCallback;
		}
		else
		{
			PhaseEndedCallback.ExecuteIfBound(nullptr);
		}
	}
}

	{
		const FTenkaichiGamePhaseEntry& PhaseEntry = KVP.Value;

// 新阶段开始：结束所有非祖先的活跃阶段，登记本阶段，并通知开始观察者
void UTenkaichiGamePhaseSubsystem::OnBeginPhase(const UTenkaichiGamePhaseAbility* PhaseAbility, const FGameplayAbilitySpecHandle PhaseAbilityHandle)
{
	const FGameplayTag IncomingPhaseTag = PhaseAbility->GetGamePhaseTag();

	UE_LOG(LogTenkaichiGamePhase, Log, TEXT("Beginning Phase '%s' (%s)"), *IncomingPhaseTag.ToString(), *GetNameSafe(PhaseAbility));

	const UWorld* World = GetWorld();
	UTenkaichiAbilitySystemComponent* GameState_ASC = World->GetGameState()->FindComponentByClass<UTenkaichiAbilitySystemComponent>();
	if (ensure(GameState_ASC))
	{
		TArray<FGameplayAbilitySpec*> ActivePhases;
		for (const auto& KVP : ActivePhaseMap)
		{
			const FGameplayAbilitySpecHandle ActiveAbilityHandle = KVP.Key;
			if (FGameplayAbilitySpec* Spec = GameState_ASC->FindAbilitySpecFromHandle(ActiveAbilityHandle))
			{
				ActivePhases.Add(Spec);
			}
		}

		for (const FGameplayAbilitySpec* ActivePhase : ActivePhases)
		{
			const UTenkaichiGamePhaseAbility* ActivePhaseAbility = CastChecked<UTenkaichiGamePhaseAbility>(ActivePhase->Ability);
			const FGameplayTag ActivePhaseTag = ActivePhaseAbility->GetGamePhaseTag();
			
			// 如果当前活跃阶段与新阶段标签匹配（即它是新阶段的祖先），就允许它继续存在。
			// 也就是说，多个技能可以关联到同一个阶段标签。
			// 举例：
			// 你可以处于 Game.Playing 阶段，然后启动一个子阶段，比如 Game.Playing.SuddenDeath。
			// Game.Playing 阶段仍然活跃，如果再启动一个，比如 Game.Playing.ActualSuddenDeath，
			// 它会结束 Game.Playing.SuddenDeath 阶段，但 Game.Playing 会继续存在。
			// 类似地，如果我们激活 Game.GameOver，所有 Game.Playing* 阶段都会结束。
			if (!IncomingPhaseTag.MatchesTag(ActivePhaseTag))
			{
				UE_LOG(LogTenkaichiGamePhase, Log, TEXT("\tEnding Phase '%s' (%s)"), *ActivePhaseTag.ToString(), *GetNameSafe(ActivePhaseAbility));

				FGameplayAbilitySpecHandle HandleToEnd = ActivePhase->Handle;
				GameState_ASC->CancelAbilitiesByFunc([HandleToEnd](const UTenkaichiGameplayAbility* TenkaichiAbility, FGameplayAbilitySpecHandle Handle) {
					return Handle == HandleToEnd;
				}, true);
			}
		}

		FTenkaichiGamePhaseEntry& Entry = ActivePhaseMap.FindOrAdd(PhaseAbilityHandle);
		Entry.PhaseTag = IncomingPhaseTag;

		// 通知所有观察本阶段的观察者：阶段已开始
		for (const FPhaseObserver& Observer : PhaseStartObservers)
		{
			if (Observer.IsMatch(IncomingPhaseTag))
			{
				Observer.PhaseCallback.ExecuteIfBound(IncomingPhaseTag);
			}
		}
	}
}

// 阶段结束：触发结束回调、从活跃表移除，并通知结束观察者
void UTenkaichiGamePhaseSubsystem::OnEndPhase(const UTenkaichiGamePhaseAbility* PhaseAbility, const FGameplayAbilitySpecHandle PhaseAbilityHandle)
{
	const FGameplayTag EndedPhaseTag = PhaseAbility->GetGamePhaseTag();
	UE_LOG(LogTenkaichiGamePhase, Log, TEXT("Ended Phase '%s' (%s)"), *EndedPhaseTag.ToString(), *GetNameSafe(PhaseAbility));

	const FTenkaichiGamePhaseEntry& Entry = ActivePhaseMap.FindChecked(PhaseAbilityHandle);
	Entry.PhaseEndedCallback.ExecuteIfBound(PhaseAbility);

	ActivePhaseMap.Remove(PhaseAbilityHandle);

	// 通知所有观察本阶段的观察者：阶段已结束
	for (const FPhaseObserver& Observer : PhaseEndObservers)
	{
		if (Observer.IsMatch(EndedPhaseTag))
		{
			Observer.PhaseCallback.ExecuteIfBound(EndedPhaseTag);
		}
	}
}

// 判断给定阶段标签是否符合本观察者的匹配规则（精确匹配 / 部分匹配）
bool UTenkaichiGamePhaseSubsystem::FPhaseObserver::IsMatch(const FGameplayTag& ComparePhaseTag) const
{
	switch(MatchType)
	{
	case EPhaseTagMatchType::ExactMatch:
		return ComparePhaseTag == PhaseTag;
	case EPhaseTagMatchType::PartialMatch:
		return ComparePhaseTag.MatchesTag(PhaseTag);
	}

	return false;
}

### 3. StartPhase（启动阶段）

```cpp

```

**关键步骤**：

1. **`World->GetGameState()->FindComponentByClass<UTenkaichiAbilitySystemComponent>()`** —— 从 GameState 上找 ASC。⚠️ **这就是为什么阶段 GA 挂在 GameState 而不是角色上**。
2. **`ensure(GameState_ASC)`** —— ⚠️ 新宏（铁律 18）：**运行时断言**。条件为假时**不崩溃**，但会弹一次错误上报（打包版只记日志）。比 `check()` 温和（check 直接崩溃）。因为"GameState 没挂 ASC"是**配置问题**不是逻辑 bug，不该让整个游戏崩。
3. **`FGameplayAbilitySpec PhaseSpec(PhaseAbility, 1, 0, this)`** —— 构造技能规格：技能类、等级 1、输入 ID 0、**SourceObject 是 `this`（本 Subsystem）**。
4. **`GiveAbilityAndActivateOnce(PhaseSpec)`** —— P05.2 教的 ASC 方法：**授予并立刻激活一次**，返回句柄。这**就是"阶段开始"**。
5. **`FindAbilitySpecFromHandle(SpecHandle)`** —— 反查规格，确认它确实激活了。
6. **`if (FoundSpec && FoundSpec->IsActive())`** → `ActivePhaseMap.FindOrAdd(SpecHandle)` 存回调。
7. **`else` 分支**：激活失败（比如被 Tag 阻止）→ `PhaseEndedCallback.ExecuteIfBound(nullptr)` —— **立刻以 nullptr 通知调用方"这阶段没起来"**，避免调用方傻等。

### 4. K2_ 三个桥接函数（套路完全一致）

```cpp
void UTenkaichiGamePhaseSubsystem::K2_StartPhase(TSubclassOf<UTenkaichiGamePhaseAbility> PhaseAbility, const FTenkaichiGamePhaseDynamicDelegate& PhaseEndedDelegate)
{
	const FTenkaichiGamePhaseDelegate EndedDelegate = FTenkaichiGamePhaseDelegate::CreateWeakLambda(const_cast<UObject*>(PhaseEndedDelegate.GetUObject()), [PhaseEndedDelegate](const UTenkaichiGamePhaseAbility* PhaseAbility) {
		PhaseEndedDelegate.ExecuteIfBound(PhaseAbility);
	});

	StartPhase(PhaseAbility, EndedDelegate);
}
```

- **套路**：动态委托 → `CreateWeakLambda` 包成原生委托 → 调 C++ 主函数。
- **`CreateWeakLambda(对象, Lambda)`** —— ⚠️ 新方法（铁律 18）：
  - 创建一个**弱引用**绑定到指定对象的 Lambda 委托。
  - **"弱"的意思**：如果那个对象被销毁了，这个回调**自动失效、不会崩**（对比 `CreateRaw` 会在对象销毁后野指针崩溃）。
  - 这是 UE 委托里**最常用、最安全**的绑定方式之一。
- **`const_cast<UObject*>(PhaseEndedDelegate.GetUObject())`** —— 动态委托的 `GetUObject()` 返回 `const`，而 `CreateWeakLambda` 要非 const，所以 `const_cast` 去掉 const。
- **Lambda 里只做一件事**：`PhaseEndedDelegate.ExecuteIfBound(PhaseAbility);` —— 把原生委托的调用**转发**给动态委托。
- **三个 `K2_` 函数结构一模一样**，只是委托类型/参数不同。照抄。

### 5. WhenPhaseStartsOrIsActive / WhenPhaseEnds（C++ 注册）

```cpp
void UTenkaichiGamePhaseSubsystem::WhenPhaseStartsOrIsActive(FGameplayTag PhaseTag, EPhaseTagMatchType MatchType, const FTenkaichiGamePhaseTagDelegate& WhenPhaseActive)
{
	FPhaseObserver Observer;
	Observer.PhaseTag = PhaseTag;
	Observer.MatchType = MatchType;
	Observer.PhaseCallback = WhenPhaseActive;
	PhaseStartObservers.Add(Observer);

	if (IsPhaseActive(PhaseTag))
	{
		WhenPhaseActive.ExecuteIfBound(PhaseTag);
	}

```

- **建一个 `FPhaseObserver` 塞进数组**（三个字段逐个赋值，Lyra 没用聚合初始化，照抄）。
- ⚠️ **`if (IsPhaseActive(PhaseTag))` 立即触发** —— 这就是函数名里 **"OrIsActive"** 的含义：**如果订阅时该阶段已经活跃了，马上补一次通知**。否则订阅者会永久错过"在我订阅之前就开始的阶段"。
- **`WhenPhaseEnds` 没有这个 `if`** —— 因为"结束"是未来事件，不存在"已经结束了"要补的情况。
- **`ExecuteIfBound`** —— 委托的安全调用：**绑定了才执行**，没绑定就静默跳过（不用判空）。

### 6. IsPhaseActive

```cpp
bool UTenkaichiGamePhaseSubsystem::IsPhaseActive(const FGameplayTag& PhaseTag) const
{
	for (const auto& KVP : ActivePhaseMap)
	{
		const FTenkaichiGamePhaseEntry& PhaseEntry = KVP.Value;
		if (PhaseEntry.PhaseTag.MatchesTag(PhaseTag))
		{
			return true;
		}
	}

```

- **遍历 `ActivePhaseMap`**，`KVP` = Key-Value Pair（键是句柄，值是阶段条目）。
- **`PhaseEntry.PhaseTag.MatchesTag(PhaseTag)`** —— ⚠️ **核心方法**：`MatchesTag` 判断**前者是否是后者的后代（或相等）**。
  - `Game.Playing.WarmUp`.MatchesTag(`Game.Playing`) → **true**（是后代）
  - `Game.Playing`.MatchesTag(`Game.Playing.WarmUp`) → **false**（父不匹配子）
- 所以**查 `Game.Playing` 时，它的所有子阶段都算"活跃"**——符合"父子共存"的设计。

### 7. OnBeginPhase（★ 本文件核心，互斥判定）

```cpp
void UTenkaichiGamePhaseSubsystem::OnBeginPhase(const UTenkaichiGamePhaseAbility* PhaseAbility, const FGameplayAbilitySpecHandle PhaseAbilityHandle)
{
	const FGameplayTag IncomingPhaseTag = PhaseAbility->GetGamePhaseTag();

	UE_LOG(LogTenkaichiGamePhase, Log, TEXT("Beginning Phase '%s' (%s)"), *IncomingPhaseTag.ToString(), *GetNameSafe(PhaseAbility));

	const UWorld* World = GetWorld();
	UTenkaichiAbilitySystemComponent* GameState_ASC = World->GetGameState()->FindComponentByClass<UTenkaichiAbilitySystemComponent>();
	if (ensure(GameState_ASC))
	{
		// 第一步：收集当前活跃阶段的 Spec
		TArray<FGameplayAbilitySpec*> ActivePhases;
		for (const auto& KVP : ActivePhaseMap)
		{
			const FGameplayAbilitySpecHandle ActiveAbilityHandle = KVP.Key;
			if (FGameplayAbilitySpec* Spec = GameState_ASC->FindAbilitySpecFromHandle(ActiveAbilityHandle))
			{
				ActivePhases.Add(Spec);
			}
		}

		// 第二步：遍历，结束"不是新阶段祖先"的那些
		for (const FGameplayAbilitySpec* ActivePhase : ActivePhases)
		{
			const UTenkaichiGamePhaseAbility* ActivePhaseAbility = CastChecked<UTenkaichiGamePhaseAbility>(ActivePhase->Ability);
			const FGameplayTag ActivePhaseTag = ActivePhaseAbility->GetGamePhaseTag();
			
			if (!IncomingPhaseTag.MatchesTag(ActivePhaseTag))
			{
				UE_LOG(LogTenkaichiGamePhase, Log, TEXT("\tEnding Phase '%s' (%s)"), *ActivePhaseTag.ToString(), *GetNameSafe(ActivePhaseAbility));

				FGameplayAbilitySpecHandle HandleToEnd = ActivePhase->Handle;
				GameState_ASC->CancelAbilitiesByFunc([HandleToEnd](const UTenkaichiGameplayAbility* TenkaichiAbility, FGameplayAbilitySpecHandle Handle) {
					return Handle == HandleToEnd;
				}, true);
			}
		}

		// 第三步：把新阶段登记进活跃表
		FTenkaichiGamePhaseEntry& Entry = ActivePhaseMap.FindOrAdd(PhaseAbilityHandle);
		Entry.PhaseTag = IncomingPhaseTag;

		// 第四步：通知开始观察者
		for (const FPhaseObserver& Observer : PhaseStartObservers)
		{
			if (Observer.IsMatch(IncomingPhaseTag))
			{
				Observer.PhaseCallback.ExecuteIfBound(IncomingPhaseTag);
			}
		}
	}
}
```

**⚠️★ 核心判定就一行**：

```cpp
if (!IncomingPhaseTag.MatchesTag(ActivePhaseTag))
```

- **`IncomingPhaseTag.MatchesTag(ActivePhaseTag)`** = **新阶段是否是老活跃阶段的后代**（即老阶段是新阶段的祖先）。
  - 例：新阶段 `Game.Playing.PostGame`，老阶段 `Game.Playing` → `PostGame`.MatchesTag(`Playing`) = **true** → 老阶段是**祖先** → **保留**（不进 if）。
  - 例：新阶段 `Game.Playing.PostGame`，老阶段 `Game.Playing.CaptureTheFlag` → `PostGame`.MatchesTag(`CaptureTheFlag`) = **false** → 老阶段是**兄弟** → **结束**（进 if）。
- **加 `!` 取反**：所以 `if` 里的是"**不是祖先** → 结束"。

**⚠️ 为什么要先收集成 `ActivePhases` 数组再遍历（两步走）？**

因为第二步里 `CancelAbilitiesByFunc` 会**触发 `OnEndPhase` → 修改 `ActivePhaseMap`**。如果直接在遍历 `ActivePhaseMap` 的同时修改它，**迭代器失效崩溃**。所以先**快照**到数组，再遍历。**这是 UE/C++ 容器操作的经典安全写法**。

**其他要点**：
- **`CastChecked<UTenkaichiGamePhaseAbility>(ActivePhase->Ability)`** —— ⚠️ 新方法（铁律 18）：**强制转换 + 断言**。转换失败会**崩溃**（不像 `Cast` 返回 nullptr）。这里能用它是因为"活跃阶段表里的必然是阶段技能"，不成立就是 bug，该崩。
- **`CancelAbilitiesByFunc(Lambda, true)`** —— P05.2 教的 ASC 方法：按 Lambda 条件取消技能。Lambda 捕获 `HandleToEnd`，返回 `Handle == HandleToEnd` 决定哪个该取消。第二个参数 `true` = **强制取消**。
- **`GetNameSafe(PhaseAbility)`** —— 安全取名（对象为空返回 `"None"`，不崩）。
- **`UE_LOG` 的 `\t`**：结束日志缩进一格，视觉上从属于"Beginning Phase"日志。
- **`ActivePhaseMap.FindOrAdd(PhaseAbilityHandle)`** —— 不存在就创建、存在就返回引用。**注意 `StartPhase` 里也调了一次**（那时只设回调不设 Tag），这里**补上 `PhaseTag`**。

### 8. OnEndPhase

```cpp
void UTenkaichiGamePhaseSubsystem::OnEndPhase(const UTenkaichiGamePhaseAbility* PhaseAbility, const FGameplayAbilitySpecHandle PhaseAbilityHandle)
{
	const FGameplayTag EndedPhaseTag = PhaseAbility->GetGamePhaseTag();
	UE_LOG(LogTenkaichiGamePhase, Log, TEXT("Ended Phase '%s' (%s)"), *EndedPhaseTag.ToString(), *GetNameSafe(PhaseAbility));

	const FTenkaichiGamePhaseEntry& Entry = ActivePhaseMap.FindChecked(PhaseAbilityHandle);
	Entry.PhaseEndedCallback.ExecuteIfBound(PhaseAbility);

	ActivePhaseMap.Remove(PhaseAbilityHandle);

	for (const FPhaseObserver& Observer : PhaseEndObservers)
	{
		if (Observer.IsMatch(EndedPhaseTag))
		{
			Observer.PhaseCallback.ExecuteIfBound(EndedPhaseTag);
		}
	}
}
```

- **`FindChecked`** —— ⚠️ 和 `FindOrAdd` 不同：`FindChecked` 要求**必须存在**，不存在就**断言崩溃**（本函数被调用时该阶段必然在表里）。
- **`Entry.PhaseEndedCallback.ExecuteIfBound(PhaseAbility)`** —— 触发 `StartPhase` 时传进来的那个"阶段结束回调"。
- **`ActivePhaseMap.Remove(...)`** —— 从活跃表移除。
- **通知 `PhaseEndObservers`**（和 `OnBeginPhase` 第四步对称）。

### 9. FPhaseObserver::IsMatch

```cpp
bool UTenkaichiGamePhaseSubsystem::FPhaseObserver::IsMatch(const FGameplayTag& ComparePhaseTag) const
{
	switch(MatchType)
	{
	case EPhaseTagMatchType::ExactMatch:
		return ComparePhaseTag == PhaseTag;
	case EPhaseTagMatchType::PartialMatch:
		return ComparePhaseTag.MatchesTag(PhaseTag);
	}

	return false;
}
```

- ⚠️ **注意函数名的嵌套作用域**：`FPhaseObserver` 是**类的内部结构体**，所以定义时写 `UTenkaichiGamePhaseSubsystem::FPhaseObserver::IsMatch`（两级 `::`）。
- **`switch` 两种匹配**：
  - `ExactMatch` → `ComparePhaseTag == PhaseTag`（**完全相等**）
  - `PartialMatch` → `ComparePhaseTag.MatchesTag(PhaseTag)`（`ComparePhaseTag` 是 `PhaseTag` 的**后代**）
- **最后的 `return false;`** —— 兜底（枚举值异常时）。⚠️ 虽然 `switch` 覆盖了所有枚举值，但编译器不知道，**没有这行会警告/报错**。

## 六、几个要记住的点

1. **★ 互斥判定就一行**：`if (!IncomingPhaseTag.MatchesTag(ActivePhaseTag))` —— 不是祖先就结束。
2. **★ 两步走是为了容器安全**：先快照 `ActivePhases` 数组，再遍历取消（取消会修改 Map，直接遍历会崩）。
3. **`ensure` vs `check` vs `CastChecked`**：`ensure` 温和（不崩，只上报）、`CastChecked`/`FindChecked` 强硬（不成立就崩）。
4. **`DEFINE_LOG_CATEGORY(LogTenkaichiGamePhase)` 写在本文件** —— P05.5-4-3 只有 `.h`。
5. **`MatchesTag` 是单向的**：子匹配父（`A.B`.MatchesTag(`A`) = true），父不匹配子。
6. **`CreateWeakLambda`**：对象销毁后自动失效，最安全的委托绑定方式。
7. **`ExecuteIfBound`**：委托的安全调用，没绑定就跳过。
8. **`ShouldCreateSubsystem` 里被注释的代码照抄保留**（铁律 23）。
9. **Log 头文件待建**：本组内部顺序，不登记备忘录；想立刻编译可临时注释 1 个 include + 3 处 `UE_LOG`。
