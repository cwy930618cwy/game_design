## 三、完整代码（注释已中文化）

```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#include "TenkaichiGlobalAbilitySystem.h"

#include "AbilitySystem/TenkaichiAbilitySystemComponent.h"

#include UE_INLINE_GENERATED_CPP_BY_NAME(TenkaichiGlobalAbilitySystem)

// 把全局技能授给指定的 ASC（若已授过，先移除旧的再授新的）
void FGlobalAppliedAbilityList::AddToASC(TSubclassOf<UGameplayAbility> Ability, UTenkaichiAbilitySystemComponent* ASC)
{
	if (FGameplayAbilitySpecHandle* SpecHandle = Handles.Find(ASC))
	{
		RemoveFromASC(ASC);
	}

	UGameplayAbility* AbilityCDO = Ability->GetDefaultObject<UGameplayAbility>();
	FGameplayAbilitySpec AbilitySpec(AbilityCDO);
	const FGameplayAbilitySpecHandle AbilitySpecHandle = ASC->GiveAbility(AbilitySpec);
	Handles.Add(ASC, AbilitySpecHandle);
}

// 从指定的 ASC 上移除这个全局技能
void FGlobalAppliedAbilityList::RemoveFromASC(UTenkaichiAbilitySystemComponent* ASC)
{
	if (FGameplayAbilitySpecHandle* SpecHandle = Handles.Find(ASC))
	{
		ASC->ClearAbility(*SpecHandle);
		Handles.Remove(ASC);
	}
}

// 从所有 ASC 上移除这个全局技能
void FGlobalAppliedAbilityList::RemoveFromAll()
{
	for (auto& KVP : Handles)
	{
		if (KVP.Key != nullptr)
		{
			KVP.Key->ClearAbility(KVP.Value);
		}
	}
	Handles.Empty();
}



// 把全局效果加给指定的 ASC（若已加过，先移除旧的再加新的）
void FGlobalAppliedEffectList::AddToASC(TSubclassOf<UGameplayEffect> Effect, UTenkaichiAbilitySystemComponent* ASC)
{
	if (FActiveGameplayEffectHandle* EffectHandle = Handles.Find(ASC))
	{
		RemoveFromASC(ASC);
	}

	const UGameplayEffect* GameplayEffectCDO = Effect->GetDefaultObject<UGameplayEffect>();
	const FActiveGameplayEffectHandle GameplayEffectHandle = ASC->ApplyGameplayEffectToSelf(GameplayEffectCDO, /*Level=*/ 1, ASC->MakeEffectContext());
	Handles.Add(ASC, GameplayEffectHandle);
}

// 从指定的 ASC 上移除这个全局效果
void FGlobalAppliedEffectList::RemoveFromASC(UTenkaichiAbilitySystemComponent* ASC)
{
	if (FActiveGameplayEffectHandle* EffectHandle = Handles.Find(ASC))
	{
		ASC->RemoveActiveGameplayEffect(*EffectHandle);
		Handles.Remove(ASC);
	}
}

// 从所有 ASC 上移除这个全局效果
void FGlobalAppliedEffectList::RemoveFromAll()
{
	for (auto& KVP : Handles)
	{
		if (KVP.Key != nullptr)
		{
			KVP.Key->RemoveActiveGameplayEffect(KVP.Value);
		}
	}
	Handles.Empty();
}

// 构造函数：全局技能系统的默认构造（无特殊初始化）
UTenkaichiGlobalAbilitySystem::UTenkaichiGlobalAbilitySystem()
{
}

// 把技能授给所有已注册的 ASC（仅服务器）
void UTenkaichiGlobalAbilitySystem::ApplyAbilityToAll(TSubclassOf<UGameplayAbility> Ability)
{
	if ((Ability.Get() != nullptr) && (!AppliedAbilities.Contains(Ability)))
	{
		FGlobalAppliedAbilityList& Entry = AppliedAbilities.Add(Ability);		
		for (ULyraAbilitySystemComponent* ASC : RegisteredASCs)
		{
			Entry.AddToASC(Ability, ASC);
		}
	}
}

// 把效果加给所有已注册的 ASC（仅服务器）
void UTenkaichiGlobalAbilitySystem::ApplyEffectToAll(TSubclassOf<UGameplayEffect> Effect)
{
	if ((Effect.Get() != nullptr) && (!AppliedEffects.Contains(Effect)))
	{
		FGlobalAppliedEffectList& Entry = AppliedEffects.Add(Effect);
		for (ULyraAbilitySystemComponent* ASC : RegisteredASCs)
		{
			Entry.AddToASC(Effect, ASC);
		}
	}
}

// 把所有 ASC 上的这个技能移除（仅服务器）
void UTenkaichiGlobalAbilitySystem::RemoveAbilityFromAll(TSubclassOf<UGameplayAbility> Ability)
{
	if ((Ability.Get() != nullptr) && AppliedAbilities.Contains(Ability))
	{
		FGlobalAppliedAbilityList& Entry = AppliedAbilities[Ability];
		Entry.RemoveFromAll();
		AppliedAbilities.Remove(Ability);
	}
}

// 把所有 ASC 上的这个效果移除（仅服务器）
void UTenkaichiGlobalAbilitySystem::RemoveEffectFromAll(TSubclassOf<UGameplayEffect> Effect)
{
	if ((Effect.Get() != nullptr) && AppliedEffects.Contains(Effect))
	{
		FGlobalAppliedEffectList& Entry = AppliedEffects[Effect];
		Entry.RemoveFromAll();
		AppliedEffects.Remove(Effect);
	}
}

// 把某个 ASC 注册进全局系统，并补授所有已全局应用的技能/效果
void UTenkaichiGlobalAbilitySystem::RegisterASC(UTenkaichiAbilitySystemComponent* ASC)
{
	check(ASC);

	for (auto& Entry : AppliedAbilities)
	{
		Entry.Value.AddToASC(Entry.Key, ASC);
	}
	for (auto& Entry : AppliedEffects)
	{
		Entry.Value.AddToASC(Entry.Key, ASC);
	}

	RegisteredASCs.AddUnique(ASC);
}

// 把某个 ASC 从全局系统移除，并移除它身上所有全局应用的技能/效果
void UTenkaichiGlobalAbilitySystem::UnregisterASC(UTenkaichiAbilitySystemComponent* ASC)
{
	check(ASC);
	for (auto& Entry : AppliedAbilities)
	{
		Entry.Value.RemoveFromASC(ASC);
	}
	for (auto& Entry : AppliedEffects)
	{
		Entry.Value.RemoveFromASC(ASC);
	}

	RegisteredASCs.Remove(ASC);
}
```

- `UE_INLINE_GENERATED_CPP_BY_NAME`：U 类反射必需，永不挂起。

### 2. FGlobalAppliedAbilityList::AddToASC（授技能给单个 ASC）

```cpp
void FGlobalAppliedAbilityList::AddToASC(TSubclassOf<UGameplayAbility> Ability, UTenkaichiAbilitySystemComponent* ASC)
{
	if (FGameplayAbilitySpecHandle* SpecHandle = Handles.Find(ASC))
	{
		RemoveFromASC(ASC);
	}

	UGameplayAbility* AbilityCDO = Ability->GetDefaultObject<UGameplayAbility>();
	FGameplayAbilitySpec AbilitySpec(AbilityCDO);
	const FGameplayAbilitySpecHandle AbilitySpecHandle = ASC->GiveAbility(AbilitySpec);
	Handles.Add(ASC, AbilitySpecHandle);
}
```

- 先查 `Handles` 里有没有这个 ASC：**已授过就先移除**（避免重复授予）。
- `Ability->GetDefaultObject<UGameplayAbility>()`：拿到技能类的**CDO（默认对象）**——`TSubclassOf` 传的是类，造技能用的是它的 CDO 配置。
- `FGameplayAbilitySpec AbilitySpec(AbilityCDO)`：用 CDO 构造一个"技能规格"（spec，描述这个技能实例）。
- `ASC->GiveAbility(AbilitySpec)`：真正把技能授给 ASC，返回一个 `FGameplayAbilitySpecHandle`（唯一句柄）。
- `Handles.Add(ASC, AbilitySpecHandle)`：记录"这个 ASC ↔ 这个句柄"，方便以后移除。

### 3. RemoveFromASC / RemoveFromAll（技能侧移除）

```cpp
void FGlobalAppliedAbilityList::RemoveFromASC(UTenkaichiAbilitySystemComponent* ASC)
{
	if (FGameplayAbilitySpecHandle* SpecHandle = Handles.Find(ASC))
	{
		ASC->ClearAbility(*SpecHandle);
		Handles.Remove(ASC);
	}
}

void FGlobalAppliedAbilityList::RemoveFromAll()
{
	for (auto& KVP : Handles)
	{
		if (KVP.Key != nullptr)
		{
			KVP.Key->ClearAbility(KVP.Value);
		}
	}
	Handles.Empty();
}
```

- `RemoveFromASC`：按 ASC 找到句柄，`ASC->ClearAbility(*SpecHandle)` 清掉技能，再从表里删掉这一项。
- `RemoveFromAll`：遍历整张表，对每个 ASC 调 `ClearAbility`，最后 `Handles.Empty()` 清空表。
  - `auto& KVP`：`KVP.Key` = ASC，`KVP.Value` = 句柄。
  - `KVP.Key != nullptr`：防空指针，安全起见才操作。

### 4. FGlobalAppliedEffectList（效果侧，和技能侧对称）

```cpp
void FGlobalAppliedEffectList::AddToASC(TSubclassOf<UGameplayEffect> Effect, UTenkaichiAbilitySystemComponent* ASC)
{
	if (FActiveGameplayEffectHandle* EffectHandle = Handles.Find(ASC))
	{
		RemoveFromASC(ASC);
	}

	const UGameplayEffect* GameplayEffectCDO = Effect->GetDefaultObject<UGameplayEffect>();
	const FActiveGameplayEffectHandle GameplayEffectHandle = ASC->ApplyGameplayEffectToSelf(GameplayEffectCDO, /*Level=*/ 1, ASC->MakeEffectContext());
	Handles.Add(ASC, GameplayEffectHandle);

```

- 逻辑和技能侧对称，但用**效果**的 API：
  - `ASC->ApplyGameplayEffectToSelf(GameplayEffectCDO, 1, ASC->MakeEffectContext())`：把效果施加到 ASC 自己身上。
    三个参数：效果 CDO、等级（这里写死 `1`）、效果上下文（`MakeEffectContext()` 现场造一个）。
  - 返回 `FActiveGameplayEffectHandle`（激活效果句柄），记入 `Handles`。
- `RemoveFromASC` / `RemoveFromAll` 用 `ASC->RemoveActiveGameplayEffect(...)` 移除效果，其余同技能侧。

### 6. ApplyAbilityToAll / ApplyEffectToAll（批量授予）

```cpp
void UTenkaichiGlobalAbilitySystem::ApplyAbilityToAll(TSubclassOf<UGameplayAbility> Ability)
{
	if ((Ability.Get() != nullptr) && (!AppliedAbilities.Contains(Ability)))
	{
		FGlobalAppliedAbilityList& Entry = AppliedAbilities.Add(Ability);		
		for (ULyraAbilitySystemComponent* ASC : RegisteredASCs)
		{
			Entry.AddToASC(Ability, ASC);
		}
	}
}
```

- 判断：`Ability.Get() != nullptr`（类有效）且 `!AppliedAbilities.Contains(Ability)`（还没全局应用过）。
- `AppliedAbilities.Add(Ability)`：往"已全局应用技能表"里加一项，返回这张表的引用 `Entry`。
- 遍历 `RegisteredASCs`（所有已注册 ASC），对每个 ASC 调 `Entry.AddToASC` 批量授予。
- `ApplyEffectToAll` 完全对称，只是用 `AppliedEffects` 表 + `AddToASC`（效果版）。

### 7. RemoveAbilityFromAll / RemoveEffectFromAll（批量移除）

```cpp
void UTenkaichiGlobalAbilitySystem::RemoveAbilityFromAll(TSubclassOf<UGameplayAbility> Ability)
{
	if ((Ability.Get() != nullptr) && AppliedAbilities.Contains(Ability))
	{
		FGlobalAppliedAbilityList& Entry = AppliedAbilities[Ability];
		Entry.RemoveFromAll();
		AppliedAbilities.Remove(Ability);
	}
}
```

- 类有效且表中确实有：取这张表，`RemoveFromAll()` 把技能从所有 ASC 移除，然后 `AppliedAbilities.Remove(Ability)` 从全局表删掉这条。
- 这样以后再 `ApplyAbilityToAll` 同一个技能时，`Contains` 判为 false，能重新应用。

### 8. RegisterASC / UnregisterASC（注册/注销 ASC）

```cpp
void UTenkaichiGlobalAbilitySystem::RegisterASC(UTenkaichiAbilitySystemComponent* ASC)
{
	check(ASC);

	for (auto& Entry : AppliedAbilities)
	{
		Entry.Value.AddToASC(Entry.Key, ASC);
	}
	for (auto& Entry : AppliedEffects)
	{
		Entry.Value.AddToASC(Entry.Key, ASC);
	}

	RegisteredASCs.AddUnique(ASC);
}
```

- `check(ASC)`：ASC 不能为空，否则直接崩溃（断言，开发期抓 bug）。
- **核心**：新 ASC 注册时，把**当前所有已全局应用的技能/效果**逐个补授给它（`Entry.Value.AddToASC(Entry.Key, ASC)`）。
- 最后 `RegisteredASCs.AddUnique(ASC)` 记入列表（`AddUnique` 防重复）。

```cpp
void UTenkaichiGlobalAbilitySystem::UnregisterASC(UTenkaichiAbilitySystemComponent* ASC)
{
	check(ASC);
	for (auto& Entry : AppliedAbilities)
	{
		Entry.Value.RemoveFromASC(ASC);
	}
	for (auto& Entry : AppliedEffects)
	{
		Entry.Value.RemoveFromASC(ASC);
	}

	RegisteredASCs.Remove(ASC);
}
```

- 反向：从所有全局技能/效果里把该 ASC 移除，然后 `RegisteredASCs.Remove(ASC)` 从列表剔除。

## 五、几个要记住的点

1. **GiveAbility / ApplyGameplayEffectToSelf**：给 ASC 授技能 / 加效果的两个核心引擎 API。
2. **CDO 造对象**：`TSubclassOf` 传类，用 `GetDefaultObject` 取配置再造 spec / effect。
3. **RegisterASC 补授**：新 ASC 注册时自动补齐已全局应用的技能/效果——这是"全局"生效的关键。
4. **全部零挂起**：依赖的 ASC 已落盘，本文件直接可编译。
