// 让本 Pawn 成为能力系统 avatar：初始化 ASC 的 Actor 信息并广播
void UTenkaichiPawnExtensionComponent::InitializeAbilitySystem(UTenkaichiAbilitySystemComponent* InASC, AActor* InOwnerActor)
{
	check(InASC);
	check(InOwnerActor);

	if (AbilitySystemComponent == InASC)
	{
		// 能力系统组件没变，直接返回
		return;
	}

	if (AbilitySystemComponent)
	{
		// 清理旧的能力系统组件
		UninitializeAbilitySystem();
	}

	APawn* Pawn = GetPawnChecked<APawn>();
	AActor* ExistingAvatar = InASC->GetAvatarActor();

	UE_LOG(LogTenkaichi, Verbose, TEXT("Setting up ASC [%s] on pawn [%s] owner [%s], existing [%s] "), *GetNameSafe(InASC), *GetNameSafe(Pawn), *GetNameSafe(InOwnerActor), *GetNameSafe(ExistingAvatar));

	if ((ExistingAvatar != nullptr) && (ExistingAvatar != Pawn))
	{
		UE_LOG(LogTenkaichi, Log, TEXT("Existing avatar (authority=%d)"), ExistingAvatar->HasAuthority() ? 1 : 0);

		// 已经有一个 Pawn 是这个 ASC 的 avatar，得把它踢出去。
		// 这种情况会出现在客户端卡顿：新 Pawn 生成并拥有，而旧的还没被移除
		ensure(!ExistingAvatar->HasAuthority());

		if (UTenkaichiPawnExtensionComponent* OtherExtensionComponent = FindPawnExtensionComponent(ExistingAvatar))
		{
			OtherExtensionComponent->UninitializeAbilitySystem();
		}
	}

	AbilitySystemComponent = InASC;
	AbilitySystemComponent->InitAbilityActorInfo(InOwnerActor, Pawn);

	if (ensure(PawnData))
	{
		InASC->SetTagRelationshipMapping(PawnData->TagRelationshipMapping);
	}

	OnAbilitySystemInitialized.Broadcast();
}

// 卸下本 Pawn 作为能力系统 avatar 的身份（并取消相关技能/输入/表现）
void UTenkaichiPawnExtensionComponent::UninitializeAbilitySystem()
{
	if (!AbilitySystemComponent)
	{
		return;
	}

	// 只有当我们还是 avatar actor 时才反初始化（否则别的 Pawn 已经在我们成为 avatar 时做过了）
	if (AbilitySystemComponent->GetAvatarActor() == GetOwner())
	{
		FGameplayTagContainer AbilityTypesToIgnore;
		AbilityTypesToIgnore.AddTag(TenkaichiGameplayTags::Ability_Behavior_SurvivesDeath);

		AbilitySystemComponent->CancelAbilities(nullptr, &AbilityTypesToIgnore);
		AbilitySystemComponent->ClearAbilityInput();
		AbilitySystemComponent->RemoveAllGameplayCues();

		if (AbilitySystemComponent->GetOwnerActor() != nullptr)
		{
			AbilitySystemComponent->SetAvatarActor(nullptr);
		}
		else
		{
			// 如果 ASC 没有有效 owner，则需要清掉*所有* Actor 信息，而不仅是 avatar 配对
			AbilitySystemComponent->ClearActorInfo();
		}

		OnAbilitySystemUninitialized.Broadcast();
	}

	AbilitySystemComponent = nullptr;
}

// 推进默认初始化：先推进所依赖的其他特性，再沿状态链推进自己
void UTenkaichiPawnExtensionComponent::CheckDefaultInitialization()
{

	ContinueInitStateChain(StateChain);
}

// 状态机核心：判断能否从当前状态进入目标状态
bool UTenkaichiPawnExtensionComponent::CanChangeInitState(UGameFrameworkComponentManager* Manager, FGameplayTag CurrentState, FGameplayTag DesiredState) const
{
	check(Manager);

	APawn* Pawn = GetPawn<APawn>();
	if (!CurrentState.IsValid() && DesiredState == TenkaichiGameplayTags::InitState_Spawned)
	{
		// 只要在合法 Pawn 上，就算作已生成
		if (Pawn)
		{
			return true;
		}
	}
	if (CurrentState == TenkaichiGameplayTags::InitState_Spawned && DesiredState == TenkaichiGameplayTags::InitState_DataAvailable)
	{
		// 需要角色数据
		if (!PawnData)
		{
			return false;
		}

		const bool bHasAuthority = Pawn->HasAuthority();
		const bool bIsLocallyControlled = Pawn->IsLocallyControlled();

		if (bHasAuthority || bIsLocallyControlled)
		{
			// 检查是否已被控制器拥有
			if (!GetController<AController>())
			{
				return false;
			}
		}

		return true;
	}
	else if (CurrentState == TenkaichiGameplayTags::InitState_DataAvailable && DesiredState == TenkaichiGameplayTags::InitState_DataInitialized)
	{
		// 所有特性都已"数据可用"才进入初始化
		return Manager->HaveAllFeaturesReachedInitState(Pawn, TenkaichiGameplayTags::InitState_DataAvailable);
	}
	else if (CurrentState == TenkaichiGameplayTags::InitState_DataInitialized && DesiredState == TenkaichiGameplayTags::InitState_GameplayReady)
	{
		return true;
	}

	return false;
}

## 二、逐块讲解（先讲"为什么"再讲"是什么"）

### 块 1：构造 + 复制属性

**为什么 + 是什么**：构造里关掉 Tick（这个组件不每帧做事）、默认开启复制（`SetIsReplicatedByDefault(true)`）、清空两个数据指针。`GetLifetimeReplicatedProps` 只复制 `PawnData`（`DOREPLIFETIME`），因为它是唯一需要跨端同步的属性。

### 块 2：OnRegister / BeginPlay / EndPlay（生命周期）

**为什么 + 是什么**：
- `OnRegister`：组件注册时校验"只能挂在 Pawn 上 + 只能有一个"（两个 `ensureAlwaysMsgf`），然后 `RegisterInitStateFeature()` **尽早注册到初始化状态系统**。
- `BeginPlay`：`BindOnActorInitStateChanged(NAME_None, ...)` 监听**所有**特性状态变化 → `TryToChangeInitState(InitState_Spawned)` 推进到"已生成" → `CheckDefaultInitialization()` 尝试继续推进。
- `EndPlay`：先 `UninitializeAbilitySystem()`（卸下能力系统）+ `UnregisterInitStateFeature()`（注销状态系统），再调父类。

> **核心**：PawnExtension 是整个 Pawn 初始化链的"发动机"，BeginPlay 里先推进自己到 Spawned。

### 块 3：SetPawnData / OnRep_PawnData（角色数据）

**为什么 + 是什么**：`SetPawnData` 只在**权威端**能设置（`GetLocalRole() != ROLE_Authority` 直接 return），且已有数据就不能重复设（`UE_LOG` 报错）。设完 `ForceNetUpdate()` 强制同步 + `CheckDefaultInitialization()` 推进。客户端收到复制后走 `OnRep_PawnData` 再推进。

### 块 4：InitializeAbilitySystem / UninitializeAbilitySystem（能力系统接入）

**为什么 + 是什么**：`InitializeAbilitySystem` 是本类最核心方法——让本 Pawn 成为 ASC 的 avatar：
1. ASC 没变直接返回；变了先清理旧的
2. 若已有别的 avatar 且不是本 Pawn（客户端卡顿场景），把它踢出去（`UninitializeAbilitySystem`）
3. `InitAbilityActorInfo(InOwnerActor, Pawn)` 建立 ASC ↔ Pawn 的关联
4. 设置 `TagRelationshipMapping`（依赖 PawnData）
5. `OnAbilitySystemInitialized.Broadcast()` 通知

`UninitializeAbilitySystem`：卸下 avatar——`CancelAbilities`（保留 `SurvivesDeath` 标签的技能）+ `ClearAbilityInput` + `RemoveAllGameplayCues` + `SetAvatarActor(nullptr)`（或 `ClearActorInfo`），再广播反初始化。⚠️ 全程硬调用 GAS（断点）。

### 块 5：HandleControllerChanged / HandlePlayerStateReplicated / SetupPlayerInputComponent（推进时机）

**为什么 + 是什么**：这三个方法**结构几乎一样**——都只调 `CheckDefaultInitialization()` 再次推进初始化链（`HandleControllerChanged` 额外在控制器改变时刷新 ASC 的 Actor 信息）。它们由 Pawn 在对应时机调用：
- `HandleControllerChanged`：控制器改变时
- `HandlePlayerStateReplicated`：PlayerState 复制到客户端时
- `SetupPlayerInputComponent`：输入组件设置好时

> 三者都是"某个依赖刚就位，试试能不能继续推进"的触发点。

### 块 6：CheckDefaultInitialization（推进初始化链）

**为什么 + 是什么**：先 `CheckDefaultInitializationForImplementers()` 推进所依赖的**其他**特性，再定义状态链 `{Spawned → DataAvailable → DataInitialized → GameplayReady}`，用 `ContinueInitStateChain(StateChain)` 沿链推进。

### 块 7：CanChangeInitState（状态机核心判断）

**为什么 + 是什么**：决定每个状态转移是否可行：
- 空 → Spawned：只要在合法 Pawn 上即可
- Spawned → DataAvailable：需要 `PawnData` + 权威端/本地控制时需有控制器
- DataAvailable → DataInitialized：**所有**特性都到 DataAvailable（`HaveAllFeaturesReachedInitState`）
- DataInitialized → GameplayReady：直接 true

### 块 8：HandleChangeInitState / OnActorInitStateChanged

**为什么 + 是什么**：`HandleChangeInitState` 在状态改变时回调（DataInitialized 阶段由其他组件监听处理）。`OnActorInitStateChanged` 监听**别的**特性——别的特性到 DataAvailable 时，触发自己的 `CheckDefaultInitialization`。

### 块 9：两个 Register（能力系统通知）

**为什么 + 是什么**：`OnAbilitySystemInitialized_RegisterAndCall`：注册 + 若已初始化立即执行（`_AndCall` 的由来）；`OnAbilitySystemUninitialized_Register`：只注册不立即执行（`IsBoundToObject` 防重复绑定）。
