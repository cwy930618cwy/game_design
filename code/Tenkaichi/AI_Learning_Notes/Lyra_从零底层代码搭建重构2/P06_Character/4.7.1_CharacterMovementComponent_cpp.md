```cpp
/** 模拟移动：如果外部传入了复制加速度，就跑完模拟再把它还原回去 */
void UTenkaichiCharacterMovementComponent::SimulateMovement(float DeltaTime)
{
	if (bHasReplicatedAcceleration)
	{
		// 保留外部传进来的复制加速度
		const FVector OriginalAcceleration = Acceleration;
		Super::SimulateMovement(DeltaTime);
		Acceleration = OriginalAcceleration;
	}
	else
	{
		Super::SimulateMovement(DeltaTime);
	}
}
```

| 分支 | 逻辑 |
|------|------|
| `if (bHasReplicatedAcceleration)` | 有外部传入的加速度：先存下原值 → 跑引擎默认模拟 → **再把加速度还原**（防止模拟覆盖掉外部值） |
| `else` | 没有外部加速度：直接走引擎默认 |

> 关键：这是"油门同步器"的实现——模拟端角色用外部传的加速度移动，跑完不让引擎把它冲掉。

	// 和 UCharacterMovementComponent 的实现一样，但没有蹲着检查

## 五、GetGroundInfo（核心：地面探测 + 帧缓存）

```cpp
/** 获取当前地面信息：如果数据过期就重新算一次，然后返回缓存 */
const FTenkaichiCharacterGroundInfo& UTenkaichiCharacterMovementComponent::GetGroundInfo()
{
	if (!CharacterOwner || (GFrameCounter == CachedGroundInfo.LastUpdateFrame))
	{
		return CachedGroundInfo;
	}

	if (MovementMode == MOVE_Walking)
	{
		CachedGroundInfo.GroundHitResult = CurrentFloor.HitResult;
		CachedGroundInfo.GroundDistance = 0.0f;
	}
	else
	{
		const UCapsuleComponent* CapsuleComp = CharacterOwner->GetCapsuleComponent();
		check(CapsuleComp);

		const float CapsuleHalfHeight = CapsuleComp->GetUnscaledCapsuleHalfHeight();
		const ECollisionChannel CollisionChannel = (UpdatedComponent ? UpdatedComponent->GetCollisionObjectType() : ECC_Pawn);
		const FVector TraceStart(GetActorLocation());
		const FVector TraceEnd(TraceStart.X, TraceStart.Y, (TraceStart.Z - LyraCharacter::GroundTraceDistance - CapsuleHalfHeight));

		FCollisionQueryParams QueryParams(SCENE_QUERY_STAT(TenkaichiCharacterMovementComponent_GetGroundInfo), false, CharacterOwner);
		FCollisionResponseParams ResponseParam;
		InitCollisionParams(QueryParams, ResponseParam);

		FHitResult HitResult;
		GetWorld()->LineTraceSingleByChannel(HitResult, TraceStart, TraceEnd, CollisionChannel, QueryParams, ResponseParam);

		CachedGroundInfo.GroundHitResult = HitResult;
		CachedGroundInfo.GroundDistance = LyraCharacter::GroundTraceDistance;

		if (MovementMode == MOVE_NavWalking)
		{
			CachedGroundInfo.GroundDistance = 0.0f;
		}
		else if (HitResult.bBlockingHit)
		{
			CachedGroundInfo.GroundDistance = FMath::Max((HitResult.Distance - CapsuleHalfHeight), 0.0f);
		}
	}

	CachedGroundInfo.LastUpdateFrame = GFrameCounter;

	return CachedGroundInfo;
}
```

| 段 | 逻辑 |
|----|------|
| **帧缓存判读** | `!CharacterOwner`（没角色）或 `GFrameCounter == LastUpdateFrame`（同帧已算过）→ **直接返回缓存**，不重算 |
| **行走模式** `MOVE_Walking` | 直接用引擎缓存的 `CurrentFloor.HitResult`，地面距离 `0`（脚就在地上） |
| **非行走模式**（空中/下坡） | 手动做一条向下的射线：起点=角色位置，终点=起点往下 `GroundTraceDistance + 胶囊半高` |
| **射线参数** | `CollisionChannel`=组件碰撞通道（默认 `ECC_Pawn`）；`QueryParams` 忽略角色自己（`CharacterOwner` 作为被忽略对象） |
| **存结果** | `GroundHitResult`=命中结果；`GroundDistance` 先记默认最大距离 |
| **距离修正** | 导航行走 `MOVE_NavWalking` → 距离 `0`；命中阻挡 → `FMath::Max(命中距离 - 胶囊半高, 0)`（扣除半高才是"脚到地"的距离） |
| **记帧号** | `CachedGroundInfo.LastUpdateFrame = GFrameCounter` 标记"这帧算过了" |

> 类比：这是"地面探测器"的完整逻辑——**同帧查速查表，跨帧才重新打一枪**（向下射线），并根据是走路/导航/悬空算出正确的"脚离地距离"。

> 注意：`LyraCharacter::GroundTraceDistance` 是我在上文命名空间里定义的探测距离；`SCENE_QUERY_STAT` 是给射线取个统计名（性能分析用）。

## 七、GetDeltaRotation（急停点 1：转身归零）+ GetMaxSpeed（急停点 2：速度归零）

```cpp
/** 覆盖"转身角速度"：挂了"移动停止"tag 就转不动（返回 0） */
FRotator UTenkaichiCharacterMovementComponent::GetDeltaRotation(float DeltaTime) const
{
	if (UAbilitySystemComponent* ASC = UAbilitySystemGlobals::GetAbilitySystemComponentFromActor(GetOwner()))
	{
		if (ASC->HasMatchingGameplayTag(TAG_Gameplay_MovementStopped))
		{
			return FRotator(0,0,0);
		}
	}

	return Super::GetDeltaRotation(DeltaTime);
}
```

| 段 | 逻辑 |
|----|------|
| `UAbilitySystemGlobals::GetAbilitySystemComponentFromActor(GetOwner())` | 从角色身上拿到它的 GAS 组件（**引擎类**） |
| `ASC->HasMatchingGameplayTag(TAG_Gameplay_MovementStopped)` | 检查这个 GAS 组件有没有挂"急停"tag |
| 挂了 → `return FRotator(0,0,0)` | **转身角速度归零**（转不了头） |
| 没挂 → `return Super::GetDeltaRotation(DeltaTime)` | 走引擎默认转身 |

```cpp
/** 覆盖"最大速度"：挂了"移动停止"tag 就跑不动（返回 0） */
float UTenkaichiCharacterMovementComponent::GetMaxSpeed() const
{
	if (UAbilitySystemComponent* ASC = UAbilitySystemGlobals::GetAbilitySystemComponentFromActor(GetOwner()))
	{
		if (ASC->HasMatchingGameplayTag(TAG_Gameplay_MovementStopped))
		{
			return 0;
		}
	}

	return Super::GetMaxSpeed();
}
```

| 段 | 逻辑 |
|----|------|
| 同样的 GAS 查询 + tag 检查 | 和 `GetDeltaRotation` 完全一样的套路 |
| 挂了 → `return 0` | **最大速度归零**（跑不动） |
| 没挂 → `return Super::GetMaxSpeed()` | 走引擎默认最大速度 |

> 类比：这两个是"急停总闸"的**两个执行点**——`GetDeltaRotation` 管"不能转头"，`GetMaxSpeed` 管"不能加速"。只要 GAS 组件挂上 `Gameplay.MovementStopped` 这个 tag，角色就彻底"冻住"。
