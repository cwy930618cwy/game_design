```cpp
// 做一次射线检测，取最近的命中点（静态工具，供子类复用）
void UAbilityTask_WaitForInteractableTargets::LineTrace(FHitResult& OutHitResult, const UWorld* World, const FVector& Start, const FVector& End, FName ProfileName, const FCollisionQueryParams Params)
{
	check(World);

	OutHitResult = FHitResult();
	TArray<FHitResult> HitResults;
	World->LineTraceMultiByProfile(HitResults, Start, End, ProfileName, Params);

	OutHitResult.TraceStart = Start;
	OutHitResult.TraceEnd = End;

	if (HitResults.Num() > 0)
	{
		OutHitResult = HitResults[0];
	}
}

// 从玩家视角计算瞄准射线的终点（考虑相机朝向、最大射程，可选忽略俯仰角）
void UAbilityTask_WaitForInteractableTargets::AimAtPlayerController(const AActor* InSourceActor, FCollisionQueryParams Params, const FVector& TraceStart, float MaxRange, FVector& OutTraceEnd, bool bIgnorePitch) const
{
	if (!Ability) // 只在服务器和发起客户端执行
	{
		return;
	}

	//@TODO: Bots?
	APlayerController* PC = Ability->GetCurrentActorInfo()->PlayerController.Get();
	check(PC);

	FVector ViewStart;
	FRotator ViewRot;
	PC->GetPlayerViewPoint(ViewStart, ViewRot);

	const FVector ViewDir = ViewRot.Vector();
	FVector ViewEnd = ViewStart + (ViewDir * MaxRange);

	// 把相机射线裁剪到能力作用半径内
	ClipCameraToAbilityRange(ViewStart, ViewDir, Start, MaxRange, ViewEnd);

	FHitResult HitResult;
	LineTrace(HitResult, InSourceActor->GetWorld(), ViewStart, ViewEnd, TraceProfile.Name, Params);

	// 若射线打中障碍物且在最大射程内，就用命中点作为瞄准终点
	const bool bUseResult = HitResult.bBlockingHit && (FVector::DistSquared(Start, HitResult.Location) <= (MaxRange * MaxRange));
	const FVector AdjustedEnd = bUseResult ? HitResult.Location : ViewEnd;

	FVector AdjustedDir = (AdjustedEnd - Start).GetSafeNormal();
	if (AdjustedDir.IsZero())
	{
		AdjustedDir = ViewDir;
	}

	// 忽略俯仰角时：保留原视线俯仰，只改水平方向
	if (!bIgnorePitch && !bUseResult)
	{
		FVector OriginalDir = (ViewEnd - Start).GetSafeNormal();
		if (!OriginalDir.IsZero())
		{
			const FRotator OriginalRot = OriginalDir.Rotation();
			FRotator AdjustedRot = AdjustedDir.Rotation();
			AdjustedRot.Pitch = OriginalRot.Pitch;
			AdjustedDir = AdjustedRot.Vector();
		}
	}

	OutTarget = Start + (AdjustedDir * MaxRange);
}
```

（`ClipCameraToAbilityRange` 和 `UpdateInteractableOptions` 见下方分段讲解。）

### 4. `AimAtPlayerController`（第 30~50 行）

```cpp
	if (!Ability) { return; }
```

### 5. `ClipCameraToAbilityRange`（第 53~65 行）

```cpp
	// 相机到能力中心的向量
	Vector CameraToCenter = AbilityCenter - CameraLocation;
	// 相机视线方向与到中心的点积：判断相机朝向能力中心
	float DotToCenter = Vector::DotProduct(CameraToCenter, CameraDirection);
	if (DotToCenter >= 0)
	{
		// 用勾股定理求相机在视线方向上的投影，裁剪到球面上
		float DistanceSquared = CameraToCenter.SizeSquared() - (DotToCenter * DotToCenter);
		float RadiusSquared = (Range * Range);
		if (DistanceSquared <= RadiusSquared)
		{
			float DistanceFromCamera = Math::Sqrt(RadiusSquared - DistanceSquared);
			float DistanceAlongRay = DotToCenter + DistanceFromCamera;
			ClippedPosition = CameraLocation + (DistanceAlongRay * CameraDirection);
			return true;
		}
	}
	return false;
```

- 纯数学函数：把相机发出的射线，裁剪到以 `AbilityCenter` 为球心、`Range` 为半径的球面上。
- 返回球面上的交点位置；若相机视线不指向球面则返回 false。

### 6. `UpdateInteractorOptions` 核心（第 68~122 行）

**第 1~3 步：收集 + 找能力规格 + 过滤**
```cpp
	for (const TScriptInterface<IInteractorTarget> & InteractiveTarget : InteractorTargets)
	{
		TArray<FInteractionOption> TempOptions;
		FInteractionOptionBuilder InteractionBuilder(InteractiveTarget, TempOptions);
		InteractiveTarget->GatherInteractorOptions(InteractQuery, InteractionBuilder);

		for (FInteractionOption& Option : TempOptions)
		{
			FGameplayAbilitySpec* InteractionAbilitySpec = nullptr;

			// 情况 A：目标自身有指定能力
			if (Option.TargetAbilitySystem && Option.TargetInteractionAbilityHandle.IsValid())
			{
				InteractionAbilitySpec = Option.TargetAbilitySystem->FindAbilitySpecFromHandle(Option.TargetInteractionAbilityHandle);
			}
			// 情况 B：需要授予玩家的能力
			else if (Option.InteractionAbilityToGrant)
			{
				InteractionAbilitySpec = AbilitySystemComponent->FindAbilitySpecFromClass(Option.InteractionAbilityToGrant);
				if (InteractionAbilitySpec)
				{
					Option.TargetAbilitySystem = AbilitySystemComponent.Get();
					Option.TargetInteractionAbilityHandle = InteractionAbilitySpec->Handle;
				}
			}

			// 过滤：现在无法激活的不加入新列表
			if (InteractionAbilitySpec && InteractionAbilitySpec->Ability->CanActivateAbility(InteractionAbilitySpec->Handle, AbilitySystemComponent->AbilityActorInfo.Get()))
			{
				NewOptions.Add(Option);
			}
		}
	}
```

- 遍历每个目标，收集原始选项到 `TempOptions`。
- 对每个选项，查它对应的 `FGameplayAbilitySpec`：
  - A：目标自身能力（用 Handle 查）；
  - B：需要授予玩家的能力（用 Class 查，并把选项更新为"玩家身上"）。
- 用 `CanActivateAbility` 过滤掉现在用不了的。

**第 4 步：对比 + 广播**
```cpp
	bool bOptionsChanged = false;
	if (NewOptions.Num() == CurrentOptions.Num())
	{
		NewOptions.Sort();
		for (int i = 0; i < NewOptions.Num(); i++)
		{
			if (NewOptions[i] != CurrentOptions[i]) { bOptionsChanged = true; break; }
		}
	}
	else { bOptionsChanged = true; }

	if (bOptionsChanged)
	{
		CurrentOptions = NewOptions;
		InteractorObjectsChanged.Broadcast(CurrentOptions);
	}
```

- 新旧列表无差异则不广播，避免 UI 频繁刷新。
- 有差异才更新并广播 `InteractorObjectsChanged`。
