	{
		UTenkaichiTeamSubsystem* TeamSubsystem = GetWorld()->GetSubsystem<UTenkaichiTeamSubsystem>();

void ATenkaichiTeamInfoBase::TryRegisterWithTeamSubsystem()
{
	if (TeamId != INDEX_NONE)
	{
		UTenkaichiTeamSubsystem* TeamSubsystem = GetWorld()->GetSubsystem<UTenkaichiTeamSubsystem>();
		if (ensure(TeamSubsystem))
		{
			RegisterWithTeamSubsystem(TeamSubsystem);
		}
	}
}

{
	check(HasAuthority());
	check(TeamId == INDEX_NONE);
	check(NewTeamId != INDEX_NONE);

	TeamId = NewTeamId;

	TryRegisterWithTeamSubsystem();
}

// 客户端收到队号复制时触发，重新尝试注册
void ATenkaichiTeamInfoBase::OnRep_TeamId()
{
	TryRegisterWithTeamSubsystem();
}

## 二、逐段讲解（先讲"为什么"再讲"是什么"）

### 1. 构造函数

- `TeamId(INDEX_NONE)`：初始"无队伍"（`INDEX_NONE` = -1）。
- `bReplicates = true`：这个 Actor 要跨网络复制。
- `bAlwaysRelevant = true`：对**所有客户端**始终相关（队伍信息每个人都要知道）。
- `NetPriority = 3.0f`：网络优先级较高（队伍信息要及时同步）。
- `SetReplicatingMovement(false)`：不复制移动（`AInfo` 无碰撞，不用同步位置）。

### 2. `GetLifetimeReplicatedProps` —— 声明复制字段

- `DOREPLIFETIME(ThisClass, TeamTags)`：`TeamTags` 持续复制。
- `DOREPLIFETIME_CONDITION(ThisClass, TeamId, COND_InitialOnly)`：`TeamId` 只在**初次复制**时同步（队号建好后基本不变）。

### 3. `BeginPlay` / `EndPlay`

- `BeginPlay` → `TryRegisterWithTeamSubsystem()`：开局把自己登记进队伍子系统。
- `EndPlay` → 若已有队号，从子系统 `UnregisterTeamInfo` 注销。判空是因为 `EndPlay` 可能在子系统已被销毁时触发。

### 4. `RegisterWithTeamSubsystem` / `TryRegisterWithTeamSubsystem`

- `RegisterWithTeamSubsystem(Subsystem)`：**protected 虚**，把队伍注册到指定子系统（派生类可覆写增强）。
- `TryRegisterWithTeamSubsystem()`：**只有有队号**才注册；`ensure(TeamSubsystem)` 断言子系统存在。

### 5. `SetTeamId` / `OnRep_TeamId`

- `SetTeamId`：服务器端分配队号，**三个 check**（有权、当前无队、新队号有效）保证"只设一次、合法"。
- `OnRep_TeamId`：客户端收到复制的队号后，重新尝试注册。
