```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#include "Teams/AsyncAction_ObserveTeamColors.h"

#include "Engine/Engine.h"
#include "Engine/World.h"
#include "Teams/TenkaichiTeamAgentInterface.h"
#include "Teams/TenkaichiTeamStatics.h"
#include "Teams/TenkaichiTeamSubsystem.h"

#include UE_INLINE_GENERATED_CPP_BY_NAME(AsyncAction_ObserveTeamColors)

// 构造函数：把对象初始化器原样转交给父类，不做任何额外初始化
UAsyncAction_ObserveTeamColors::UAsyncAction_ObserveTeamColors(const FObjectInitializer& ObjectInitializer)
	: Super(ObjectInitializer)
{
}

// 监听指定对象的队伍颜色变化（蓝图异步节点入口）：造节点、记住两个弱引用并注册到 GameInstance 防 GC
UAsyncAction_ObserveTeamColors* UAsyncAction_ObserveTeamColors::ObserveTeamColors(UObject* TeamAgent)
{
	UAsyncAction_ObserveTeamColors* Action = nullptr;

	if (TeamAgent != nullptr)
	{
		Action = NewObject<UAsyncAction_ObserveTeamColors>();
		Action->TeamInterfacePtr = TWeakInterfacePtr<ITenkaichiTeamAgentInterface>(TeamAgent);
		Action->TeamInterfaceObj = TeamAgent;
		Action->RegisterWithGameInstance(TeamAgent);
	}

	return Action;
}

// 节点准备销毁时的收尾：先让父类解除 GameInstance 注册，再从"换队"委托上解绑本对象的所有绑定
void UAsyncAction_ObserveTeamColors::SetReadyToDestroy()
{
	Super::SetReadyToDestroy();

	// 如果我们被取消了，需要把所有可能监听过的东西都解绑
	if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
	{
		TeamInterface->GetTeamChangedDelegateChecked().RemoveAll(this);
	}
}

// 节点激活：取当前队伍与外观资产、绑定"换队"委托、立刻广播一次，取不到 World 或绑定失败则自我销毁
void UAsyncAction_ObserveTeamColors::Activate()
{
	bool bCouldSucceed = false;
	int32 CurrentTeamIndex = INDEX_NONE;
	UTenkaichiTeamDisplayAsset* CurrentDisplayAsset = nullptr;

	if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
	{
		if (UWorld* World = GEngine->GetWorldFromContextObject(TeamInterfaceObj.Get(), EGetWorldErrorMode::LogAndReturnNull))
		{
			// 取得当前的队伍信息
			CurrentTeamIndex = GenericTeamIdToInteger(TeamInterface->GetGenericTeamId());
			CurrentDisplayAsset = UTenkaichiTeamStatics::GetTeamDisplayAsset(World, CurrentTeamIndex);

			// 监听未来的队伍变化
			TeamInterface->GetTeamChangedDelegateChecked().AddDynamic(this, &ThisClass::OnWatchedAgentChangedTeam);

			bCouldSucceed = true;
		}
	}

	// 广播一次，让使用者拿到当前状态
	BroadcastChange(CurrentTeamIndex, CurrentDisplayAsset);

	// 我们没能绑定到任何委托，所以之后永远不会再有更新了
	if (!bCouldSucceed)
	{
		SetReadyToDestroy();
	}
}

// 统一广播出口：先把外观监听从旧队伍解绑，再广播当前队伍与外观，最后绑到新队伍上
void UAsyncAction_ObserveTeamColors::BroadcastChange(int32 NewTeam, const UTenkaichiTeamDisplayAsset* DisplayAsset)
{
	UWorld* World = GEngine->GetWorldFromContextObject(TeamInterfaceObj.Get(), EGetWorldErrorMode::LogAndReturnNull);
	UTenkaichiTeamSubsystem* TeamSubsystem = UWorld::GetSubsystem<UTenkaichiTeamSubsystem>(World);

	const bool bTeamChanged = (LastBroadcastTeamId != NewTeam);

	// 停止监听旧队伍
	if ((TeamSubsystem != nullptr) && bTeamChanged && (LastBroadcastTeamId != INDEX_NONE))
	{
		TeamSubsystem->GetTeamDisplayAssetChangedDelegate(LastBroadcastTeamId).RemoveAll(this);
	}

	// 广播
	LastBroadcastTeamId = NewTeam;
	OnTeamChanged.Broadcast(NewTeam != INDEX_NONE, NewTeam, DisplayAsset);

	// 开始监听新队伍
	if ((TeamSubsystem != nullptr) && bTeamChanged && (NewTeam != INDEX_NONE))
	{
		TeamSubsystem->GetTeamDisplayAssetChangedDelegate(NewTeam).AddDynamic(this, &ThisClass::OnDisplayAssetChanged);
	}
}

// 被监听对象换队时的回调：查出新队伍的外观资产后统一广播
void UAsyncAction_ObserveTeamColors::OnWatchedAgentChangedTeam(UObject* TeamAgent, int32 OldTeam, int32 NewTeam)
{
	UTenkaichiTeamDisplayAsset* DisplayAsset = UTenkaichiTeamStatics::GetTeamDisplayAsset(TeamAgent, NewTeam);
	BroadcastChange(NewTeam, DisplayAsset);
}

// 当前队伍外观资产被修改时的回调：沿用上次广播的队伍号再播一次
void UAsyncAction_ObserveTeamColors::OnDisplayAssetChanged(const UTenkaichiTeamDisplayAsset* DisplayAsset)
{
	BroadcastChange(LastBroadcastTeamId, DisplayAsset);
}
```

|----|------|--------|

#### ③ 为什么要存两份（和 P07.9 的关键差异）

P07.9 是这样的：

```cpp
static UAsyncAction_ObserveTeam* InternalObserveTeamChanges(TScriptInterface<ITenkaichiTeamAgentInterface> TeamActor)
{
	...
	Action->TeamInterfacePtr = TeamActor;   // ← 直接赋值，TScriptInterface → TWeakInterfacePtr
	Action->RegisterWithGameInstance(TeamActor.GetObject());
}
```

它靠 `TScriptInterface` 形参做隐式转换，**只要一份**。

本类**没有 `TScriptInterface` 形参**，所以：

```cpp
Action->TeamInterfacePtr = TWeakInterfacePtr<ITenkaichiTeamAgentInterface>(TeamAgent);  // ← 显式构造
Action->TeamInterfaceObj = TeamAgent;                                                    // ← 裸 UObject 弱引用
Action->RegisterWithGameInstance(TeamAgent);                                             // ← 直接用裸指针
```

| 成员 | 用途 |
|------|------|
| `TeamInterfacePtr` | 调 `GetGenericTeamId()`、绑**换队**委托（需要接口类型） |
| `TeamInterfaceObj` | 传给 `GetWorldFromContextObject()` / `GetTeamDisplayAsset(WorldContextObject, ...)`（它们收 `const UObject*`） |

> 💡 注意 `TWeakInterfacePtr<ITenkaichiTeamAgentInterface>(TeamAgent)` 这种**显式构造**写法——
> 从裸 `UObject*` 直接构造一个接口弱引用。Lyra 原样，别改成 `Cast`。

#### ④ `RegisterWithGameInstance` 为什么必须调

| 问题 | 答案 |
|------|------|
| **它是谁的方法？** | 祖父类 `UBlueprintAsyncActionBase`（`BlueprintAsyncActionBase.h:34`） |
| **参数是啥？** | `const UObject* WorldContextObject`——传"能找到 World 的对象"即可，这里传被监听对象本身 |
| **不调会怎样？** | 本对象只是普通 `UObject`，**没有任何 UPROPERTY 引用它** → **下一帧 GC 直接回收** → 节点永远不回调 |
| **调了之后？** | 挂到 GameInstance 保活列表，**直到 `SetReadyToDestroy()` 才释放** |

> 💡 注意这里传的是 **`TeamAgent`（裸指针）**，而 P07.9 传的是 **`TeamActor.GetObject()`**（从 `TScriptInterface` 取）。
> 结果一样，写法因形参类型不同而异。

### 4.4 `SetReadyToDestroy` —— 退订（第 33~42 行）

```cpp
// 节点准备销毁时的收尾：先让父类解除 GameInstance 注册，再从"换队"委托上解绑本对象的所有绑定
void UAsyncAction_ObserveTeamColors::SetReadyToDestroy()
{
	Super::SetReadyToDestroy();

	// 如果我们被取消了，需要把所有可能监听过的东西都解绑
	if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
	{
		TeamInterface->GetTeamChangedDelegateChecked().RemoveAll(this);

```

| 步 | 说明 |
|----|------|
| `Super::SetReadyToDestroy()` | 先让父类解除 GameInstance 注册（销户） |
| `TeamInterfacePtr.Get()` | 弱引用取值。**对象已销毁 → 返回 `nullptr`，整段跳过**（安全！） |
| `RemoveAll(this)` | 从**换队**委托上移除本对象的所有绑定 |

> 🔥 **这段代码和 P07.9 一模一样**——只有一个 `RemoveAll`。
>
> ⚠️ **它只解绑了"换队"委托，没有解绑"队伍外观变更"委托**（`TeamSubsystem` 那个）。
> 这是 **Lyra 原样**。别"顺手补一个"（铁律 16）。
> （真要追究：`TeamSubsystem` 是 World 子系统，World 销毁时委托随之消失，不会野指针。）

> 🔥 **为什么必须退订（最常见的崩溃源）**
> 委托是**强持有**绑定的：你 `AddDynamic` 之后，被监听对象手里就攥着你的指针。
> 先销毁、不解绑 → 下次换队时委托会调用已销毁对象的成员函数 → **野指针崩溃**。

### 4.5 `Activate` —— 订阅 + 首播（第 44~73 行）

```cpp
// 节点激活：取当前队伍与外观资产、绑定"换队"委托、立刻广播一次，取不到 World 或绑定失败则自我销毁
void UAsyncAction_ObserveTeamColors::Activate()
{
	bool bCouldSucceed = false;
	int32 CurrentTeamIndex = INDEX_NONE;
	UTenkaichiTeamDisplayAsset* CurrentDisplayAsset = nullptr;

	if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
	{
		if (UWorld* World = GEngine->GetWorldFromContextObject(TeamInterfaceObj.Get(), EGetWorldErrorMode::LogAndReturnNull))
		{
			// 取得当前的队伍信息
			CurrentTeamIndex = GenericTeamIdToInteger(TeamInterface->GetGenericTeamId());
			CurrentDisplayAsset = UTenkaichiTeamStatics::GetTeamDisplayAsset(World, CurrentTeamIndex);

			// 监听未来的队伍变化
			TeamInterface->GetTeamChangedDelegateChecked().AddDynamic(this, &ThisClass::OnWatchedAgentChangedTeam);

			bCouldSucceed = true;
		}
	}

	// 广播一次，让使用者拿到当前状态
	BroadcastChange(CurrentTeamIndex, CurrentDisplayAsset);

	// 我们没能绑定到任何委托，所以之后永远不会再有更新了
	if (!bCouldSucceed)
	{
		SetReadyToDestroy();
	}
}
```

#### 第 1 步：三个局部变量（比 P07.9 多一个）

```cpp

int32 CurrentTeamIndex = INDEX_NONE;
UTenkaichiTeamDisplayAsset* CurrentDisplayAsset = nullptr;   // ★ 新增
```

| 变量 | 初值 | 含义 |
|------|------|------|
| `bCouldSucceed` | `false` | "我到底绑定成功了没"——决定最后要不要自毁 |
| `CurrentTeamIndex` | `INDEX_NONE`（`-1`） | 当前队伍号。初值就是"没队伍" |
| `CurrentDisplayAsset` | `nullptr` | ★ **新增**：当前外观资产。初值就是"没外观" |

> 💡 三个初值都代表"**什么都没有**"的合法状态。
> 这样即使后面的 `if` 一层都不进，广播出去的也是合法的 `(false, -1, nullptr)`。

#### 第 2 步：两层 `if`（★ 本类比 P07.9 多的一层）

```cpp

{
	if (UWorld* World = GEngine->GetWorldFromContextObject(TeamInterfaceObj.Get(), EGetWorldErrorMode::LogAndReturnNull))

| 层 | 判什么 | 失败会怎样 |
|----|--------|-----------|
| 外层 | 被监听对象还活着吗（弱引用取值） | 拿不到接口 → 后面全跳过 |
| 内层 ★ | **能找到 World 吗** | 拿不到 World → **查不到外观资产** → `bCouldSucceed` 保持 `false` |

> 📌 **为什么必须多这一层？**
> 因为 `UTenkaichiTeamStatics::GetTeamDisplayAsset(WorldContextObject, TeamId)` 内部要：
> `GEngine->GetWorldFromContextObject(...)` → `World->GetSubsystem<UTenkaichiTeamSubsystem>()` → 查 `TeamMap`。
> **没有 World 就整条链断掉**。P07.9 不需要查外观，所以不需要 World。

> 📌 **`EGetWorldErrorMode::LogAndReturnNull`** = 拿不到就**打日志并返回 `nullptr`**，不崩溃。
> 引擎 3 档错误模式之一（`Assert` / `LogAndReturnNull` / `ReturnNull`）。
> 定义在 `Engine.h:61` 上方注释区。

#### 第 3 步：取当前队伍 + 外观

| 调用 | 来源 |

| `GetGenericTeamId()` | **引擎** `IGenericTeamAgentInterface` → `FGenericTeamId` |
| `GenericTeamIdToInteger()` | **P07.1** `TenkaichiTeamAgentInterface.h:19` 内联 |
| `UTenkaichiTeamStatics::GetTeamDisplayAsset(World, TeamId)` | **P07.7** `TenkaichiTeamStatics.h:30` / `.cpp:49` |

> 💡 注意这里传的是 **`World`**（不是 `TeamAgent`）——
> 因为已经在第 2 步拿到了 World，直接用它当 WorldContextObject。
> 对比 `OnWatchedAgentChangedTeam` 里传的是 **`TeamAgent`**（那里没有现成的 World 变量）。

#### 第 4 步：绑定"换队"委托

> 那个是在 `BroadcastChange` 里绑的（见 4.6）。

#### 第 5 步：首播（★ 用的是 `BroadcastChange` 不是直接 `Broadcast`）

**注意这行在两层 `if` 外面**——不管绑没绑成功都会执行一次。

> 📌 **这是和 P07.9 的另一个差异**：
> P07.9 是 `OnTeamChanged.Broadcast(...)` **直接广播**；
> 本类是 **`BroadcastChange(...)`** —— 因为首播时**顺便要把外观监听绑到当前队伍上**（见 4.6）。

#### 第 6 步：绑定失败就自毁

```cpp

	{
		SetReadyToDestroy();
	}

**什么情况会走到这里？** 两种：

1. `TeamInterfacePtr.Get()` 返 `nullptr`（被监听对象已销毁 or 从一开始就是空）
2. `GetWorldFromContextObject` 返 `nullptr`（对象不在任何世界里）← ★ **本类新增的失败路径**

这时委托绑不上 → 之后永远不会再有更新 → 主动 `SetReadyToDestroy()` 销户放 GC 回收。

> 💡 但注意：**首播已经执行过了**（第 5 步在 if 外）。
> 所以即使失败，蓝图也已经收到了一次 `(false, -1, nullptr)`。
> **Lyra 的顺序安排很讲究：先给结论，再决定要不要留下。**（和 P07.9 一致）

### 4.6 `BroadcastChange` —— 搬家的广播员（第 75~97 行，★ 本文件灵魂）

```cpp
// 统一广播出口：先把外观监听从旧队伍解绑，再广播当前队伍与外观，最后绑到新队伍上
void UAsyncAction_ObserveTeamColors::BroadcastChange(int32 NewTeam, const UTenkaichiTeamDisplayAsset* DisplayAsset)
{
	UWorld* World = GEngine->GetWorldFromContextObject(TeamInterfaceObj.Get(), EGetWorldErrorMode::LogAndReturnNull);
	UTenkaichiTeamSubsystem* TeamSubsystem = UWorld::GetSubsystem<UTenkaichiTeamSubsystem>(World);

	const bool bTeamChanged = (LastBroadcastTeamId != NewTeam);

	// 停止监听旧队伍
	if ((TeamSubsystem != nullptr) && bTeamChanged && (LastBroadcastTeamId != INDEX_NONE))
	{
		TeamSubsystem->GetTeamDisplayAssetChangedDelegate(LastBroadcastTeamId).RemoveAll(this);
	}

	// 广播
	LastBroadcastTeamId = NewTeam;
	OnTeamChanged.Broadcast(NewTeam != INDEX_NONE, NewTeam, DisplayAsset);

	// 开始监听新队伍
	if ((TeamSubsystem != nullptr) && bTeamChanged && (NewTeam != INDEX_NONE))
	{
		TeamSubsystem->GetTeamDisplayAssetChangedDelegate(NewTeam).AddDynamic(this, &ThisClass::OnDisplayAssetChanged);
	}
}
```

#### 第 0 步：取 World + TeamSubsystem

```cpp
UWorld* World = GEngine->GetWorldFromContextObject(TeamInterfaceObj.Get(), EGetWorldErrorMode::LogAndReturnNull);
UTenkaichiTeamSubsystem* TeamSubsystem = UWorld::GetSubsystem<UTenkaichiTeamSubsystem>(World);
```

| 行 | 说明 |
|----|------|
| 第 1 行 | 用 `TeamInterfaceObj`（裸 UObject 弱引用）找 World —— **这就是为什么要存第二个弱引用** |
| 第 2 行 | `UWorld::GetSubsystem<T>(World)` 是**静态方法**，World 为 `nullptr` 时安全返回 `nullptr` |

> 📌 所以后面两个 `if` 都必须先判 `TeamSubsystem != nullptr`。

#### 第 1 步：算 `bTeamChanged`

```cpp
const bool bTeamChanged = (LastBroadcastTeamId != NewTeam);
```

| 场景 | `LastBroadcastTeamId` | `NewTeam` | `bTeamChanged` | 要不要搬监听 |
|------|----------------------|-----------|----------------|-------------|
| **首次广播** | `INDEX_NONE(-1)` | `1` | `true` | 只绑新的（旧的 `-1` 跳过） |
| **换队** | `1` | `2` | `true` | **解绑 1 + 绑定 2** |
| **只改颜色** | `1` | `1` | **`false`** | **不搬**，只广播 |

> 💡 这就是总 md 里说的 **T2 场景**（策划改颜色）：`bTeamChanged == false` → 跳过两段搬家 `if`，只执行广播。

#### 第 2 步：停止监听旧队伍

```cpp
if ((TeamSubsystem != nullptr) && bTeamChanged && (LastBroadcastTeamId != INDEX_NONE))
{
	TeamSubsystem->GetTeamDisplayAssetChangedDelegate(LastBroadcastTeamId).RemoveAll(this);
}
```

三个条件的含义：

| 条件 | 为什么 |
|------|--------|
| `TeamSubsystem != nullptr` | 没子系统就无从解绑 |
| `bTeamChanged` | 队伍没变就不用搬 |
| **`LastBroadcastTeamId != INDEX_NONE`** | ★ **首次广播时是 `-1`**，没有"旧队伍"可解绑 → 跳过 |

#### 第 3 步：广播（注意顺序！）

```cpp
LastBroadcastTeamId = NewTeam;
OnTeamChanged.Broadcast(NewTeam != INDEX_NONE, NewTeam, DisplayAsset);
```

**先更新状态，再广播。**

> ⚠️ **为什么先更新 `LastBroadcastTeamId`？**
> 因为蓝图收到广播后**可能立即反问**（比如调 `GetTeamDisplayAsset`），
> 也可能**在回调里触发新的 `BroadcastChange`**（递归）。
> 先把状态更新到最新，任何重入都不会读到脏的旧值。

| 参数 | 值 |
|------|-----|
| `bTeamSet` | `NewTeam != INDEX_NONE` → 有队 `true`，没队 `false` |
| `TeamId` | 队号，没队时 `-1` |
| `DisplayAsset` | 调用方查好的外观资产（**本函数不负责查**） |

#### 第 4 步：开始监听新队伍

```cpp
if ((TeamSubsystem != nullptr) && bTeamChanged && (NewTeam != INDEX_NONE))
{
	TeamSubsystem->GetTeamDisplayAssetChangedDelegate(NewTeam).AddDynamic(this, &ThisClass::OnDisplayAssetChanged);
}
```

三个条件：

| 条件 | 为什么 |
|------|--------|
| `TeamSubsystem != nullptr` | 没子系统就无从绑定 |
| `bTeamChanged` | 队伍没变就不用搬 |
| **`NewTeam != INDEX_NONE`** | ★ **没队伍就无可监听**（没有"无队伍"这个桶） |

#### 为什么要"搬家"（本方法存在的根本原因 ★）

`TeamSubsystem` 的委托是**按队伍 ID 分桶**的：

```cpp
// TenkaichiTeamSubsystem.cpp:445
FOnTenkaichiTeamDisplayAssetChangedDelegate& UTenkaichiTeamSubsystem::GetTeamDisplayAssetChangedDelegate(int32 TeamId)
{
	return TeamMap.FindOrAdd(TeamId).OnTeamDisplayAssetChanged;
}
```

你监听的是"**1 队**的外观"。对象换到 2 队了，如果不搬：

- 还在收 **1 队**的改版通知 → **错的**（跟你无关了）
- **2 队**改版了反而收不到 → **漏的**（你现在是 2 队）

> 💡 **类比**：你从 1 号楼搬到 2 号楼，**订阅的牛奶得改送到新地址**。
> `BroadcastChange` 就是那个"**广播 + 顺便改牛奶地址**"的管家。
>
> 📌 **顺序很讲究**：先解绑旧的 → 再广播 → 最后绑新的。
> 这样广播时蓝图拿到的 `DisplayAsset` 是新的，监听也已经切换到新队伍了。

#### 三个调用入口汇总

```
Activate()                  → BroadcastChange(CurrentTeamIndex, CurrentDisplayAsset)   ← 首播（bTeamChanged 恒 true）
OnWatchedAgentChangedTeam() → BroadcastChange(NewTeam, 新查出的 DisplayAsset)          ← 换队（true）
OnDisplayAssetChanged()     → BroadcastChange(LastBroadcastTeamId, 新 DisplayAsset)    ← 外观变（false）
```

### 4.7 `OnWatchedAgentChangedTeam` —— 转播（换队）（第 99~103 行）

```cpp
// 被监听对象换队时的回调：查出新队伍的外观资产后统一广播
void UAsyncAction_ObserveTeamColors::OnWatchedAgentChangedTeam(UObject* TeamAgent, int32 OldTeam, int32 NewTeam)
{
	UTenkaichiTeamDisplayAsset* DisplayAsset = UTenkaichiTeamStatics::GetTeamDisplayAsset(TeamAgent, NewTeam);
	BroadcastChange(NewTeam, DisplayAsset);
}
```

| 行 | 说明 |
|----|------|
| `GetTeamDisplayAsset(TeamAgent, NewTeam)` | ★ 用 **`TeamAgent`** 当 WorldContextObject（回调参数里现成的），查出**新队伍**的外观资产 |
| `BroadcastChange(NewTeam, DisplayAsset)` | 转交统一出口（会自动把监听从旧队搬到新队） |

> 📌 **和 P07.9 的对比**：
> P07.9 这里是 `OnTeamChanged.Broadcast(NewTeam != INDEX_NONE, NewTeam);` —— **直接广播，一行完事**。
> 本类因为**要查外观 + 要搬监听**，所以变成"**查 → 转交 `BroadcastChange`**"。

> 💡 **三个参数里 `TeamAgent` 和 `OldTeam` 的用法**：
> `TeamAgent` 在这里被用来当 WorldContextObject 了（P07.9 里它完全没用到）。
> `OldTeam` 依然没用到——保留只是为匹配委托签名。Lyra 原样。

### 4.8 `OnDisplayAssetChanged` —— 转播（外观变）（第 105~108 行）

```cpp
// 当前队伍外观资产被修改时的回调：沿用上次广播的队伍号再播一次
void UAsyncAction_ObserveTeamColors::OnDisplayAssetChanged(const UTenkaichiTeamDisplayAsset* DisplayAsset)
{
	BroadcastChange(LastBroadcastTeamId, DisplayAsset);
}
```

| 点 | 说明 |
|----|------|
| 触发者 | `TeamSubsystem::NotifyTeamDisplayAssetModified()` → 遍历 `TeamMap` 广播 `OnTeamDisplayAssetChanged`（`.cpp:432-442`） |
| 队伍号 | **沿用 `LastBroadcastTeamId`**（队伍没变，只是外观变了） |
| 外观资产 | 用回调带来的 `DisplayAsset` 参数 |
| `bTeamChanged` | 因为 `LastBroadcastTeamId == NewTeam` → **`false`** → 跳过搬家，只广播 |

> 💡 **这就是总 md 里 T2 场景（策划改颜色）的完整路径**：
> ```
> 策划改 TeamDisplayAsset → PostEditChangeProperty → TeamSubsystem::NotifyTeamDisplayAssetModified
>   → OnTeamDisplayAssetChanged.Broadcast → OnDisplayAssetChanged(新外观)
>   → BroadcastChange(LastBroadcastTeamId, 新外观)  ← bTeamChanged = false，只广播
>   → OnTeamChanged.Broadcast(true, 队号, 新外观) → 蓝图 UI 刷新颜色
> ```

> ⚠️ **注意参数类型**：回调收的是 `const UTenkaichiTeamDisplayAsset*`（**带 const**），
> 而 `BroadcastChange` 的形参也是 `const UTenkaichiTeamDisplayAsset*` → **类型完全匹配，可直接传**。
> 对比 `OnWatchedAgentChangedTeam` 里查出来的是**非 const** 的 `UTenkaichiTeamDisplayAsset*`，
> 传给 const 形参是**隐式加 const**，合法。

### 五个必须记住的点

1. **`BroadcastChange` 是本类灵魂**：解绑旧队 → 更新状态 → 广播 → 绑定新队，**顺序不能乱**。
2. **`RegisterWithGameInstance` 不能漏**——漏了对象下一帧被 GC 回收，节点永远不回调。
3. **`Activate()` 的首播在两层 `if` 外面**——绑定成功与否都会给蓝图一次当前值。
4. **`Activate()` 比 P07.9 多一层 World 判断**——查外观资产必须先有 World。
5. **构造函数不能省**——`.h` 用了 `GENERATED_UCLASS_BODY()`，少了会链接报错。
