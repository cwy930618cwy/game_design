```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#include "Teams/AsyncAction_ObserveTeam.h"

#include "Teams/TenkaichiTeamAgentInterface.h"

#include UE_INLINE_GENERATED_CPP_BY_NAME(AsyncAction_ObserveTeam)

// 构造函数：把对象初始化器原样转交给父类，不做任何额外初始化
UAsyncAction_ObserveTeam::UAsyncAction_ObserveTeam(const FObjectInitializer& ObjectInitializer)
	: Super(ObjectInitializer)
{
}

// 监听指定对象上的队伍变化（蓝图异步节点入口），内部转调真正的创建函数
UAsyncAction_ObserveTeam* UAsyncAction_ObserveTeam::ObserveTeam(UObject* TeamAgent)
{
	return InternalObserveTeamChanges(TeamAgent);
}

// 内部创建入口：判定传入对象非空后，造出本异步节点对象、记住监听目标并注册到 GameInstance 防 GC
UAsyncAction_ObserveTeam* UAsyncAction_ObserveTeam::InternalObserveTeamChanges(TScriptInterface<ITenkaichiTeamAgentInterface> TeamActor)
{
	UAsyncAction_ObserveTeam* Action = nullptr;

	if (TeamActor != nullptr)
	{
		Action = NewObject<UAsyncAction_ObserveTeam>();
		Action->TeamInterfacePtr = TeamActor;
		Action->RegisterWithGameInstance(TeamActor.GetObject());
	}

	return Action;
}

// 节点准备销毁时的收尾：先让父类解除 GameInstance 注册，再从队伍变化委托上解绑本对象的所有绑定
void UAsyncAction_ObserveTeam::SetReadyToDestroy()
{
	Super::SetReadyToDestroy();

	// 如果我们被取消了，需要把所有可能监听过的东西都解绑
	if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
	{
		TeamInterface->GetTeamChangedDelegateChecked().RemoveAll(this);
	}
}

// 节点激活：绑定队伍变化委托、立刻广播一次当前队伍，绑定失败则自我销毁
void UAsyncAction_ObserveTeam::Activate()
{
	bool bCouldSucceed = false;
	int32 CurrentTeamIndex = INDEX_NONE;

	if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
	{
		CurrentTeamIndex = GenericTeamIdToInteger(TeamInterface->GetGenericTeamId());

		TeamInterface->GetTeamChangedDelegateChecked().AddDynamic(this, &ThisClass::OnWatchedAgentChangedTeam);

		bCouldSucceed = true;
	}

	// 广播一次，让使用者拿到当前状态
	OnTeamChanged.Broadcast(CurrentTeamIndex != INDEX_NONE, CurrentTeamIndex);

	// 我们没能绑定到任何委托，所以之后永远不会再有更新了
	if (!bCouldSucceed)
	{
		SetReadyToDestroy();
	}
}

// 被监听对象换队时的回调：把新队伍号转成蓝图委托再广播一次（旧队号与本对象指针在此不使用）
void UAsyncAction_ObserveTeam::OnWatchedAgentChangedTeam(UObject* TeamAgent, int32 OldTeam, int32 NewTeam)
{
	OnTeamChanged.Broadcast(NewTeam != INDEX_NONE, NewTeam);
}
```

### 4.4 `InternalObserveTeamChanges` —— 造对象（第 19~31 行）

```cpp
// 内部创建入口：判定传入对象非空后，造出本异步节点对象、记住监听目标并注册到 GameInstance 防 GC
UAsyncAction_ObserveTeam* UAsyncAction_ObserveTeam::InternalObserveTeamChanges(TScriptInterface<ITenkaichiTeamAgentInterface> TeamActor)
{
	UAsyncAction_ObserveTeam* Action = nullptr;

	if (TeamActor != nullptr)
	{
		Action = NewObject<UAsyncAction_ObserveTeam>();
		Action->TeamInterfacePtr = TeamActor;
		Action->RegisterWithGameInstance(TeamActor.GetObject());
	}

	return Action;
}
```

三步走：

| 步 | 代码 | 为什么 |
|----|------|--------|
| ① 判空 | `if (TeamActor != nullptr)` | `TScriptInterface` 的 `operator!=` 比较内部 `UObject*` 是否为空。**传空就返回 `nullptr`**，不造对象 |
| ② `NewObject` + 存弱引用 | `Action->TeamInterfacePtr = TeamActor;` | `TWeakInterfacePtr` 可以直接从 `TScriptInterface` 赋值 → **弱引用**保存，不延长对象生命 |
| ③ 注册保活 | `RegisterWithGameInstance(TeamActor.GetObject())` | ★ **防 GC**，见下方详解 |

#### ② 为什么用弱引用（复习 `.h` 4.8）

```cpp
Action->TeamInterfacePtr = TeamActor;   // TWeakInterfacePtr ← TScriptInterface
```

这个 AsyncAction 挂在 GameInstance 上，**可能比被监听的 Actor 活得久**。
强引用会把 Actor 钉住不放 → 泄漏。弱引用则 Actor 销毁后 `Get()` 自动返 `nullptr`。

#### ③ `RegisterWithGameInstance` 为什么必须调（本类最关键的一行）

```cpp
Action->RegisterWithGameInstance(TeamActor.GetObject());
```

| 问题 | 答案 |
|------|------|
| **它是谁的方法？** | 祖父类 `UBlueprintAsyncActionBase`（`BlueprintAsyncActionBase.h:34`） |
| **参数是啥？** | `const UObject* WorldContextObject` —— 传一个"能找到 World 的对象"就行，这里传被监听对象本身 |
| **不调会怎样？** | `UAsyncAction_ObserveTeam` 只是个普通 `UObject`，**没有任何 UPROPERTY 引用它** → **下一帧 GC 直接回收** → 节点永远不回调 |
| **调了之后？** | 挂到 GameInstance 的保活列表里，**直到 `SetReadyToDestroy()` 才释放** |

> 💡 **类比**：`RegisterWithGameInstance` 就像给对象办了张"暂住证"——
> 没有它 GC 会把你当流浪对象清走；办了之后 GameInstance 知道"这人我罩着"，
> 你主动说"我走了"（`SetReadyToDestroy`）它才销户。

> ⚠️ **注意**：注册只是**保活**，不代表节点已激活。激活是引擎另外调 `Activate()` 的事（见 4.6）。

### 4.5 `SetReadyToDestroy` —— 退订（第 33~42 行）

```cpp
// 节点准备销毁时的收尾：先让父类解除 GameInstance 注册，再从队伍变化委托上解绑本对象的所有绑定
void UAsyncAction_ObserveTeam::SetReadyToDestroy()
{
	Super::SetReadyToDestroy();

	// 如果我们被取消了，需要把所有可能监听过的东西都解绑
	if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
	{
		TeamInterface->GetTeamChangedDelegateChecked().RemoveAll(this);
	}
}
```

| 步 | 说明 |
|----|------|
| `Super::SetReadyToDestroy()` | 先让父类解除 GameInstance 注册（销户） |
| `TeamInterfacePtr.Get()` | 弱引用取值。**对象已销毁 → 返回 `nullptr`，整段跳过**（安全！） |
| `RemoveAll(this)` | 从委托上移除**本对象绑定的所有函数** |

> 📌 **为什么用 `RemoveAll(this)` 而不是 `RemoveDynamic(this, &ThisClass::OnWatchedAgentChangedTeam)`？**
> Lyra 用 `RemoveAll`——一次清掉本对象在这个委托上的**所有绑定**，更彻底也更省事。
> 本类其实只绑了一个，两种写法等价，但**照抄 Lyra 的 `RemoveAll`**。

> 🔥 **为什么必须退订（最常见的崩溃源）**
> 委托是**强持有**绑定的：你 `AddDynamic` 之后，被监听对象手里就攥着你的指针。
> 如果你先销毁、不解绑 → 下次换队时委托会调用一个已销毁对象的成员函数 → **野指针崩溃**。
> `SetReadyToDestroy()` 是 Lyra 给"取消/结束"留的统一出口，必须在这里清干净。
>
> **谁来调它？** 两种：
> ① 蓝图调用 `Cancel()`（`UCancellableAsyncAction` 提供）→ 内部转调 `SetReadyToDestroy`
> ② 本类自己——`Activate()` 里绑定失败时主动调（见 4.6 最后）

### 4.6 `Activate` —— 订阅 + 首播（第 44~66 行，本文件核心）

```cpp
// 节点激活：绑定队伍变化委托、立刻广播一次当前队伍，绑定失败则自我销毁
void UAsyncAction_ObserveTeam::Activate()
{
	bool bCouldSucceed = false;
	int32 CurrentTeamIndex = INDEX_NONE;

	if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
	{
		CurrentTeamIndex = GenericTeamIdToInteger(TeamInterface->GetGenericTeamId());

		TeamInterface->GetTeamChangedDelegateChecked().AddDynamic(this, &ThisClass::OnWatchedAgentChangedTeam);

		bCouldSucceed = true;
	}

	// 广播一次，让使用者拿到当前状态
	OnTeamChanged.Broadcast(CurrentTeamIndex != INDEX_NONE, CurrentTeamIndex);

	// 我们没能绑定到任何委托，所以之后永远不会再有更新了
	if (!bCouldSucceed)
	{
		SetReadyToDestroy();
	}
}
```

#### 第 1 步：两个局部变量

```cpp
bool bCouldSucceed = false;
int32 CurrentTeamIndex = INDEX_NONE;
```

| 变量 | 初值 | 含义 |
|------|------|------|
| `bCouldSucceed` | `false` | "我到底绑定成功了没"——决定最后要不要自毁 |
| `CurrentTeamIndex` | `INDEX_NONE`（即 `-1`） | 当前队伍号。**初值就是"没队伍"**，这样即使后面 if 不进，广播出去的也是合法的"无队"状态 |

#### 第 2 步：取弱引用 + 拿当前队伍

```cpp
if (ITenkaichiTeamAgentInterface* TeamInterface = TeamInterfacePtr.Get())
{
	CurrentTeamIndex = GenericTeamIdToInteger(TeamInterface->GetGenericTeamId());
	...
}
```

| 调用 | 来源 | 说明 |
|------|------|------|
| `TeamInterfacePtr.Get()` | `TWeakInterfacePtr` | 弱引用取值，对象没了返回 `nullptr`（整段跳过） |
| `GetGenericTeamId()` | **引擎** `IGenericTeamAgentInterface` | 拿 `FGenericTeamId`（引擎的队号类型） |
| `GenericTeamIdToInteger()` | **P07.1** `TenkaichiTeamAgentInterface.h:19` | 转成 `int32`；`NoTeam` → `INDEX_NONE` |

> 💡 注意这里的**判空 + 赋值合一**写法（`if (T* P = ...Get())`）：C++17 风格的紧凑写法，UE 代码里很常见。

#### 第 3 步：绑定委托（订阅）

```cpp
TeamInterface->GetTeamChangedDelegateChecked().AddDynamic(this, &ThisClass::OnWatchedAgentChangedTeam);
```

| 元素 | 说明 |
|------|------|
| `GetTeamChangedDelegateChecked()` | **P07.1** `.h:48` 内联实现。取委托引用，若实现类没覆写 `GetOnTeamIndexChangedDelegate()` 会 `check` 崩溃 |
| `AddDynamic(this, &ThisClass::Func)` | UHT 宏：把 `this` 对象的 `Func` 绑到委托上 |
| `&ThisClass::OnWatchedAgentChangedTeam` | `ThisClass` = `UAsyncAction_ObserveTeam`（`GENERATED_UCLASS_BODY()` 生成的 typedef），等价于写全类名 |

> ⚠️ **签名必须严格匹配**：
> 委托 `FOnTenkaichiTeamIndexChangedDelegate`（P07.1 `.h:16`）是 `(UObject*, int32, int32)`；
> `OnWatchedAgentChangedTeam(UObject*, int32, int32)` 完全一致。
> **对不上编译不报错，但运行时绑定失败、回调永不触发**——这是最隐蔽的坑。

#### 第 4 步：首次广播（Lyra 的刻意设计 ★）

```cpp
	// 广播一次，让使用者拿到当前状态
	OnTeamChanged.Broadcast(CurrentTeamIndex != INDEX_NONE, CurrentTeamIndex);
```

**注意这行在 `if` 外面**——不管绑没绑成功都会执行一次。

| 参数 | 值 |
|------|-----|
| `bTeamSet` | `CurrentTeamIndex != INDEX_NONE` → 有队 = `true`，没队 = `false` |
| `TeamId` | 队号，没队时是 `-1` |

**为什么要先播一次？** 蓝图用法是：

```
BeginPlay → [Observe Team] → OnTeamChanged 事件里刷新 UI
```

如果不首播，蓝图就得**再额外调一次** `TeamStatics::FindTeamFromObject` 拿初始队伍 → 两处逻辑、容易漏。
Lyra 让它**一激活就把当前值送上来**，蓝图只写一个事件处理就够。

> 📌 这就是 `.h` 注释里那句"**它会立刻触发一次，给出当前的队伍归属**"的落地点。

#### 第 5 步：绑定失败就自毁

```cpp
	// 我们没能绑定到任何委托，所以之后永远不会再有更新了
	if (!bCouldSucceed)
	{
		SetReadyToDestroy();
	}
```

**什么情况会走到这里？** `TeamInterfacePtr.Get()` 返回 `nullptr`（被监听对象已销毁 or 从一开始就是空）。

这时：委托绑不上 → 之后永远不会再有更新 → **留着这个对象没意义** → 主动 `SetReadyToDestroy()` 销户放 GC 回收。

> 💡 但注意：**首播已经执行过了**（第 4 步在 if 外）。
> 所以即使绑定失败，蓝图也已经收到了一次 `(false, -1)` —— 告诉它"这对象没队伍"。
> **Lyra 的顺序安排很讲究**：先给结论，再决定要不要留下。

### 4.7 `OnWatchedAgentChangedTeam` —— 转播（第 68~71 行）

```cpp
// 被监听对象换队时的回调：把新队伍号转成蓝图委托再广播一次（旧队号与本对象指针在此不使用）
void UAsyncAction_ObserveTeam::OnWatchedAgentChangedTeam(UObject* TeamAgent, int32 OldTeam, int32 NewTeam)
{
	OnTeamChanged.Broadcast(NewTeam != INDEX_NONE, NewTeam);
}
```

- **回调入口**：被监听对象换队时，P07.1 的 `ConditionalBroadcastTeamChanged` 会 `Broadcast(ThisObj, OldTeamIndex, NewTeamIndex)` → 触发到这里
- 只有一个动作：**转成蓝图委托再广播一次**
- 三个参数里**只用 `NewTeam`**，`TeamAgent` 和 `OldTeam` **都没用到**——Lyra 原样（保留签名是为了匹配委托）

> 📌 **注意两个委托的参数语义转换**：
> | 委托 | 参数 |
> |------|------|
> | C++ 侧 `FOnTenkaichiTeamIndexChangedDelegate` | `(谁, 旧队, 新队)` —— 3 个参数 |
> | 蓝图侧 `FTeamObservedAsyncDelegate` | `(有没有队, 队号)` —— 2 个参数 |
>
> 本函数就是**两层的翻译官**：把"3 参数的 C++ 事件"翻译成"2 参数的蓝图事件"。
> 蓝图不需要知道"谁"和"旧队"（它本来就在监听特定对象），只关心"现在几队"。

> ⚠️ `NewTeam != INDEX_NONE` 这个判断在首播和转播里**重复了两次**——
> 因为两种路径都得算 `bTeamSet`。Lyra 没有抽成辅助函数，**照抄，不要"顺手重构"**（铁律 16）。

## 五、五个函数的调用时序（一张图）

```
蓝图拖出 [Observe Team]，传入一个 Actor
   │
   ├─► ObserveTeam(Actor)                    ← 静态工厂（蓝图节点入口）
   │      └─► InternalObserveTeamChanges(Actor)
   │             ├─ Actor != nullptr ? 
   │             ├─ NewObject<UAsyncAction_ObserveTeam>()      ← 造对象
   │             ├─ TeamInterfacePtr = Actor                   ← 弱引用记住"监听谁"
   │             └─ RegisterWithGameInstance(Actor)            ← 办暂住证（防 GC）
   │
   ├─► [引擎自动] Activate()                  ← 委托已绑定好后调用
   │      ├─ TeamInterfacePtr.Get() 取接口指针
   │      ├─ CurrentTeamIndex = GenericTeamIdToInteger(GetGenericTeamId())
   │      ├─ AddDynamic(this, &OnWatchedAgentChangedTeam)      ← 订阅
   │      ├─ OnTeamChanged.Broadcast(有队?, 队号)              ← ★ 首播当前值
   │      └─ 绑定失败? → SetReadyToDestroy()                   ← 自毁
   │
   ├─► [以后某刻] 对象换队
   │      └─ P07.1 ConditionalBroadcastTeamChanged 广播
   │             └─► OnWatchedAgentChangedTeam(谁, 旧, 新)      ← 回调
   │                    └─► OnTeamChanged.Broadcast(新!=NONE, 新)  ← 转播给蓝图
   │
   └─► [蓝图调 Cancel() 或对象结束]
          └─► SetReadyToDestroy()
                 ├─ Super::SetReadyToDestroy()                 ← 销户（GameInstance）
                 └─ GetTeamChangedDelegateChecked().RemoveAll(this)  ← 退订（防野指针）
```

### 四个必须记住的点

1. **`RegisterWithGameInstance` 不能漏**——漏了对象下一帧被 GC 回收，节点永远不回调。
2. **`SetReadyToDestroy` 里必须 `RemoveAll(this)`**——漏了会野指针崩溃。和 `AddDynamic` 成对。
3. **`Activate()` 的首播在 `if` 外面**——绑定成功与否都会给蓝图一次当前值，这是 Lyra 的刻意设计。
4. **构造函数不能省**——`.h` 用了 `GENERATED_UCLASS_BODY()`，少了会链接报错。
