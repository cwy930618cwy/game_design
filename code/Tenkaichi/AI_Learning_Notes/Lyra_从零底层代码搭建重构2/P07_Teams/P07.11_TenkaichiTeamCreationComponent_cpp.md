// 返回当前人数最少的队伍 ID，若没有任何有效队伍则返回 INDEX_NONE
int32 UTenkaichiTeamCreationComponent::GetLeastPopulatedTeamID() const
{
	const int32 NumTeams = TeamsToCreate.Num();
	if (NumTeams > 0)
	{
		TMap<int32, uint32> TeamMemberCounts;
		TeamMemberCounts.Reserve(NumTeams);

		for (const auto& KVP : TeamsToCreate)
		{
			const int32 TeamId = KVP.Key;
			TeamMemberCounts.Add(TeamId, 0);
		}

		AGameStateBase* GameState = GetGameStateChecked<AGameStateBase>();
		for (APlayerState* PS : GameState->PlayerArray)
		{
			if (ATenkaichiPlayerState* TenkaichiPS = Cast<ATenkaichiPlayerState>(PS))
			{
				const int32 PlayerTeamID = TenkaichiPS->GetTeamId();

				if ((PlayerTeamID != INDEX_NONE) && !TenkaichiPS->IsInactive())	// 不统计未分配队伍或已断线的玩家
				{
					check(TeamMemberCounts.Contains(PlayerTeamID))
					TeamMemberCounts[PlayerTeamID] += 1;
				}
			}
		}

		// 先按队伍人数从少到多排序，人数相同再按队伍 ID 排序
		int32 BestTeamId = INDEX_NONE;
		uint32 BestPlayerCount = TNumericLimits<uint32>::Max();
		for (const auto& KVP : TeamMemberCounts)
		{
			const int32 TestTeamId = KVP.Key;
			const uint32 TestTeamPlayerCount = KVP.Value;

			if (TestTeamPlayerCount < BestPlayerCount)
			{
				BestTeamId = TestTeamId;
				BestPlayerCount = TestTeamPlayerCount;
			}
			else if (TestTeamPlayerCount == BestPlayerCount)
			{
				if ((TestTeamId < BestTeamId) || (BestTeamId == INDEX_NONE))
				{
					BestTeamId = TestTeamId;
					BestPlayerCount = TestTeamPlayerCount;
				}
			}
		}

		return BestTeamId;
	}

	return INDEX_NONE;
}

### 4.11 `GetLeastPopulatedTeamID`（第 143~189 行）—— 本文件最"有内容"的函数

```cpp
// 返回当前人数最少的队伍 ID，若没有任何有效队伍则返回 INDEX_NONE
int32 UTenkaichiTeamCreationComponent::GetLeastPopulatedTeamID() const
{
	const int32 NumTeams = TeamsToCreate.Num();
	if (NumTeams > 0)
	{
		TMap<int32, uint32> TeamMemberCounts;
		TeamMemberCounts.Reserve(NumTeams);

		for (const auto& KVP : TeamsToCreate)
		{
			const int32 TeamId = KVP.Key;
			TeamMemberCounts.Add(TeamId, 0);
		}

		AGameStateBase* GameState = GetGameStateChecked<AGameStateBase>();
		for (APlayerState* PS : GameState->PlayerArray)
		{
			if (ATenkaichiPlayerState* TenkaichiPS = Cast<ATenkaichiPlayerState>(PS))
			{
				const int32 PlayerTeamID = TenkaichiPS->GetTeamId();

				if ((PlayerTeamID != INDEX_NONE) && !TenkaichiPS->IsInactive())	// 不统计未分配队伍或已断线的玩家
				{
					check(TeamMemberCounts.Contains(PlayerTeamID))
					TeamMemberCounts[PlayerTeamID] += 1;
				}
			}
		}

		// 先按队伍人数从少到多排序，人数相同再按队伍 ID 排序
		int32 BestTeamId = INDEX_NONE;
		uint32 BestPlayerCount = TNumericLimits<uint32>::Max();
		for (const auto& KVP : TeamMemberCounts)
		{
			const int32 TestTeamId = KVP.Key;
			const uint32 TestTeamPlayerCount = KVP.Value;

			if (TestTeamPlayerCount < BestPlayerCount)
			{
				BestTeamId = TestTeamId;
				BestPlayerCount = TestTeamPlayerCount;
			}
			else if (TestTeamPlayerCount == BestPlayerCount)
			{
				if ((TestTeamId < BestTeamId) || (BestTeamId == INDEX_NONE))
				{
					BestTeamId = TestTeamId;
					BestPlayerCount = TestTeamPlayerCount;
				}
			}
		}

		return BestTeamId;
	}

	return INDEX_NONE;
}
```

#### 第 1 步：初始化计数表

```cpp
TMap<int32, uint32> TeamMemberCounts;
TeamMemberCounts.Reserve(NumTeams);

for (const auto& KVP : TeamsToCreate)
{

}
```

| 点 | 说明 |
|----|------|
| `Reserve(NumTeams)` | 预分配空间，避免扩容（性能习惯） |
| 所有队伍初值 `0` | **保证每个配置的队伍都在表里**（哪怕一个人没有） |

> 💡 **为什么要预填 0？** 因为"**0 人的队**"正是**最优先**的选择——
> 如果表里没有它，后面的循环就选不到它。

#### 第 2 步：遍历玩家计数

```cpp

	check(TeamMemberCounts.Contains(PlayerTeamID))

| 条件 | 为什么 |

| `Cast<ATenkaichiPlayerState>` 成功 | 只统计我们自己的 PlayerState |
| `PlayerTeamID != INDEX_NONE` | **未分配队伍的玩家不算**（他们还没进任何队） |
| `!IsInactive()` | **已断线的玩家不算**（`APlayerState` 引擎成员，`PlayerState.h:381`） |

> 📌 `IsInactive()` 引擎源码核实：`PlayerState.h:381` 返回 `bIsInactive`（`:91` 的 `UPROPERTY(ReplicatedUsing=OnRep_bIsInactive)`）。
> 断线玩家的 PlayerState 会**继续留在 `PlayerArray` 里一段时间**，但**不该占位**。

#### 第 3 步：选最优（含平局规则 ★）

		BestPlayerCount = TestTeamPlayerCount;

| 分支 1 `<` | 人更少 → 换 |
| 分支 2 `==` | 人一样多 → **队号小的胜出** |
| `(BestTeamId == INDEX_NONE)` | 处理"第一个就是平局"的边界（`INDEX_NONE` 是 `-1`，比任何队号都小） |

> 💡 **为什么要平局规则？** 保证**确定性**：
> 相同局面下永远得到相同结果，不会因为 `TMap` 遍历顺序（哈希序）而抖动。
> **没有它，分队会变成"随机"**——玩家重连可能被分到不同队。

> ⚠️ 注意两个分支里 `BestPlayerCount = TestTeamPlayerCount;` **都写了**——
> 平局时值其实没变，但 Lyra 原样写了。**照抄**（铁律 14）。

```

      │

      ├─ FindComponentByClass<UTenkaichiExperienceManagerComponent>()   ⚠️ 挂起③
      └─ CallOrRegister_OnExperienceLoaded_HighPriority(OnExperienceLoaded)
             │  （已加载 → 立即回调；未加载 → 等着）
             ▼
[Experience 加载完] OnExperienceLoaded(Experience)                     ⚠️ 挂起④（整函数）
      │
      ├─ HasAuthority() ?
