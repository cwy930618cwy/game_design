// 版权声明（可保留或省略，不影响编译）
#pragma once

#include "GameFeatureAction_WorldActionBase.h"
#include "UObject/ObjectKey.h"

#include "GameFeatureAction_SplitscreenConfig.generated.h"

class UObject;
struct FGameFeatureDeactivatingContext;
struct FGameFeatureStateChangeContext;
struct FWorldContext;

//////////////////////////////////////////////////////////////////////
// UGameFeatureAction_SplitscreenConfig

/**
 * 负责配置分屏的 GameFeatureAction。
 */
UCLASS(MinimalAPI, meta = (DisplayName = "Splitscreen Config"))
class UGameFeatureAction_SplitscreenConfig final : public UGameFeatureAction_WorldActionBase
{
	GENERATED_BODY()

public:
	//~UGameFeatureAction 接口开始
	// 游戏特性停用时调用：撤掉本 Action 投的票。
	virtual void OnGameFeatureDeactivating(FGameFeatureDeactivatingContext& Context) override;
	//~UGameFeatureAction 接口结束

	//~UGameFeatureAction_WorldActionBase 接口开始
	// 在某个 world 里执行：给 viewport 投票禁用分屏。
	virtual void AddToWorld(const FWorldContext& WorldContext, const FGameFeatureStateChangeContext& ChangeContext) override;
	//~UGameFeatureAction_WorldActionBase 接口结束

public:
	// 是否禁用分屏。
	UPROPERTY(EditAnywhere, Category=Action)
	bool bDisableSplitscreen = true;

private:
	// 本 Action 投过票的 viewport 列表。
	TArray<FObjectKey> LocalDisableVotes;

	// 全局投票表：viewport → 票数（静态，所有 Action 实例共享）。
	static TMap<FObjectKey, int32> GlobalDisableVotes;
};

## 二、关键点讲解

### 1. `final` + 继承 `WorldActionBase`

```cpp
class UGameFeatureAction_SplitscreenConfig final : public UGameFeatureAction_WorldActionBase
```

- `final`：终点类（不再被继承）。
- 基类是 **P15.1 的 `WorldActionBase`**（不是直接 `UGameFeatureAction`）——所以它只需要实现 `AddToWorld`（基类处理 world 遍历），额外再覆写 `OnGameFeatureDeactivating` 做撤票。

> 对比 P15.2 直接继承 `UGameFeatureAction`（不需要 world），这个继承 `WorldActionBase`（需要 world）。

### 2. 两个覆写：接口分界标记

```cpp
//~UGameFeatureAction 接口开始
virtual void OnGameFeatureDeactivating(FGameFeatureDeactivatingContext& Context) override;
//~UGameFeatureAction 接口结束

//~UGameFeatureAction_WorldActionBase 接口开始
virtual void AddToWorld(const FWorldContext& WorldContext, const FGameFeatureStateChangeContext& ChangeContext) override;
//~UGameFeatureAction_WorldActionBase 接口结束
```

两组接口分界标记（铁律 36：中文化）。第一组是基类的接口（`UGameFeatureAction`），第二组是 P15.1 的接口（`WorldActionBase`）。

### 3. `static TMap<FObjectKey, int32> GlobalDisableVotes`

```cpp
static TMap<FObjectKey, int32> GlobalDisableVotes;
```

- `static`：**类静态成员**——所有 `UGameFeatureAction_SplitscreenConfig` 实例**共享同一份**投票表。
- 为什么必须静态？因为多个插件各自有一个 Action 实例，但它们要共同决定"这个 viewport 要不要禁用分屏"，必须共享计数。
- `.cpp` 里要**定义**它：`TMap<FObjectKey, int32> UGameFeatureAction_SplitscreenConfig::GlobalDisableVotes;`（静态成员声明在 `.h`、定义在 `.cpp`）。

### 4. `FObjectKey`——对象键

```cpp
TArray<FObjectKey> LocalDisableVotes;
TMap<FObjectKey, int32> GlobalDisableVotes;
```

`FObjectKey` 是 UE 的"对象唯一键"——用对象指针包装成可哈希的 Key（能放 TMap/TArray）。用它而不是裸指针，因为：
- 对象销毁后 `FObjectKey` 还能安全 `ResolveObjectPtr()` 返回 nullptr（不悬空）。
- 可哈希（能作为 TMap 的 Key）。

### 5. `bDisableSplitscreen` 配置项

```cpp
UPROPERTY(EditAnywhere, Category=Action)
bool bDisableSplitscreen = true;
```

编辑器可配（默认 true = 禁用分屏）。注意 `Category=Action`（不是 `Game Feature | XXX`，Lyra 这里简写了）。

### 6. `class UObject;` 前置声明（未使用）

Lyra 源码里前置声明了 `class UObject;`，但实际没用到。按铁律 23 照抄保留。
