// 游戏特性注册时调用：把 Cue 路径真正注册进 GameplayCueManager。
void UTenkaichiGameFeature_AddGameplayCuePaths::OnGameFeatureRegistering(const UGameFeatureData* GameFeatureData, const FString& PluginName, const FString& PluginURL)
{
	TRACE_CPUPROFILER_EVENT_SCOPE(UTenkaichiGameFeature_AddGameplayCuePaths::OnGameFeatureRegistering);
	
	const FString PluginRootPath = TEXT("/") + PluginName;
	// 遍历这个 GameFeature 的所有 Action，找出"加 Cue 路径"的 Action
	for (const UGameFeatureAction* Action : GameFeatureData->GetActions())
	{
		if (const UGameFeatureAction_AddGameplayCuePath* AddGameplayCueGFA = Cast<UGameFeatureAction_AddGameplayCuePath>(Action))
		{
			const TArray<FDirectoryPath>& DirsToAdd = AddGameplayCueGFA->GetDirectoryPathsToAdd();
			
			if (UTenkaichiGameplayCueManager* GCM = UTenkaichiGameplayCueManager::Get())
			{
				UGameplayCueSet* RuntimeGameplayCueSet = GCM->GetRuntimeCueSet();
				// 记下注册前有多少个 Cue
				const int32 PreInitializeNumCues = RuntimeGameplayCueSet ? RuntimeGameplayCueSet->GameplayCueData.Num() : 0;

				for (const FDirectoryPath& Directory : DirsToAdd)
				{
					FString MutablePath = Directory.Path;
					// 把插件相对路径修正成完整路径
					UGameFeaturesSubsystem::FixPluginPackagePath(MutablePath, PluginRootPath, false);
					GCM->AddGameplayCueNotifyPath(MutablePath, /** bShouldRescanCueAssets = */ false);	
				}
				
				// 有新路径就重建运行时库
				if (!DirsToAdd.IsEmpty())
				{
					GCM->InitializeRuntimeObjectLibrary();	
				}

				// 对比注册前后的 Cue 数量，变了就刷新主资产
				const int32 PostInitializeNumCues = RuntimeGameplayCueSet ? RuntimeGameplayCueSet->GameplayCueData.Num() : 0;
				if (PreInitializeNumCues != PostInitializeNumCues)
				{
					GCM->RefreshGameplayCuePrimaryAsset();
				}
			}
		}
	}
}

// 游戏特性注销时调用：把 Cue 路径从 GameplayCueManager 移除。
void UTenkaichiGameFeature_AddGameplayCuePaths::OnGameFeatureUnregistering(const UGameFeatureData* GameFeatureData, const FString& PluginName, const FString& PluginURL)
{
	const FString PluginRootPath = TEXT("/") + PluginName;
	for (const UGameFeatureAction* Action : GameFeatureData->GetActions())
	{
		if (const UGameFeatureAction_AddGameplayCuePath* AddGameplayCueGFA = Cast<UGameFeatureAction_AddGameplayCuePath>(Action))
		{
			const TArray<FDirectoryPath>& DirsToAdd = AddGameplayCueGFA->GetDirectoryPathsToAdd();
			
			if (UGameplayCueManager* GCM = UAbilitySystemGlobals::Get().GetGameplayCueManager())
			{
				int32 NumRemoved = 0;
				for (const FDirectoryPath& Directory : DirsToAdd)
				{
					FString MutablePath = Directory.Path;
					UGameFeaturesSubsystem::FixPluginPackagePath(MutablePath, PluginRootPath, false);
					NumRemoved += GCM->RemoveGameplayCueNotifyPath(MutablePath, /** bShouldRescanCueAssets = */ false);
				}

				ensure(NumRemoved == DirsToAdd.Num());
				
				// 只有确实移除了才重建运行时库
				if (NumRemoved > 0)
				{
					GCM->InitializeRuntimeObjectLibrary();	
				}			
			}
	}
	}
}

## 二、关键点讲解

### 1. "就近 include" 风格

```cpp
#include "AbilitySystemGlobals.h"
#include "GameFeatureAction_AddGameplayCuePath.h"

#include UE_INLINE_GENERATED_CPP_BY_NAME(TenkaichiGameFeaturePolicy)
```

Lyra 把这些 include 放在**文件中途**（刚好在用到它们的函数之前），而不是全部堆在文件顶部。这是"就近 include"风格——一眼看出"这个 include 服务于下面这个函数"。

> `UE_INLINE_GENERATED_CPP_BY_NAME` 也在中途——它必须在**第一个需要 UHT 生成的类型使用之前**。

### 2. `UTenkaichiGameFeature_HotfixManager::OnGameFeatureLoading`（挂起）

```cpp
if (UTenkaichiHotfixManager* HotfixManager = Cast<UTenkaichiHotfixManager>(UOnlineHotfixManager::Get(nullptr)))
{
	HotfixManager->RequestPatchAssetsFromIniFiles();
}
```

逻辑：拿到当前的 OnlineHotfixManager，转成 `UTenkaichiHotfixManager`（成功说明项目用的是自定义 hotfix 管理器），然后让它**从 ini 文件请求补丁资产**（热更新）。

依赖 P18 的 `UTenkaichiHotfixManager`，所以挂起。

### 3. `TRACE_CPUPROFILER_EVENT_SCOPE`——性能标记

```cpp
TRACE_CPUPROFILER_EVENT_SCOPE(UTenkaichiGameFeature_AddGameplayCuePaths::OnGameFeatureRegistering);
```

在 Unreal Insights / CPU Profiler 里给这个函数起个名字，方便性能分析时定位。零运行时开销（非 profile 构建时展开为空）。

### 4. `FixPluginPackagePath`——路径修正

```cpp
const FString PluginRootPath = TEXT("/") + PluginName;
FString MutablePath = Directory.Path;
UGameFeaturesSubsystem::FixPluginPackagePath(MutablePath, PluginRootPath, false);
```

P15.2 的 `DirectoryPathsToAdd` 存的是**相对插件根目录**的路径（比如 `/GameplayCues`）。这里用插件名拼出根路径（`/插件名`），把相对路径修正成**完整包路径**（`/插件名/GameplayCues`）。

### 5. 注册时的"前后数量对比"

```cpp
const int32 PreInitializeNumCues = RuntimeGameplayCueSet ? RuntimeGameplayCueSet->GameplayCueData.Num() : 0;
... AddGameplayCueNotifyPath + InitializeRuntimeObjectLibrary ...
const int32 PostInitializeNumCues = RuntimeGameplayCueSet ? RuntimeGameplayCueSet->GameplayCueData.Num() : 0;
if (PreInitializeNumCues != PostInitializeNumCues)
{
	GCM->RefreshGameplayCuePrimaryAsset();
}
```

- 记下注册前 Cue 数量 → 加路径 + 重建运行时库 → 记下注册后数量。
- **数量变了**才刷新主资产（`RefreshGameplayCuePrimaryAsset`）——避免不必要的刷新开销。

### 6. `AddGameplayCueNotifyPath` 的 `false` 参数

```cpp
GCM->AddGameplayCueNotifyPath(MutablePath, /** bShouldRescanCueAssets = */ false);
```

第二个参数 `bShouldRescanCueAssets = false`：**不立即重新扫描** Cue 资产（因为后面统一调 `InitializeRuntimeObjectLibrary` 重建，避免重复扫描）。

> `/****/` 是"参数名注释"——把参数名写在注释里（因为调用处看不出参数含义）。按铁律 28 保留英文 `bShouldRescanCueAssets`（它是**参数名**，不是注释文案；保留原名有助于对照源码）。

### 7. 注销时用引擎基类

```cpp
if (UGameplayCueManager* GCM = UAbilitySystemGlobals::Get().GetGameplayCueManager())
```

注意对比：**注册用 `UTenkaichiGameplayCueManager::Get()`（自定义类），注销用 `UAbilitySystemGlobals::Get().GetGameplayCueManager()`（引擎基类）**。

为什么？因为 `RemoveGameplayCueNotifyPath` 是基类方法，用基类指针就够。这是 Lyra 源码的不对称写法，按铁律 23 照抄。

### 8. `ensure(NumRemoved == DirsToAdd.Num())`

```cpp
ensure(NumRemoved == DirsToAdd.Num());
```

断言：移除的路径数应该等于要移除的路径数。不等说明有路径没找到（数据不一致），报警告。
