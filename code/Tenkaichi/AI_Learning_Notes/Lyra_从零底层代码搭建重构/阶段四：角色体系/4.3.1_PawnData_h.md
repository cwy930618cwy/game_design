```cpp
// Copyright Epic Games, Inc. All Rights Reserved.

#pragma once

#include "Engine/DataAsset.h"

#include "TenkaichiPawnData.generated.h"

#define UE_API TENKAICHIGAME_API

class APawn;
class UTenkaichiAbilitySet;
class UTenkaichiAbilityTagRelationshipMapping;
class UTenkaichiCameraMode;
class UTenkaichiInputConfig;
class UObject;


/**
 * UTenkaichiPawnData
 *
 *	不可变数据资产，用来定义"一个 Pawn 需要哪些配置"。
 */
UCLASS(MinimalAPI, BlueprintType, Const, Meta = (DisplayName = "Tenkaichi Pawn Data", ShortTooltip = "Data asset used to define a Pawn."))
class UTenkaichiPawnData : public UPrimaryDataAsset
{
	GENERATED_BODY()

public:

	UE_API UTenkaichiPawnData(const FObjectInitializer& ObjectInitializer);

public:

	// 该 Pawn 实例化用的类（通常继承自 ATenkaichiPawn 或 ATenkaichiCharacter）
	UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Tenkaichi|Pawn")
	TSubclassOf<APawn> PawnClass;

	// 授予该 Pawn 能力系统的一组能力集
	UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Tenkaichi|Abilities")
	TArray<TObjectPtr<UTenkaichiAbilitySet>> AbilitySets;

	// 该 Pawn 发起动作时使用的"能力标签关系映射"
	UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Tenkaichi|Abilities")
	TObjectPtr<UTenkaichiAbilityTagRelationshipMapping> TagRelationshipMapping;

	// 玩家控制的 Pawn 用这个输入配置来创建输入映射并绑定输入动作
	UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Tenkaichi|Input")
	TObjectPtr<UTenkaichiInputConfig> InputConfig;

	// 玩家控制的 Pawn 默认使用的相机模式
	UPROPERTY(EditDefaultsOnly, BlueprintReadOnly, Category = "Tenkaichi|Camera")
	TSubclassOf<UTenkaichiCameraMode> DefaultCameraMode;
};

#undef UE_API
```

## 二、逐段讲解（先讲"为什么"再讲"是什么"）

### 1. 前向声明（软引用，不阻塞编译）

```cpp
class APawn;
class UTenkaichiAbilitySet;
class UTenkaichiAbilityTagRelationshipMapping;
class UTenkaichiCameraMode;
class UTenkaichiInputConfig;
class UObject;
```

- 这些类在 `.h` 里**只做前向声明**（`class 类名;`），不 `#include` 完整定义。
- 因为下面的字段用的是**指针容器**（`TSubclassOf`、`TObjectPtr`、`TArray<TObjectPtr>`），
  编译器只需要知道"有这么个类型"，不需要完整定义。
- **铁律 30 关键**：`UTenkaichiAbilitySet`、`UTenkaichiCameraMode` 尚未教，但因为是软引用，
  **编译 `.h` 不需要它们的完整定义**，所以本文件可独立编译。

### 2. 类声明

```cpp
UCLASS(MinimalAPI, BlueprintType, Const, Meta = (DisplayName = "Tenkaichi Pawn Data", ShortTooltip = "Data asset used to define a Pawn."))
class UTenkaichiPawnData : public UPrimaryDataAsset
```

- `MinimalAPI`：类本身导出，成员不强制导出。
- `BlueprintType`：可在蓝图里作为变量类型使用。
- `Const`：数据资产是只读的（不运行时修改）。
- `Meta=(DisplayName=..., ShortTooltip=...)`：编辑器里显示的名字和提示。
- 继承 `UPrimaryDataAsset`：**主数据资产**，可被 `AssetManager` 全局加载（区别于普通 `UDataAsset`）。

### 3. 构造函数声明

```cpp
UE_API UTenkaichiPawnData(const FObjectInitializer& ObjectInitializer);
```

- 带 `UE_API`（= `TENKAICHIGAME_API`）导出，保证其他模块能构造。
- 实现在 `.cpp`（下一步教），把 3 个指针初始化为 `nullptr`。

### 4. 五个配置字段（UPROPERTY）

| 字段 | 类型 | 干嘛的 |
|------|------|--------|
| `PawnClass` | `TSubclassOf<APawn>` | 该 Pawn 用哪个类（`TSubclassOf` = 指定一个"类"作为数据） |
| `AbilitySets` | `TArray<TObjectPtr<UTenkaichiAbilitySet>>` | 能力集数组（软引用，未教） |
| `TagRelationshipMapping` | `TObjectPtr<UTenkaichiAbilityTagRelationshipMapping>` | 能力标签关系映射（✅ 3.4.2 已教） |
| `InputConfig` | `TObjectPtr<UTenkaichiInputConfig>` | 输入配置（✅ 3.4.3 已教） |
| `DefaultCameraMode` | `TSubclassOf<UTenkaichiCameraMode>` | 默认相机模式（软引用，未教） |

- `EditDefaultsOnly`：只能在"类默认值"里编辑（数据资产里填默认值）。
- `BlueprintReadOnly`：蓝图可读，不可写。
- `Category`：编辑器分类分组。

### 5. `TSubclassOf<T>` vs `TObjectPtr<T>` 的区别（关键概念）

| 类型 | 存什么 | 例子 |
|------|--------|------|
| `TSubclassOf<APawn>` | 存"**一个类**"（不是实例） | `PawnClass` = 某角色的类 |
| `TObjectPtr<UInputConfig>` | 存"**一个对象实例**" | `InputConfig` = 某份输入配置资产 |

> 简单记：`TSubclassOf` 存"类"（要 new 出来的），`TObjectPtr` 存"对象"（已有资产）。
