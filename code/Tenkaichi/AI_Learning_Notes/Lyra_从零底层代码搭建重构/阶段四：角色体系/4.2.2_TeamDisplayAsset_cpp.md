// 把这份队伍的标量/颜色/贴图参数，全部刷到一个动态材质上
void UTenkaichiTeamDisplayAsset::ApplyToMaterial(UMaterialInstanceDynamic* Material)
{
	if (Material)
	{
		for (const auto& KVP : ScalarParameters)
		{
			Material->SetScalarParameterValue(KVP.Key, KVP.Value);
		}

		for (const auto& KVP : ColorParameters)
		{
			Material->SetVectorParameterValue(KVP.Key, FVector(KVP.Value));
		}

		for (const auto& KVP : TextureParameters)
		{
			Material->SetTextureParameterValue(KVP.Key, KVP.Value);
		}
	}
}

// 把这份队伍的标量/颜色/贴图参数，刷到网格组件的所有材质上（不足的动态材质会自动创建）
void UTenkaichiTeamDisplayAsset::ApplyToMeshComponent(UMeshComponent* MeshComponent)
{
	if (MeshComponent)
	{
		for (const auto& KVP : ScalarParameters)
		{
			MeshComponent->SetScalarParameterValueOnMaterials(KVP.Key, KVP.Value);
		}

		for (const auto& KVP : ColorParameters)
		{
			MeshComponent->SetVectorParameterValueOnMaterials(KVP.Key, FVector(KVP.Value));
		}

		const TArray<UMaterialInterface*> MaterialInterfaces = MeshComponent->GetMaterials();
		for (int32 MaterialIndex = 0; MaterialIndex < MaterialInterfaces.Num(); ++MaterialIndex)
		{
			if (UMaterialInterface* MaterialInterface = MaterialInterfaces[MaterialIndex])
			{
				UMaterialInstanceDynamic* DynamicMaterial = Cast<UMaterialInstanceDynamic>(MaterialInterface);
				if (!DynamicMaterial)
				{
					DynamicMaterial = MeshComponent->CreateAndSetMaterialInstanceDynamic(MaterialIndex);
				}

				for (const auto& KVP : TextureParameters)
				{
					DynamicMaterial->SetTextureParameterValue(KVP.Key, KVP.Value);
				}
			}
		}
	}
}

// 把这份队伍的标量/颜色/贴图参数，刷到一个 Niagara 粒子组件上
void UTenkaichiTeamDisplayAsset::ApplyToNiagaraComponent(UNiagaraComponent* NiagaraComponent)
{
	if (NiagaraComponent)
	{
		for (const auto& KVP : ScalarParameters)
		{
			NiagaraComponent->SetVariableFloat(KVP.Key, KVP.Value);
		}

		for (const auto& KVP : ColorParameters)
		{
			NiagaraComponent->SetVariableLinearColor(KVP.Key, KVP.Value);
		}

		for (const auto& KVP : TextureParameters)
		{
			UTexture* Texture = KVP.Value;
			NiagaraComponent->SetVariableTexture(KVP.Key, Texture);
		}
	}
}

// 遍历目标 Actor 的所有组件，把队伍外观刷到每个可用的组件上（可含子组件）
void UTenkaichiTeamDisplayAsset::ApplyToActor(AActor* TargetActor, bool bIncludeChildActors)
{
	if (TargetActor != nullptr)
	{
		TargetActor->ForEachComponent(bIncludeChildActors, [this](UActorComponent* InComponent)
		{
			if (UMeshComponent* MeshComponent = Cast<UMeshComponent>(InComponent))
			{
				ApplyToMeshComponent(MeshComponent);
			}
			else if (UNiagaraComponent* NiagaraComponent = Cast<UNiagaraComponent>(InComponent))
			{
				ApplyToNiagaraComponent(NiagaraComponent);
			}
		});
	}
}

## 二、逐段讲解（先讲"为什么"再讲"是什么"）

### 1. 头文件依赖

- `#include "Teams/TenkaichiTeamSubsystem.h"`：**这就是断点**——`.cpp` 用它，但它是 GAS 断点类。
- 其余（`MeshComponent`/`NiagaraComponent`/`MaterialInstanceDynamic`/`Texture`/`UObjectIterator`）全是引擎类。
- `UE_INLINE_GENERATED_CPP_BY_NAME`：UE5 自动生成的反射代码宏（和 `.generated.h` 配对）。

### 2. `ApplyToMaterial` —— 刷到动态材质

- 遍历三个参数表，分别调材质 API：
  - `SetScalarParameterValue`（标量）
  - `SetVectorParameterValue`（颜色，转成 `FVector`）
  - `SetTextureParameterValue`（贴图）
- 用 `const auto& KVP` 遍历 `TMap`：`KVP.Key` 是参数名，`KVP.Value` 是值。

### 3. `ApplyToMeshComponent` —— 刷到网格组件（比材质多一步）

- 标量/颜色直接调 `SetXxxParameterValueOnMaterials`（一次刷到网格所有材质）。
- 贴图**不能**直接刷，要遍历网格的每个材质槽：
  - 先取 `GetMaterials()`；
  - 如果该槽不是动态材质（`Cast<UMaterialInstanceDynamic>` 失败），就 `CreateAndSetMaterialInstanceDynamic` 创建一个动态实例（否则不能改参数）；
  - 再刷贴图参数。

> 核心：**静态材质不能改运行时参数，必须先转成动态材质（MID）**。这是材质参数化的关键。

### 4. `ApplyToNiagaraComponent` —— 刷到粒子

- 粒子 API 和材质不同：
  - `SetVariableFloat`（标量）
  - `SetVariableLinearColor`（颜色，直接用 `FLinearColor`）
  - `SetVariableTexture`（贴图，取 `UTexture*`）
- 注意粒子用 `FLinearColor` 原样，材质用 `FVector`（API 签名不同）。

### 5. `ApplyToActor` —— 遍历组件分发

- `TargetActor->ForEachComponent(bIncludeChildActors, Lambda)`：遍历 Actor 所有组件（含/不含子组件）。
- 在 Lambda 里按类型分发：`Cast<UMeshComponent>` → 走网格；`Cast<UNiagaraComponent>` → 走粒子。
- 其他组件类型忽略。

### 6. `PostEditChangeProperty` —— 编辑器专属 + 断点（铁律 35）

- `#if WITH_EDITOR`：只在编辑器构建编译。
- 逻辑：改了属性 → `Super::` → 遍历所有 `TeamSubsystem` 实例，通知"这份队伍外观改了"（即时刷新预览）。
- **断点**：`UTenkaichiTeamSubsystem` 是 GAS 断点类，尚未就位。所以这段**完整写上但注释挂起**（`/* ... */`），
  并登记 `注释备忘录.md`。到阶段五 `TeamSubsystem` 就位后放开注释即可。

> ⚠️ **注意**：`PostEditChangeProperty` 的 `.h` 声明保留（声明不依赖 TeamSubsystem），
> 只有 `.cpp` 函数体里依赖 `TeamSubsystem` 的部分注释挂起。这样 `.h` + `.cpp` 其它部分都能编译。

## 三、依赖核对（铁律 30/31）

- `.cpp` 依赖：引擎类（`MeshComponent`/`NiagaraComponent`/`MaterialInstanceDynamic`/`Texture`）+ **`TenkaichiTeamSubsystem`（断点 ✗）**。
- 断点处理：`PostEditChangeProperty` 内依赖 `TeamSubsystem` 的代码**注释挂起**（`/* */`），
  **其余代码（4 个 `ApplyToXxx`）完整编译，无未教依赖**。

**结论：注释挂起断点段后，`.cpp` 可编译；`ApplyToXxx` 四个方法是纯引擎实现，无 GAS 依赖。**
