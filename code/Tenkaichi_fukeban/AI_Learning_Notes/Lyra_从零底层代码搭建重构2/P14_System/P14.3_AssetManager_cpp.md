// 同步加载指定路径的资产。
UObject* UTenkaichiAssetManager::SynchronousLoadAsset(const FSoftObjectPath& AssetPath)
{
	if (AssetPath.IsValid())
	{
		TUniquePtr<FScopeLogTime> LogTimePtr;

		if (ShouldLogAssetLoads())
		{
			LogTimePtr = MakeUnique<FScopeLogTime>(*FString::Printf(TEXT("Synchronously loaded asset [%s]"), *AssetPath.ToString()), nullptr, FScopeLogTime::ScopeLog_Seconds);
		}

		if (UAssetManager::IsInitialized())
		{
			return UAssetManager::GetStreamableManager().LoadSynchronous(AssetPath, false);
		}

		// 资产管理器还没就绪，就用 LoadObject。
		return AssetPath.TryLoad();
	}

	return nullptr;
}

// 真正加载指定类的游戏数据。
UPrimaryDataAsset* UTenkaichiAssetManager::LoadGameDataOfClass(TSubclassOf<UPrimaryDataAsset> DataClass, const TSoftObjectPtr<UPrimaryDataAsset>& DataClassPath, FPrimaryAssetType PrimaryAssetType)
{
	UPrimaryDataAsset* Asset = nullptr;

	DECLARE_SCOPE_CYCLE_COUNTER(TEXT("Loading GameData Object"), STAT_GameData, STATGROUP_LoadTime);
	if (!DataClassPath.IsNull())
	{
#if WITH_EDITOR
		FScopedSlowTask SlowTask(0, FText::Format(NSLOCTEXT("TenkaichiEditor", "BeginLoadingGameDataTask", "Loading GameData {0}"), FText::FromName(DataClass->GetFName())));
		const bool bShowCancelButton = false;
		const bool bAllowInPIE = true;
		SlowTask.MakeDialog(bShowCancelButton, bAllowInPIE);
#endif
		UE_LOG(LogTenkaichi, Log, TEXT("Loading GameData: %s ..."), *DataClassPath.ToString());
		SCOPE_LOG_TIME_IN_SECONDS(TEXT("    ... GameData loaded!"), nullptr);

		// 这在编辑器里可能被递归调用（因为它是在 PostLoad 里按需调用的），
		// 所以那种情况下强制同步加载主资产、异步加载其余部分
		if (GIsEditor)
		{
			Asset = DataClassPath.LoadSynchronous();
			LoadPrimaryAssetsWithType(PrimaryAssetType);
		}
		else
		{
			TSharedPtr<FStreamableHandle> Handle = LoadPrimaryAssetsWithType(PrimaryAssetType);
			if (Handle.IsValid())
			{
				Handle->WaitUntilComplete(0.0f, false);

				// 这应该总是能成功
				Asset = Cast<UPrimaryDataAsset>(Handle->GetLoadedAsset());
			}
		}
	}

	if (Asset)
	{
		GameDataMap.Add(DataClass, Asset);
	}
	else
	{
		// 加载任何 GameData 资产失败都是不可接受的，会导致难以诊断的软失败。
		UE_LOG(LogTenkaichi, Fatal, TEXT("Failed to load GameData asset at %s. Type %s. This is not recoverable and likely means you do not have the correct data to run %s."), *DataClassPath.ToString(), *PrimaryAssetType.ToString(), FApp::GetProjectName());
	}

	return Asset;
}


// 跑所有启动任务。
void UTenkaichiAssetManager::DoAllStartupJobs()
{
	SCOPED_BOOT_TIMING("UTenkaichiAssetManager::DoAllStartupJobs");
	const double AllStartupJobsStartTime = FPlatformTime::Seconds();

	if (IsRunningDedicatedServer())
	{
		// 专用服务器不需要周期性进度更新，直接跑任务
		for (const FTenkaichiAssetManagerStartupJob& StartupJob : StartupJobs)
		{
			StartupJob.DoJob();
		}
	}
	else
	{
		if (StartupJobs.Num() > 0)
		{
			float TotalJobValue = 0.0f;
			for (const FTenkaichiAssetManagerStartupJob& StartupJob : StartupJobs)
			{
				TotalJobValue += StartupJob.JobWeight;
			}

			float AccumulatedJobValue = 0.0f;
			for (FTenkaichiAssetManagerStartupJob& StartupJob : StartupJobs)
			{
				const float JobValue = StartupJob.JobWeight;
				StartupJob.SubstepProgressDelegate.BindLambda([This = this, AccumulatedJobValue, JobValue, TotalJobValue](float NewProgress)
					{
						const float SubstepAdjustment = FMath::Clamp(NewProgress, 0.0f, 1.0f) * JobValue;
						const float OverallPercentWithSubstep = (AccumulatedJobValue + SubstepAdjustment) / TotalJobValue;

						This->UpdateInitialGameContentLoadPercent(OverallPercentWithSubstep);
					});

				StartupJob.DoJob();

				StartupJob.SubstepProgressDelegate.Unbind();

				AccumulatedJobValue += JobValue;

				UpdateInitialGameContentLoadPercent(AccumulatedJobValue / TotalJobValue);
			}
		}
		else
		{
			UpdateInitialGameContentLoadPercent(1.0f);
		}
	}

	StartupJobs.Empty();

	UE_LOG(LogTenkaichi, Display, TEXT("All startup jobs took %.2f seconds to complete"), FPlatformTime::Seconds() - AllStartupJobsStartTime);
}

## 二、关键点讲解

### 1. 单例 `Get()`——从引擎拿

```cpp
if (UTenkaichiAssetManager* Singleton = Cast<UTenkaichiAssetManager>(GEngine->AssetManager))
{
	return *Singleton;
}
UE_LOG(LogTenkaichi, Fatal, ...);  // 没配好就 Fatal 崩溃
```

`GEngine->AssetManager` 是引擎全局的资产管理器指针。配置里 `AssetManagerClassName` 设成 `TenkaichiAssetManager` 后，它就是这个类型。`Cast` 成功就返回，失败说明没配好，`Fatal` 崩溃（带明确错误信息）。

### 2. `STARTUP_JOB` / `STARTUP_JOB_WEIGHTED` 宏

```cpp
#define STARTUP_JOB_WEIGHTED(JobFunc, JobWeight) StartupJobs.Add(FTenkaichiAssetManagerStartupJob(#JobFunc, [this](...){JobFunc;}, JobWeight))
#define STARTUP_JOB(JobFunc) STARTUP_JOB_WEIGHTED(JobFunc, 1.f)
```

- `#JobFunc`：字符串化（`#` 把宏参数变成字符串），作为任务名。
- `[this](...){JobFunc;}`：lambda 捕获 this，调用 `JobFunc`。
- `JobWeight`：权重（`GetGameData` 权重 25，表示它占启动进度的 25/26 ≈ 96%）。

用法：`STARTUP_JOB(InitializeGameplayCueManager());` 展开成"添加一个启动任务，执行 `InitializeGameplayCueManager()`"。

### 3. `SynchronousLoadAsset`——同步加载

```cpp
if (UAssetManager::IsInitialized())
{
	return UAssetManager::GetStreamableManager().LoadSynchronous(AssetPath, false);
}
return AssetPath.TryLoad();  // AssetManager 没就绪，用 LoadObject
```

- AssetManager 就绪 → `LoadSynchronous`（流式管理器同步加载）。
- 没就绪 → `TryLoad`（直接 LoadObject）。

### 4. `FScopeLock`——线程安全

```cpp
FScopeLock LoadedAssetsLock(&LoadedAssetsCritical);
LoadedAssets.Add(Asset);
```

`FScopeLock` 是 RAII 锁：构造时加锁，析构时解锁。保护 `LoadedAssets` 这个 `TSet`（可能被多线程访问）。

### 5. `DoAllStartupJobs`——跑启动任务的两种模式

- **专用服务器**（`IsRunningDedicatedServer()`）：直接 `DoJob()`，不上报进度（服务器没加载屏）。
- **普通客户端**：算总权重，每个任务绑一个 lambda 上报进度（`UpdateInitialGameContentLoadPercent`），跑完解绑。

关键：`SubstepProgressDelegate.BindLambda([This = this, AccumulatedJobValue, JobValue, TotalJobValue](float NewProgress){...})` 用 lambda 捕获 `AccumulatedJobValue`（值捕获，快照）+ `JobValue` + `TotalJobValue`，算出全局进度百分比。

### 6. `LoadGameDataOfClass`——加载游戏数据（编辑器/运行时两套）

```cpp
if (GIsEditor)
{
	Asset = DataClassPath.LoadSynchronous();       // 编辑器：直接同步加载
	LoadPrimaryAssetsWithType(PrimaryAssetType);   // 再加载同类型所有主资产
}
else
{
	TSharedPtr<FStreamableHandle> Handle = LoadPrimaryAssetsWithType(PrimaryAssetType);
	Handle->WaitUntilComplete(0.0f, false);        // 运行时：异步加载 + 阻塞等待
	Asset = Cast<UPrimaryDataAsset>(Handle->GetLoadedAsset());
}
```

编辑器里可能递归调用（PostLoad 按需），所以走同步；运行时走流式 + 阻塞等待。

### 7. `FScopedSlowTask` / `SCOPE_LOG_TIME_IN_SECONDS` / `SCOPED_BOOT_TIMING` / `DECLARE_SCOPE_CYCLE_COUNTER`

这几个都是**性能统计/编辑器进度**工具宏：

| 宏 | 作用 |
|----|------|
| `SCOPED_BOOT_TIMING` | 统计启动阶段耗时 |
| `DECLARE_SCOPE_CYCLE_COUNTER` | 声明一个周期计数器（Profiler 里看） |
| `SCOPE_LOG_TIME_IN_SECONDS` | 作用域结束打印耗时 |
| `FScopedSlowTask` | 编辑器里显示进度对话框 |
