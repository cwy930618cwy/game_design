// 版权声明（可保留或省略，不影响编译）
#include "TenkaichiUserFacingExperienceDefinition.h"
#include "CommonSessionSubsystem.h"
#include "Containers/UnrealString.h"
#include "UObject/NameTypes.h"
#include "Engine/GameInstance.h"
#include "Engine/Engine.h"
//#include "Replays/TenkaichiReplaySubsystem.h"   // ⚠️ 注释挂起：依赖 P17 ReplaySubsystem

#include UE_INLINE_GENERATED_CPP_BY_NAME(TenkaichiUserFacingExperienceDefinition)

// 创建一个请求对象，用于按这些设置真正开启一场会话。
UCommonSession_HostSessionRequest* UTenkaichiUserFacingExperienceDefinition::CreateHostingRequest(const UObject* WorldContextObject) const
{
	const FString ExperienceName = ExperienceID.PrimaryAssetName.ToString();
	const FString UserFacingExperienceName = GetPrimaryAssetId().PrimaryAssetName.ToString();

	UWorld* World = GEngine->GetWorldFromContextObject(WorldContextObject, EGetWorldErrorMode::ReturnNull);
	UGameInstance* GameInstance = World ? World->GetGameInstance() : nullptr;
	UCommonSession_HostSessionRequest* Result = nullptr;

	if (UCommonSessionSubsystem* Subsystem = GameInstance ? GameInstance->GetSubsystem<UCommonSessionSubsystem>() : nullptr)
	{
		Result = Subsystem->CreateOnlineHostSessionRequest();
	}

	if (!Result)
	{
		// 用不了子系统，就自己新建一个
		Result = NewObject<UCommonSession_HostSessionRequest>();
		Result->OnlineMode = ECommonSessionOnlineMode::Online;
		Result->bUseLobbies = true;
		Result->bUseLobbiesVoiceChat = false;
		// 我们总是为这个会话启用 presence，因为它是用于匹配的主会话。
		// 对于在意 presence 的在线系统，只有主会话才应该启用 presence。
		Result->bUsePresence = !IsRunningDedicatedServer();
	}
	Result->MapID = MapID;
	Result->ModeNameForAdvertisement = UserFacingExperienceName;
	Result->ExtraArgs = ExtraArgs;
	Result->ExtraArgs.Add(TEXT("Experience"), ExperienceName);
	Result->MaxPlayerCount = MaxPlayerCount;

#if 0
	// ⚠️ 注释挂起：依赖 P17 UTenkaichiReplaySubsystem（DoesPlatformSupportReplays）
	if (UTenkaichiReplaySubsystem::DoesPlatformSupportReplays())
	{
		if (bRecordReplay)
		{
			Result->ExtraArgs.Add(TEXT("DemoRec"), FString());
		}
	}
#endif

	return Result;
}

## 二、关键点讲解

### 1. `ExperienceName` 和 `UserFacingExperienceName` 的区别

```cpp
const FString ExperienceName = ExperienceID.PrimaryAssetName.ToString();
const FString UserFacingExperienceName = GetPrimaryAssetId().PrimaryAssetName.ToString();
```

- `ExperienceName`：`ExperienceID`（指向玩法经验的 `FPrimaryAssetId`）的资产名，作为 URL 参数 `Experience=xxx` 传给游戏。
- `UserFacingExperienceName`：**这个卡片定义自己**的资产名（`GetPrimaryAssetId()`），用于大厅里展示的"模式名"。

### 2. `GetWorldFromContextObject` + 三元判断

```cpp
UWorld* World = GEngine->GetWorldFromContextObject(WorldContextObject, EGetWorldErrorMode::ReturnNull);
UGameInstance* GameInstance = World ? World->GetGameInstance() : nullptr;
```

- `EGetWorldErrorMode::ReturnNull`：拿不到 World 就返回 `nullptr`（不报错）。对比 P13.6 用的是 `LogAndReturnNull`（打印日志 + 返回 null）。
- 三元 `World ? ... : nullptr`：World 可能为空，空就 GameInstance 也为空。

### 3. 两级创建 `HostSessionRequest`

```cpp
if (UCommonSessionSubsystem* Subsystem = GameInstance ? GameInstance->GetSubsystem<UCommonSessionSubsystem>() : nullptr)
{
	Result = Subsystem->CreateOnlineHostSessionRequest();
}

if (!Result)
{
	Result = NewObject<UCommonSession_HostSessionRequest>();
	...
}
```

- **优先**：从 `UCommonSessionSubsystem` 拿（子系统创建的请求带默认配置）。
- **回退**：子系统拿不到，就 `NewObject` 自己 new 一个，手动填默认值。

### 4. `IsRunningDedicatedServer()`

```cpp
Result->bUsePresence = !IsRunningDedicatedServer();
```

`IsRunningDedicatedServer()` 是引擎全局函数（在 `Engine/Engine.h` 里），判断当前是不是专用服务器。专用服务器不需要 presence（在线状态），所以取反。

### 5. 组装请求字段

```cpp
Result->MapID = MapID;                                    // 要加载的地图
Result->ModeNameForAdvertisement = UserFacingExperienceName;  // 大厅展示的模式名
Result->ExtraArgs = ExtraArgs;                            // 复制额外参数
Result->ExtraArgs.Add(TEXT("Experience"), ExperienceName);    // 追加"经验名"URL 参数
Result->MaxPlayerCount = MaxPlayerCount;                  // 最大玩家数
```

关键：`Result->ExtraArgs.Add(TEXT("Experience"), ExperienceName)` 把"这局用哪个玩法经验"塞进 URL 参数——这正是 GameMode 里 `HandleMatchAssignmentIfNotExpectingOne` 用 `ParseOption(OptionsString, TEXT("Experience"))` 读的那个值（P13.7 讲的优先级第 2 级"URL 选项"）。

### 6. 断点挂起（1 处）

```cpp
#if 0
	if (UTenkaichiReplaySubsystem::DoesPlatformSupportReplays())
	{
		if (bRecordReplay)
		{
			Result->ExtraArgs.Add(TEXT("DemoRec"), FString());
		}
	}
#endif
```

`UTenkaichiReplaySubsystem` 在 P17（Replays 文件夹）才教。这段逻辑是"如果平台支持回放、且卡片勾了 `bRecordReplay`，就加一个 `DemoRec` URL 参数"。依赖未就位，`#if 0` 挂起。
