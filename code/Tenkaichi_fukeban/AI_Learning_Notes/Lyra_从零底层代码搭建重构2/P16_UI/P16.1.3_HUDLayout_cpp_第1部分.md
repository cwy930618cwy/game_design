```cpp
#if WITH_EDITOR

#include UE_INLINE_GENERATED_CPP_BY_NAME(TenkaichiHUDLayout)

// 定义 3 个本地静态 GameplayTag。
UE_DEFINE_GAMEPLAY_TAG_STATIC(TAG_UI_LAYER_MENU, "UI.Layer.Menu");
UE_DEFINE_GAMEPLAY_TAG_STATIC(TAG_UI_ACTION_ESCAPE, "UI.Action.Escape");
UE_DEFINE_GAMEPLAY_TAG_STATIC(TAG_Platform_Trait_Input_PrimarlyController, "Platform.Trait.Input.PrimarlyController");

// 构造函数：加默认的平台 trait tag。
UTenkaichiHUDLayout::UTenkaichiHUDLayout(const FObjectInitializer& ObjectInitializer)
	: Super(ObjectInitializer)
	, SpawnedControllerDisconnectScreen(nullptr)
{
	// 默认只有"主要以手柄输入"的平台才需要断线屏。
	PlatformRequiresControllerDisconnectScreen.AddTag(TAG_Platform_Trait_Input_PrimarlyController);
}

// 控件初始化（注册 Esc 动作 + 绑手柄事件）。
void UTenkaichiHUDLayout::NativeOnInitialized()
{
	Super::NativeOnInitialized();

	// 注册 Esc 输入动作 → HandleEscapeAction
	RegisterUIActionBinding(FBindUIActionArgs(FUIActionTag::ConvertChecked(TAG_UI_ACTION_ESCAPE), false, FSimpleDelegate::CreateUObject(this, &ThisClass::HandleEscapeAction)));

	// 如果这个平台能显示控制器断线屏，就监听控制器状态变化委托
	if (ShouldPlatformDisplayControllerDisconnectScreen())
	{
		// 绑到输入设备连接变化
		IPlatformInputDeviceMapper& DeviceMapper = IPlatformInputDeviceMapper::Get();
		DeviceMapper.GetOnInputDeviceConnectionChange().AddUObject(this, &ThisClass::HandleInputDeviceConnectionChanged);
		DeviceMapper.GetOnInputDevicePairingChange().AddUObject(this, &ThisClass::HandleInputDevicePairingChanged);	
	}
}

// 控件销毁（解绑手柄事件 + 移除 Ticker）。
void UTenkaichiHUDLayout::NativeDestruct()
{
	Super::NativeDestruct();

	// 移除输入设备连接变化的绑定
	IPlatformInputDeviceMapper& DeviceMapper = IPlatformInputDeviceMapper::Get();
	DeviceMapper.GetOnInputDeviceConnectionChange().RemoveAll(this);
	DeviceMapper.GetOnInputDevicePairingChange().RemoveAll(this);

	// 如果还有排队的 Ticker，移除它
	if (RequestProcessControllerStateHandle.IsValid())
	{
		FTSTicker::GetCoreTicker().RemoveTicker(RequestProcessControllerStateHandle);
		RequestProcessControllerStateHandle.Reset();
	}
}

// 处理 Esc 动作：推暂停菜单。
void UTenkaichiHUDLayout::HandleEscapeAction()
{
	// 确保配置了暂停菜单类
	if (ensure(!EscapeMenuClass.IsNull()))
	{
		UCommonUIExtensions::PushStreamedContentToLayer_ForPlayer(GetOwningLocalPlayer(), TAG_UI_LAYER_MENU, EscapeMenuClass);
	}
}
```

## 二、关键点讲解

### 1. `UE_DEFINE_GAMEPLAY_TAG_STATIC`——本地静态 tag

```cpp
UE_DEFINE_GAMEPLAY_TAG_STATIC(TAG_UI_LAYER_MENU, "UI.Layer.Menu");
UE_DEFINE_GAMEPLAY_TAG_STATIC(TAG_UI_ACTION_ESCAPE, "UI.Action.Escape");
UE_DEFINE_GAMEPLAY_TAG_STATIC(TAG_Platform_Trait_Input_PrimarlyController, "Platform.Trait.Input.PrimarlyController");
```

`UE_DEFINE_GAMEPLAY_TAG_STATIC(变量名, "Tag.Name")`：在**当前 `.cpp` 文件内**定义一个静态 GameplayTag 变量。

对比 P0 的 `TenkaichiGameplayTags`（全局 tag，放 GameplayTags 文件里，跨文件可用）——这两个是**本地 tag**，只在这个文件用，所以就地定义。

> 注意 tag 名字里没有 `Lyra`（都是 `UI.*` / `Platform.*`），所以铁律 24 无事可做。

> ⚠️ 注意 `PrimarlyController` 是 Lyra 源码的**拼写错误**（应该是 `Primarily`）——按铁律 23 照抄，**不修正**。

### 2. `RegisterUIActionBinding(FBindUIActionArgs(...))`——注册 UI 输入动作

```cpp
RegisterUIActionBinding(FBindUIActionArgs(FUIActionTag::ConvertChecked(TAG_UI_ACTION_ESCAPE), false, FSimpleDelegate::CreateUObject(this, &ThisClass::HandleEscapeAction)));
```

`RegisterUIActionBinding`：CommonUI 的方法，把"某个 UI 动作 tag"绑到"某个回调"。

`FBindUIActionArgs(动作tag, bConsumeInput, 委托)`：
- `FUIActionTag::ConvertChecked(TAG_UI_ACTION_ESCAPE)`：把 GameplayTag 转成 `FUIActionTag`（带检查）。
- `false`：`bConsumeInput`（是否吃掉输入，false = 不阻止继续传给游戏）。
- `FSimpleDelegate::CreateUObject(this, &HandleEscapeAction)`：无参回调。

### 3. `IPlatformInputDeviceMapper::Get()`——平台输入设备映射器

```cpp
IPlatformInputDeviceMapper& DeviceMapper = IPlatformInputDeviceMapper::Get();
DeviceMapper.GetOnInputDeviceConnectionChange().AddUObject(this, &ThisClass::HandleInputDeviceConnectionChanged);
DeviceMapper.GetOnInputDevicePairingChange().AddUObject(this, &ThisClass::HandleInputDevicePairingChanged);
```

`IPlatformInputDeviceMapper`：引擎的"输入设备映射器"（管理哪个手柄属于哪个玩家）。两个全局事件：
- `GetOnInputDeviceConnectionChange()`：设备连接/断开时。
- `GetOnInputDevicePairingChange()`：设备改变所属玩家时。

### 4. `NativeDestruct` 的清理三件套

```cpp
DeviceMapper.GetOnInputDeviceConnectionChange().RemoveAll(this);
DeviceMapper.GetOnInputDevicePairingChange().RemoveAll(this);

if (RequestProcessControllerStateHandle.IsValid())
{
	FTSTicker::GetCoreTicker().RemoveTicker(RequestProcessControllerStateHandle);
	RequestProcessControllerStateHandle.Reset();
}
```

1. 解绑两个设备事件（`RemoveAll(this)`）。
2. 如果还有排队的 Ticker，移除它（`IsValid()` 判断是因为可能没加过）。
3. `Reset()` 句柄（清空）。

> **关键**：不清干净会导致控件销毁后还被回调 → 崩溃。UI 控件生命周期比 Actor 短，更容易踩这个坑。

### 5. `PushStreamedContentToLayer_ForPlayer`——异步推层

```cpp
UCommonUIExtensions::PushStreamedContentToLayer_ForPlayer(GetOwningLocalPlayer(), TAG_UI_LAYER_MENU, EscapeMenuClass);
```

`Push**Streamed**ContentToLayer_ForPlayer`：CommonUI 的"**流式**推层"（异步加载类再推）。

对比 P15.7 用的 `PushContentToLayer_ForPlayer`（同步，类已加载）——这里 `EscapeMenuClass` 是 `TSoftClassPtr`（软引用），所以要流式加载。

> 这正是 `.h` 里"Esc 菜单用软引用"的原因——**大菜单延迟加载**。

### 6. `ensure(!EscapeMenuClass.IsNull())`

```cpp
if (ensure(!EscapeMenuClass.IsNull()))
```

`ensure(条件)`：条件为假时**打警告并继续**。这里"没配置暂停菜单类"是可容忍的配置错误（不崩，只是 Esc 没反应），所以用 `ensure` 而非 `check`。
