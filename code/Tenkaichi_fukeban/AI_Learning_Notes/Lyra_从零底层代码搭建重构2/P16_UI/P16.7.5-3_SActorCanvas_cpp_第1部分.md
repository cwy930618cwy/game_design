```cpp
// 内部箭头控件：画一个可旋转的箭头图片。
class SActorCanvasArrowWidget : public SLeafWidget
{
public:

	SLATE_BEGIN_ARGS(SActorCanvasArrowWidget)
	{}
	/** 这个永远放最后 */
	SLATE_END_ARGS()

	/** 构造函数 */
	SActorCanvasArrowWidget()
	: Rotation(0.0f)
	, Arrow(nullptr)
	{

	}

	/** 每个控件都需要一个这个 */
	void Construct(const FArguments& InArgs, const FSlateBrush* ActorCanvasArrowBrush)
	{
		Arrow = ActorCanvasArrowBrush;
		SetCanTick(false);
	}

	// 绘制：画旋转后的箭头。
	virtual int32 OnPaint(const FPaintArgs& Args,
		const FGeometry& AllottedGeometry,
		const FSlateRect& MyClippingRect,
		FSlateWindowElementList& OutDrawElements,
		int32 LayerId,
		const FWidgetStyle& InWidgetStyle,
		bool bParentEnabled) const override
	{
		int32 MaxLayerId = LayerId;

		if (Arrow)
		{
			const bool bIsEnabled = ShouldBeEnabled(bParentEnabled);
			const ESlateDrawEffect DrawEffects = bIsEnabled ? ESlateDrawEffect::None : ESlateDrawEffect::DisabledEffect;
			const FColor FinalColorAndOpacity = (InWidgetStyle.GetColorAndOpacityTint() * Arrow->GetTint(InWidgetStyle)).ToFColor(true);

			// 画旋转盒（箭头）
			FSlateDrawElement::MakeRotatedBox(
				OutDrawElements,
				MaxLayerId++,
				AllottedGeometry.ToPaintGeometry(Arrow->ImageSize, FSlateLayoutTransform()),
				Arrow,
				DrawEffects,
				FMath::DegreesToRadians(GetRotation()),
				TOptional<FVector2D>(),
				FSlateDrawElement::RelativeToElement,
				FinalColorAndOpacity
			);
		}

		return MaxLayerId;
	}

	// 设置旋转角度（取模 360）。
	FORCEINLINE void SetRotation(float InRotation)
	{
		Rotation = FMath::Fmod(InRotation, 360.0f);
	}

	// 取旋转角度。
	FORCEINLINE float GetRotation() const
	{
		return Rotation;
	}

	// 期望尺寸：箭头图片大小。
	virtual FVector2D ComputeDesiredSize(float) const override
	{
		if (Arrow)
		{
			return Arrow->ImageSize;
		}
		else
		{
			return FVector2D::ZeroVector;
		}
	}

private:
	// 当前旋转角度。
	float Rotation;
	
	// 箭头画刷。
	const FSlateBrush* Arrow;
};
```

```cpp
// 为一条指示器创建控件 + 槽位（异步加载控件类）。
void SActorCanvas::AddIndicatorForEntry(UIndicatorDescriptor* Indicator)
{
	// 异步加载指示器控件，并池化结果，方便使用和复用控件。
	TSoftClassPtr<UUserWidget> IndicatorClass = Indicator->GetIndicatorClass();
	if (!IndicatorClass.IsNull())
	{
		TWeakObjectPtr<UIndicatorDescriptor> IndicatorPtr(Indicator);
		AsyncLoad(IndicatorClass, [this, IndicatorPtr, IndicatorClass]() {
			if (UIndicatorDescriptor* Indicator = IndicatorPtr.Get())
			{
				// 异步加载期间可能已经被移除了。
				if (!AllIndicators.Contains(Indicator))
				{
					return;
				}

				// 从池里创建（或复用）控件。
				if (UUserWidget* IndicatorWidget = IndicatorPool.GetOrCreateInstance(TSubclassOf<UUserWidget>(IndicatorClass.Get())))
				{
					if (IndicatorWidget->GetClass()->ImplementsInterface(UIndicatorWidgetInterface::StaticClass()))
					{
						IIndicatorWidgetInterface::Execute_BindIndicator(IndicatorWidget, Indicator);
					}

					Indicator->IndicatorWidget = IndicatorWidget;

					InactiveIndicators.Remove(Indicator);

					AddActorSlot(Indicator)
					[
						SAssignNew(Indicator->CanvasHost, SBox)
						[
							IndicatorWidget->TakeWidget()
						]
					];
				}
			}
		});
		StartAsyncLoading();
	}
}
```

```cpp
// 移除一条指示器的控件 + 槽位。
void SActorCanvas::RemoveIndicatorForEntry(UIndicatorDescriptor* Indicator)
{
	if (UUserWidget* IndicatorWidget = Indicator->IndicatorWidget.Get())
	{
		if (IndicatorWidget->GetClass()->ImplementsInterface(UIndicatorWidgetInterface::StaticClass()))
		{
			IIndicatorWidgetInterface::Execute_UnbindIndicator(IndicatorWidget, Indicator);
		}

		Indicator->IndicatorWidget = nullptr;
		
		IndicatorPool.Release(IndicatorWidget);
	}

	TSharedPtr<SWidget> CanvasHost = Indicator->CanvasHost.Pin();
	if (CanvasHost.IsValid())
	{
		RemoveActorSlot(CanvasHost.ToSharedRef());
		Indicator->CanvasHost.Reset();
	}
}
```

### 1.8 `AddActorSlot` / `RemoveActorSlot`（第 626~654 行）

```cpp
// 添加一个指示器槽位。
SActorCanvas::FScopedWidgetSlotArguments SActorCanvas::AddActorSlot(UIndicatorDescriptor* Indicator)
{
	TWeakPtr<SActorCanvas> WeakCanvas = SharedThis(this);
	return FScopedWidgetSlotArguments{ MakeUnique<FSlot>(Indicator), this->CanvasChildren, INDEX_NONE
		, [WeakCanvas](const FSlot*, int32)
		{
			if (TSharedPtr<SActorCanvas> Canvas = WeakCanvas.Pin())
			{
				Canvas->UpdateActiveTimer();
			}
		}};
}

// 移除一个槽位。
int32 SActorCanvas::RemoveActorSlot(const TSharedRef<SWidget>& SlotWidget)
{
	for (int32 SlotIdx = 0; SlotIdx < CanvasChildren.Num(); ++SlotIdx)
	{
		if ( SlotWidget == CanvasChildren[SlotIdx].GetWidget() )
		{
			CanvasChildren.RemoveAt(SlotIdx);

			UpdateActiveTimer();

			return SlotIdx;
		}
	}

	return -1;
}
```

## 二、关键点讲解

### 1. `namespace EArrowDirection { enum Type {...} }`——UE 旧式枚举

```cpp
namespace EArrowDirection
{
	enum Type
	{
		Left, Top, Right, Bottom, MAX
	};
}
```

这是 UE 的**旧式枚举写法**（不是 `enum class`）：
- 用 `namespace` 包裹，避免全局污染。
- 枚举名是 `Type`，用 `EArrowDirection::Left` 访问。
- `MAX` 是"数量哨兵"（用于数组大小）。

> 对比 `enum class`（P13.5 起用的）——旧式枚举能隐式转 int（所以能当数组下标）。

### 2. `ArrowRotations` / `ArrowOffsets` 平行数组

```cpp
const float ArrowRotations[EArrowDirection::MAX] = { 270.0f, 0.0f, 90.0f, 180.0f };
const FVector2D ArrowOffsets[EArrowDirection::MAX] = { (-1,0), (0,-1), (1,0), (0,1) };
```

两个数组**按 `EArrowDirection` 顺序**索引：
| 方向 | 旋转 | 偏移 |
|------|------|------|
| `Left` | 270° | (-1, 0) |
| `Top` | 0° | (0, -1) |
| `Right` | 90° | (1, 0) |
| `Bottom` | 180° | (0, 1) |

> 偏移是"从屏幕边缘往内指"的方向向量。

### 3. `SActorCanvasArrowWidget`——文件内私有 Slate 控件

```cpp
class SActorCanvasArrowWidget : public SLeafWidget
```

**注意**：这个类**没有 `S` 前缀导出**（不在 `.h` 里声明），是 `.cpp` 内部的**私有实现类**。

它只做一件事：画一个**可旋转**的箭头图片（`FSlateDrawElement::MakeRotatedBox`）。

### 4. `MakeRotatedBox`——画旋转盒

```cpp
FSlateDrawElement::MakeRotatedBox(
	OutDrawElements,
	MaxLayerId++,
	AllottedGeometry.ToPaintGeometry(Arrow->ImageSize, FSlateLayoutTransform()),
	Arrow,
	DrawEffects,
	FMath::DegreesToRadians(GetRotation()),      // 旋转弧度
	TOptional<FVector2D>(),                       // 轴心（空 = 中心）
	FSlateDrawElement::RelativeToElement,         // 旋转相对谁
	FinalColorAndOpacity
);
```

对比 P16.6.3 的 `MakeBox`（不旋转）——`MakeRotatedBox` 多了旋转参数。

### 5. `SetCanTick(false)`——禁用 Tick

```cpp
SetCanTick(false);
```

`SWidget::SetCanTick(false)`：**禁用 Slate 的 Tick**。因为画布用"活跃定时器"（`RegisterActiveTimer`）而不是 Tick。

> 对比 P16.6.3 的 `ComputeVolatility() = true`（每帧重绘）——这里用定时器驱动更新，更可控。

### 6. `Construct` 里预建 10 个箭头

```cpp
for (int32 i = 0; i < 10; ++i)
{
	TSharedRef<SActorCanvasArrowWidget> ArrowWidget = SNew(SActorCanvasArrowWidget, ActorCanvasArrowBrush);
	ArrowWidget->SetVisibility(EVisibility::Collapsed);
	
	ArrowChildren.AddSlot(MoveTemp(
		FArrowSlot::FSlotArguments(MakeUnique<FArrowSlot>())
		[
			ArrowWidget
		]
	));
}
```

**为什么预建 10 个？** 因为箭头是"钳制到屏幕边缘时"才用的，数量不确定但通常不多。预建 10 个避免运行时频繁创建（性能优化）。

`FArrowSlot::FSlotArguments(MakeUnique<FArrowSlot>())[ArrowWidget]`：Slate 的槽位构造语法（`[]` 里放子控件）。

### 7. `AsyncLoad` + `StartAsyncLoading`——FAsyncMixin 的异步加载

```cpp
AsyncLoad(IndicatorClass, [this, IndicatorPtr, IndicatorClass]() {
	// 加载完成后的回调
});
StartAsyncLoading();
```

`FAsyncMixin`（AsyncMixin 插件）提供：
- `AsyncLoad(软引用, 回调)`：注册一个异步加载任务。
- `StartAsyncLoading()`：**开始执行**所有注册的任务。

> 这是"先注册多个任务，再统一启动"的模式（可以并行加载多个）。

### 8. 异步回调里的**双重安全检查**

```cpp
if (UIndicatorDescriptor* Indicator = IndicatorPtr.Get())      // ① Descriptor 还活着吗
{
	if (!AllIndicators.Contains(Indicator))                     // ② 还在列表里吗
	{
		return;   // 异步期间被移除了，放弃
	}
	...
}
```

**为什么需要？** 因为异步加载可能耗时（几十毫秒），期间指示器可能已经被移除。不检查就会操作已删除的对象 → 崩溃。

> 这是**异步编程的必备防御**（P16.6.1 的 `GetInstigator() != nullptr` 也是同类思路）。

### 9. `IndicatorPool.GetOrCreateInstance(...)`——对象池取/建

```cpp
if (UUserWidget* IndicatorWidget = IndicatorPool.GetOrCreateInstance(TSubclassOf<UUserWidget>(IndicatorClass.Get())))
```

`FUserWidgetPool::GetOrCreateInstance(类)`：**从池里取一个该类的实例，没有就创建**。

对比 `CreateWidget`（每次新建）——对象池会复用已回收的实例。

### 10. `SAssignNew(Indicator->CanvasHost, SBox)[...]`——创建并赋值

```cpp
AddActorSlot(Indicator)
[
	SAssignNew(Indicator->CanvasHost, SBox)
	[
		IndicatorWidget->TakeWidget()
	]
];
```

`SAssignNew(变量, 类)`：**创建 Slate 控件并赋给变量**（等价于 `变量 = SNew(类)`）。

`SBox`：Slate 的"盒子"容器（包一层，用于尺寸控制）。

`IndicatorWidget->TakeWidget()`：把 `UUserWidget` 转成 `SWidget`（UMG → Slate 的桥）。

### 11. `AddActorSlot` 的 `FScopedWidgetSlotArguments`

```cpp
SActorCanvas::FScopedWidgetSlotArguments SActorCanvas::AddActorSlot(UIndicatorDescriptor* Indicator)
{
	TWeakPtr<SActorCanvas> WeakCanvas = SharedThis(this);
	return FScopedWidgetSlotArguments{ MakeUnique<FSlot>(Indicator), this->CanvasChildren, INDEX_NONE
		, [WeakCanvas](const FSlot*, int32)
		{
			if (TSharedPtr<SActorCanvas> Canvas = WeakCanvas.Pin())
			{
				Canvas->UpdateActiveTimer();
			}
		}};
}
```

`FScopedWidgetSlotArguments`：Slate 的"槽位参数"（用于 `AddSlot` 的返回值，支持 `[...]` 语法添加子控件）。

四个参数：
1. `MakeUnique<FSlot>(Indicator)`：新建槽位。
2. `this->CanvasChildren`：目标容器。
3. `INDEX_NONE`：插入位置（末尾）。
4. lambda：**槽位添加后的回调**（这里调 `UpdateActiveTimer`）。

`SharedThis(this)`：拿到 `this` 的 `TSharedPtr`（Slate 对象都是共享指针管理）。

### 12. `UpdateActiveTimer`——按需启停

```cpp
const bool NeedsTicks = AllIndicators.Num() > 0 || !IndicatorComponentPtr.IsValid();

if (NeedsTicks && !TickHandle.IsValid())
{
	TickHandle = RegisterActiveTimer(0, FWidgetActiveTimerDelegate::CreateSP(this, &SActorCanvas::UpdateCanvas));
}
```

**需要 tick 的条件**（任一）：
1. 有指示器（`AllIndicators.Num() > 0`）。
2. 还没拿到管理器组件（`!IndicatorComponentPtr.IsValid()`）——需要持续找。

`RegisterActiveTimer(0, 委托)`：注册活跃定时器（`0` = 每帧）。

> **注意**：这里**只注册不注销**——Lyra 没写"不需要时移除定时器"的逻辑（`TickHandle` 一旦有效就一直有效）。按铁律 23 照抄。
