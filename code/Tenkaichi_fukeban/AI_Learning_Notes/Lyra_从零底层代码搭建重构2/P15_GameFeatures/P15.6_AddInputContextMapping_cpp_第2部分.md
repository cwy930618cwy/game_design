// 在某个 world 里执行：注册 APlayerController 的扩展处理器。
void UGameFeatureAction_AddInputContextMapping::AddToWorld(const FWorldContext& WorldContext, const FGameFeatureStateChangeContext& ChangeContext)
{
	UWorld* World = WorldContext.World();
	UGameInstance* GameInstance = WorldContext.OwningGameInstance;
	FPerContextData& ActiveData = ContextData.FindOrAdd(ChangeContext);

	if ((GameInstance != nullptr) && (World != nullptr) && World->IsGameWorld())
	{
		if (UGameFrameworkComponentManager* ComponentManager = UGameInstance::GetSubsystem<UGameFrameworkComponentManager>(GameInstance))
		{
			// 注册"APlayerController 出现/消失"的扩展处理器
			UGameFrameworkComponentManager::FExtensionHandlerDelegate AddAbilitiesDelegate =
				UGameFrameworkComponentManager::FExtensionHandlerDelegate::CreateUObject(this, &ThisClass::HandleControllerExtension, ChangeContext);
			TSharedPtr<FComponentRequestHandle> ExtensionRequestHandle =
				ComponentManager->AddExtensionHandler(APlayerController::StaticClass(), AddAbilitiesDelegate);

			ActiveData.ExtensionRequestHandles.Add(ExtensionRequestHandle);
		}
	}
}

// 重置某个上下文的数据（清空句柄 + 移除所有已加的输入映射）。
void UGameFeatureAction_AddInputContextMapping::Reset(FPerContextData& ActiveData)
{
	ActiveData.ExtensionRequestHandles.Empty();

	while (!ActiveData.ControllersAddedTo.IsEmpty())
	{
		TWeakObjectPtr<APlayerController> ControllerPtr = ActiveData.ControllersAddedTo.Top();
		if (ControllerPtr.IsValid())
		{
			RemoveInputMapping(ControllerPtr.Get(), ActiveData);
		}
		else
		{
			// 控制器已销毁，直接弹出
			ActiveData.ControllersAddedTo.Pop();
		}
	}
}

// 玩家控制器扩展事件回调：按事件名决定加还是移除输入映射。
void UGameFeatureAction_AddInputContextMapping::HandleControllerExtension(AActor* Actor, FName EventName, FGameFeatureStateChangeContext ChangeContext)
{
	APlayerController* AsController = CastChecked<APlayerController>(Actor);
	FPerContextData& ActiveData = ContextData.FindOrAdd(ChangeContext);

	// TODO 为什么这段代码把控制器和本地玩家混着用？ControllersAddedTo 从没被修改过
	// 扩展/接收者被移除 → 移除输入映射
	if ((EventName == UGameFrameworkComponentManager::NAME_ExtensionRemoved) || (EventName == UGameFrameworkComponentManager::NAME_ReceiverRemoved))
	{
		RemoveInputMapping(AsController, ActiveData);
	}
	// 扩展被添加 / HeroComponent 准备好绑定输入 → 给本地玩家添加输入映射
	else if ((EventName == UGameFrameworkComponentManager::NAME_ExtensionAdded) || (EventName == UTenkaichiHeroComponent::NAME_BindInputsNow))
	{
		AddInputMappingForPlayer(AsController->GetLocalPlayer(), ActiveData);
	}
}

// 给本地玩家添加输入映射上下文（真正生效）。
void UGameFeatureAction_AddInputContextMapping::AddInputMappingForPlayer(UPlayer* Player, FPerContextData& ActiveData)
{
	if (ULocalPlayer* LocalPlayer = Cast<ULocalPlayer>(Player))
	{
		if (UEnhancedInputLocalPlayerSubsystem* InputSystem = LocalPlayer->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())
		{
			for (const FInputMappingContextAndPriority& Entry : InputMappings)
			{
				if (const UInputMappingContext* IMC = Entry.InputMapping.Get())
				{
					// 真正把 IMC 加到输入子系统（带优先级）
					InputSystem->AddMappingContext(IMC, Entry.Priority);
				}
			}
		}
		else
		{
			UE_LOG(LogGameFeatures, Error, TEXT("Failed to find `UEnhancedInputLocalPlayerSubsystem` for local player. Input mappings will not be added. Make sure you're set to use the EnhancedInput system via config file."));
		}
	}
}

// 移除输入映射。
void UGameFeatureAction_AddInputContextMapping::RemoveInputMapping(APlayerController* PlayerController, FPerContextData& ActiveData)
{
	if (ULocalPlayer* LocalPlayer = PlayerController->GetLocalPlayer())
	{
		if (UEnhancedInputLocalPlayerSubsystem* InputSystem = LocalPlayer->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())
		{
			for (const FInputMappingContextAndPriority& Entry : InputMappings)
			{
				if (const UInputMappingContext* IMC = Entry.InputMapping.Get())
				{
					InputSystem->RemoveMappingContext(IMC);
				}
			}
		}
	}

	ActiveData.ControllersAddedTo.Remove(PlayerController);
}

## 二、关键点讲解

### 1. `AddToWorld` 注册 `APlayerController`（不是 APawn）

```cpp
ComponentManager->AddExtensionHandler(APlayerController::StaticClass(), AddAbilitiesDelegate);
```

对比：
- P15.5 `AddInputBinding`：注册 `APawn`（输入绑定跟 Pawn 走）。
- P15.6 `AddInputContextMapping`：注册 `APlayerController`（IMC 跟本地玩家/控制器走，因为 `UEnhancedInputLocalPlayerSubsystem` 挂在 LocalPlayer 上）。

### 2. `HandleControllerExtension` 的 `AsController->GetLocalPlayer()`

```cpp
AddInputMappingForPlayer(AsController->GetLocalPlayer(), ActiveData);
```

从控制器拿 `LocalPlayer` 传进去。`GetLocalPlayer()` 可能返回 null（远端玩家控制器），所以 `AddInputMappingForPlayer` 里再 `Cast<ULocalPlayer>` 检查。

### 3. `AddInputMappingForPlayer`——真正生效

```cpp
if (ULocalPlayer* LocalPlayer = Cast<ULocalPlayer>(Player))
{
	if (UEnhancedInputLocalPlayerSubsystem* InputSystem = LocalPlayer->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())
	{
		for (const FInputMappingContextAndPriority& Entry : InputMappings)
		{
			if (const UInputMappingContext* IMC = Entry.InputMapping.Get())
			{
				InputSystem->AddMappingContext(IMC, Entry.Priority);   // ← 真正加
			}
		}
	}
}
```

`AddMappingContext(IMC, Priority)`：把 IMC 加进输入子系统，**带优先级**（数值大优先）。这是"按键真正生效"的一步。

> 对比第 1 部分的 `Settings->RegisterInputMappingContext(IMC)`——那是"注册到设置界面"，这才是"加进运行时输入栈"。

### 4. `RemoveInputMapping` 接收 `APlayerController*`（不是 UPlayer*）

```cpp
void RemoveInputMapping(APlayerController* PlayerController, FPerContextData& ActiveData)
{
	if (ULocalPlayer* LocalPlayer = PlayerController->GetLocalPlayer())
	...
	ActiveData.ControllersAddedTo.Remove(PlayerController);
}
```

不对称的地方：Add 收 `UPlayer*`，Remove 收 `APlayerController*`——因为 Add 是从 `GetLocalPlayer()` 传进来的，Remove 是从 `ControllersAddedTo` 列表（存的是控制器）里取的。

### 5. Lyra 源码里的 TODO 注释

```cpp
// TODO Why does this code mix and match controllers and local players? ControllersAddedTo is never modified
```

Lyra 自己留的 TODO：**吐槽代码混用 controller 和 local player，而且 `ControllersAddedTo` 从来没被 Add 过**（只 Remove）。

> 这是 Lyra 原源码的真实问题（`AddInputMappingForPlayer` 里没有 `ControllersAddedTo.AddUnique`）。按铁律 23 **照抄保留，不修**。

### 6. `CastChecked<APlayerController>` vs `Cast<ULocalPlayer>`

- `CastChecked`：必须是这个类型，不是就崩（注册给 APlayerController，必须对）。
- `Cast`：可能不是，不是就返回 null（LocalPlayer 可能为空）。

### 7. 链接级挂起

`UTenkaichiHeroComponent::NAME_BindInputsNow` 常量定义在 `TenkaichiHeroComponent.cpp`（备忘录 ⑦ 挂起）。`static const FName` 的**定义**在 `.cpp` 里，所以链接时报未定义符号，等 HeroComponent.cpp 解开后消失。
