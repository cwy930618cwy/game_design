```cpp
void UGameFeatureAction_AddAbilities::OnGameFeatureActivating(FGameFeatureActivatingContext& Context)
{
	FPerContextData& ActiveData = ContextData.FindOrAdd(Context);

	// 如果上下文数据不是空的（异常情况），先重置
	if (!ensureAlways(ActiveData.ActiveExtensions.IsEmpty()) ||
		!ensureAlways(ActiveData.ComponentRequests.IsEmpty()))
	{
		Reset(ActiveData);
	}
	Super::OnGameFeatureActivating(Context);
}

{

	if (ensure(ActiveData))
	{
		Reset(*ActiveData);
	}
}

// 在某个 world 里执行：为每个配置项注册"类扩展处理器"。
void UGameFeatureAction_AddAbilities::AddToWorld(const FWorldContext& WorldContext, const FGameFeatureStateChangeContext& ChangeContext)
{
	UWorld* World = WorldContext.World();
	UGameInstance* GameInstance = WorldContext.OwningGameInstance;
	FPerContextData& ActiveData = ContextData.FindOrAdd(ChangeContext);

	if ((GameInstance != nullptr) && (World != nullptr) && World->IsGameWorld())
	{
		if (UGameFrameworkComponentManager* ComponentMan = UGameInstance::GetSubsystem<UGameFrameworkComponentManager>(GameInstance))
		{
			int32 EntryIndex = 0;
			for (const FGameFeatureAbilitiesEntry& Entry : AbilitiesList)
			{
				if (!Entry.ActorClass.IsNull())
				{
					// 创建"Actor 扩展"处理器：这类 Actor 出现时回调 HandleActorExtension
					UGameFrameworkComponentManager::FExtensionHandlerDelegate AddAbilitiesDelegate = UGameFrameworkComponentManager::FExtensionHandlerDelegate::CreateUObject(
						this, &UGameFeatureAction_AddAbilities::HandleActorExtension, EntryIndex, ChangeContext);
					TSharedPtr<FComponentRequestHandle> ExtensionRequestHandle = ComponentMan->AddExtensionHandler(Entry.ActorClass, AddAbilitiesDelegate);

					ActiveData.ComponentRequests.Add(ExtensionRequestHandle);
					EntryIndex++;
				}
			}
		}
	}
}

// 重置某个上下文的数据（移除所有已授予扩展 + 清空组件请求）。
void UGameFeatureAction_AddAbilities::Reset(FPerContextData& ActiveData)
{
	while (!ActiveData.ActiveExtensions.IsEmpty())
	{
		auto ExtensionIt = ActiveData.ActiveExtensions.CreateIterator();
		RemoveActorAbilities(ExtensionIt->Key, ActiveData);
	}

	ActiveData.ComponentRequests.Empty();
}

// Actor 扩展事件回调：按事件名决定授予还是移除。
void UGameFeatureAction_AddAbilities::HandleActorExtension(AActor* Actor, FName EventName, int32 EntryIndex, FGameFeatureStateChangeContext ChangeContext)
{
	FPerContextData* ActiveData = ContextData.Find(ChangeContext);
	if (AbilitiesList.IsValidIndex(EntryIndex) && ActiveData)
	{
		const FGameFeatureAbilitiesEntry& Entry = AbilitiesList[EntryIndex];
		// 扩展/接收者被移除 → 移除能力
		if ((EventName == UGameFrameworkComponentManager::NAME_ExtensionRemoved) || (EventName == UGameFrameworkComponentManager::NAME_ReceiverRemoved))
		{
			RemoveActorAbilities(Actor, *ActiveData);
		}
		// 扩展被添加 / 能力就绪 → 授予能力
		else if ((EventName == UGameFrameworkComponentManager::NAME_ExtensionAdded) || (EventName == ATenkaichiPlayerState::NAME_TenkaichiAbilityReady))
		{
			AddActorAbilities(Actor, Entry, *ActiveData);
		}
	}
}
```

## 二、第一部分讲解（激活/停用/校验/注册）

### 1. `OnGameFeatureActivating`——防御性重置

```cpp
FPerContextData& ActiveData = ContextData.FindOrAdd(Context);

if (!ensureAlways(ActiveData.ActiveExtensions.IsEmpty()) ||
	!ensureAlways(ActiveData.ComponentRequests.IsEmpty()))
{
	Reset(ActiveData);
}
Super::OnGameFeatureActivating(Context);
```

`ensureAlways`：**每次都检查**（不像 `ensure` 只报一次）。如果发现上次的残留（不该有），就先 `Reset` 清干净——防御性编程。

`Super::OnGameFeatureActivating(Context)`：调 P15.1 基类，由它遍历 world 调 `AddToWorld`。

### 2. `OnGameFeatureDeactivating`——先基类后重置

```cpp
Super::OnGameFeatureDeactivating(Context);   // 先解绑 GameInstance 启动回调
FPerContextData* ActiveData = ContextData.Find(Context);
if (ensure(ActiveData))
{
	Reset(*ActiveData);   // 再移除所有已授予扩展
}
```

顺序很重要：先让基类清理它的注册，再清理自己的扩展。

### 3. `IsDataValid`——4 层校验

检查：① `ActorClass` 非空；② 至少配了一样东西；③ 每个能力非空；④ 每个属性集非空；⑤ 每个能力集非空。任一出错就 `Result = Invalid`。

> 注意源码末尾有两个 `return`（`return Result;` 之后又有 `return EDataValidationResult::NotValidated;`）——这是 Lyra 原源码的**冗余代码**（第二条 return 永远执行不到）。按铁律 23 **照抄保留，不删**。

### 4. `AddToWorld`——注册扩展处理器

```cpp
UGameFrameworkComponentManager::FExtensionHandlerDelegate AddAbilitiesDelegate = UGameFrameworkComponentManager::FExtensionHandlerDelegate::CreateUObject(
	this, &UGameFeatureAction_AddAbilities::HandleActorExtension, EntryIndex, ChangeContext);
TSharedPtr<FComponentRequestHandle> ExtensionRequestHandle = ComponentMan->AddExtensionHandler(Entry.ActorClass, AddAbilitiesDelegate);
```

- `AddExtensionHandler(类, 委托)`：向 ComponentManager 注册"这个类（及子类）的 Actor 出现/消失时通知我"。
- `CreateUObject(this, &HandleActorExtension, EntryIndex, ChangeContext)`：绑定成员函数 + **两个附加参数**（配置索引、上下文）。
- 返回的 `FComponentRequestHandle` 存起来（取消注册用）。

### 5. `Reset`——while 循环移空

```cpp
while (!ActiveData.ActiveExtensions.IsEmpty())
{
	auto ExtensionIt = ActiveData.ActiveExtensions.CreateIterator();
	RemoveActorAbilities(ExtensionIt->Key, ActiveData);   // 内部会 Remove，所以每次重取迭代器
}
```

因为 `RemoveActorAbilities` 会从 `ActiveExtensions` 移除元素，所以每轮重新取迭代器（不能缓存）。

### 6. `HandleActorExtension`——事件分发

```cpp
if (EventName == NAME_ExtensionRemoved || EventName == NAME_ReceiverRemoved)  → RemoveActorAbilities
else if (EventName == NAME_ExtensionAdded || EventName == ATenkaichiPlayerState::NAME_TenkaichiAbilityReady) → AddActorAbilities
```

- `NAME_ExtensionAdded`：Actor 的扩展组件就绪 → 授予。
- `ATenkaichiPlayerState::NAME_TenkaichiAbilityReady`：**PlayerState 的能力系统就绪**（P12 定义的自定义事件，第 94 行）→ 授予。这是玩家类特有的时机（等 ASC 建好）。
- `NAME_ExtensionRemoved` / `NAME_ReceiverRemoved`：移除。

## 三、完整代码（接上，第 161~300 行）

```cpp
// 给 Actor 授予能力 / 属性 / 能力集。
void UGameFeatureAction_AddAbilities::AddActorAbilities(AActor* Actor, const FGameFeatureAbilitiesEntry& AbilitiesEntry, FPerContextData& ActiveData)
{
	check(Actor);
	// 只有服务器权威才授予（能力是服务器权威的）
	if (!Actor->HasAuthority())
	{
		return;
	}

	// 已经加过扩展就提前退出
	if (ActiveData.ActiveExtensions.Find(Actor) != nullptr)
	{
		return;	
	}

	// 找（或请求添加）这个 Actor 的 ASC 组件
	if (UAbilitySystemComponent* AbilitySystemComponent = FindOrAddComponentForActor<UAbilitySystemComponent>(Actor, AbilitiesEntry, ActiveData))
	{
		FActorExtensions AddedExtensions;
		AddedExtensions.Abilities.Reserve(AbilitiesEntry.GrantedAbilities.Num());
		AddedExtensions.Attributes.Reserve(AbilitiesEntry.GrantedAttributes.Num());
		AddedExtensions.AbilitySetHandles.Reserve(AbilitiesEntry.GrantedAbilitySets.Num());

		// ① 授予单个能力
		for (const FTenkaichiAbilityGrant& Ability : AbilitiesEntry.GrantedAbilities)
		{
			if (!Ability.AbilityType.IsNull())
			{
				FGameplayAbilitySpec NewAbilitySpec(Ability.AbilityType.LoadSynchronous());
				FGameplayAbilitySpecHandle AbilityHandle = AbilitySystemComponent->GiveAbility(NewAbilitySpec);

				AddedExtensions.Abilities.Add(AbilityHandle);
			}
		}

		// ② 授予属性集
		for (const FTenkaichiAttributeSetGrant& Attributes : AbilitiesEntry.GrantedAttributes)
		{
			if (!Attributes.AttributeSetType.IsNull())
			{
				TSubclassOf<UAttributeSet> SetType = Attributes.AttributeSetType.LoadSynchronous();
				if (SetType)
				{
					UAttributeSet* NewSet = NewObject<UAttributeSet>(AbilitySystemComponent->GetOwner(), SetType);
					// 有初始化数据表就用它初始化属性
					if (!Attributes.InitializationData.IsNull())
					{
						UDataTable* InitData = Attributes.InitializationData.LoadSynchronous();
						if (InitData)
						{
							NewSet->InitFromMetaDataTable(InitData);
						}
					}

					AddedExtensions.Attributes.Add(NewSet);
					AbilitySystemComponent->AddAttributeSetSubobject(NewSet);
				}
			}
		}

		// ③ 授予能力集
		UTenkaichiAbilitySystemComponent* TenkaichiASC = CastChecked<UTenkaichiAbilitySystemComponent>(AbilitySystemComponent);
		for (const TSoftObjectPtr<const UTenkaichiAbilitySet>& SetPtr : AbilitiesEntry.GrantedAbilitySets)
		{
			if (const UTenkaichiAbilitySet* Set = SetPtr.Get())
			{
				Set->GiveToAbilitySystem(TenkaichiASC, &AddedExtensions.AbilitySetHandles.AddDefaulted_GetRef());
			}
		}

		// 记到收据里
		ActiveData.ActiveExtensions.Add(Actor, AddedExtensions);
	}
	else
	{
		UE_LOG(LogGameFeatures, Error, TEXT("Failed to find/add an ability component to '%s'. Abilities will not be granted."), *Actor->GetPathName());
	}
}

// 从 Actor 移除已授予的能力 / 属性 / 能力集。
void UGameFeatureAction_AddAbilities::RemoveActorAbilities(AActor* Actor, FPerContextData& ActiveData)
{
	if (FActorExtensions* ActorExtensions = ActiveData.ActiveExtensions.Find(Actor))
	{
		if (UAbilitySystemComponent* AbilitySystemComponent = Actor->FindComponentByClass<UAbilitySystemComponent>())
		{
			// ① 移除属性集
			for (UAttributeSet* AttribSetInstance : ActorExtensions->Attributes)
			{
				AbilitySystemComponent->RemoveSpawnedAttribute(AttribSetInstance);
			}

			// ② 移除能力（标记为"结束时移除"）
			for (FGameplayAbilitySpecHandle AbilityHandle : ActorExtensions->Abilities)
			{
				AbilitySystemComponent->SetRemoveAbilityOnEnd(AbilityHandle);
			}

			// ③ 收回能力集
			UTenkaichiAbilitySystemComponent* TenkaichiASC = CastChecked<UTenkaichiAbilitySystemComponent>(AbilitySystemComponent);
			for (FTenkaichiAbilitySet_GrantedHandles& SetHandle : ActorExtensions->AbilitySetHandles)
			{
				SetHandle.TakeFromAbilitySystem(TenkaichiASC);
			}
		}

		ActiveData.ActiveExtensions.Remove(Actor);
	}
}

// 按组件类型找，找不到就通过 ComponentManager 请求添加（引用计数）。
UActorComponent* UGameFeatureAction_AddAbilities::FindOrAddComponentForActor(UClass* ComponentType, AActor* Actor, const FGameFeatureAbilitiesEntry& AbilitiesEntry, FPerContextData& ActiveData)
{
	UActorComponent* Component = Actor->FindComponentByClass(ComponentType);

	bool bMakeComponentRequest = (Component == nullptr);
	if (Component)
	{
		// 检查这个组件是不是由另一个 UGameFrameworkComponentManager 请求创建的。
		// `Native` 是动态添加组件的 CreationMethod 默认值。
		if (Component->CreationMethod == EComponentCreationMethod::Native)
		{
			// 尝试区分"真正的原生组件"和"由 GameFrameworkComponent 系统创建的组件"。
			// 如果它来自 UGameFrameworkComponentManager，那我们需要再请求一次（请求是引用计数的）。
			UObject* ComponentArchetype = Component->GetArchetype();
			bMakeComponentRequest = ComponentArchetype->HasAnyFlags(RF_ClassDefaultObject);
		}
	}

	if (bMakeComponentRequest)
	{
		UWorld* World = Actor->GetWorld();
		UGameInstance* GameInstance = World->GetGameInstance();

		if (UGameFrameworkComponentManager* ComponentMan = UGameInstance::GetSubsystem<UGameFrameworkComponentManager>(GameInstance))
		{
			TSharedPtr<FComponentRequestHandle> RequestHandle = ComponentMan->AddComponentRequest(AbilitiesEntry.ActorClass, ComponentType);
			ActiveData.ComponentRequests.Add(RequestHandle);
		}

		if (!Component)
		{
			Component = Actor->FindComponentByClass(ComponentType);
			ensureAlways(Component);
		}
	}

	return Component;
}
```

## 四、第二部分讲解（授予 / 移除 / 找组件）

### 1. `AddActorAbilities`——授予的三段

```cpp
if (!Actor->HasAuthority()) return;                     // ① 只服务器权威
if (ActiveData.ActiveExtensions.Find(Actor) != nullptr) return;   // ② 已加过就跳过
if (UAbilitySystemComponent* ASC = FindOrAddComponentForActor<UAbilitySystemComponent>(...))  // ③ 找/请求 ASC
```

**为什么只服务器权威？** GAS 的能力授予是**服务器权威**的（服务器决定谁有什么能力，再复制给客户端）。客户端自己 `GiveAbility` 无效。

**三个授予循环：**
1. **能力**：`FGameplayAbilitySpec(AbilityType.LoadSynchronous())` → `ASC->GiveAbility(spec)` → 返回 `FGameplayAbilitySpecHandle` 记下来。
2. **属性集**：`NewObject<UAttributeSet>(Owner, SetType)` → 可选 `InitFromMetaDataTable(InitData)` → `ASC->AddAttributeSetSubobject(NewSet)`。
3. **能力集**：`Set->GiveToAbilitySystem(TenkaichiASC, &AddedExtensions.AbilitySetHandles.AddDefaulted_GetRef())`——一次性给一套（P05 已教 `GiveToAbilitySystem`）。

**`AddDefaulted_GetRef()`**：往数组加一个默认元素并返回它的引用——一步完成"扩容 + 取引用"。

### 2. `RemoveActorAbilities`——精确回收

```cpp
for (UAttributeSet* AttribSetInstance : ActorExtensions->Attributes)
	ASC->RemoveSpawnedAttribute(AttribSetInstance);      // 属性集直接移除

for (FGameplayAbilitySpecHandle AbilityHandle : ActorExtensions->Abilities)
	ASC->SetRemoveAbilityOnEnd(AbilityHandle);           // 能力标记"结束时移除"

for (FTenkaichiAbilitySet_GrantedHandles& SetHandle : ActorExtensions->AbilitySetHandles)
	SetHandle.TakeFromAbilitySystem(TenkaichiASC);       // 能力集统一收回
```

**关键差异**：能力用 `SetRemoveAbilityOnEnd`（不是立即移除）——因为能力**可能正在执行**，立即移除会崩。标记"执行结束再移除"更安全。属性集/能力集可以直接移除。

### 3. `FindOrAddComponentForActor`——引用计数请求

```cpp
bool bMakeComponentRequest = (Component == nullptr);
if (Component && Component->CreationMethod == EComponentCreationMethod::Native)
{
	UObject* ComponentArchetype = Component->GetArchetype();
	bMakeComponentRequest = ComponentArchetype->HasAnyFlags(RF_ClassDefaultObject);
}
```

这段判断很微妙：
- 组件不存在 → 要请求。
- 组件存在且是 `Native` 创建方式 → 检查它是不是**由 ComponentManager 动态加的**（用"原型对象是不是 CDO"判断）。如果是动态加的，**还要再请求一次**——因为 `AddComponentRequest` 是**引用计数**的（多个 Action 都请求同一个组件，各自持有一个句柄，都释放才真正移除）。

```cpp
TSharedPtr<FComponentRequestHandle> RequestHandle = ComponentMan->AddComponentRequest(AbilitiesEntry.ActorClass, ComponentType);
```

`AddComponentRequest(类, 组件类型)`：请求"给这个类的 Actor 加这个组件"。返回句柄（计数 +1）。

### 4. 为什么 `ensureAlways` 在请求后

```cpp
if (!Component)
{
	Component = Actor->FindComponentByClass(ComponentType);
	ensureAlways(Component);
}
```

请求后应该就有组件了（ComponentManager 同步加的）。如果还是找不到，说明配置有问题（比如 ASC 组件不能动态加），`ensureAlways` 报错。
